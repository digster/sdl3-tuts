Recipe 25 — HDR & Tone Mapping

Render in high dynamic range, then map it down for the display. An RGBA16F framebuffer keeps colours above 1.0 intact; a fullscreen pass applies exposure, a tone-mapping curve and gamma — the bgfx counterpart to 3D recipe 23.

What you'll learn

  • Creating a floating-point (RGBA16F) render target framebuffer in bgfx.
  • Why clamping HDR to [0,1] blows out highlights and tone mapping doesn't.
  • Exposure → Reinhard / ACES → gamma, as one fullscreen fragment shader.
  • Reusing the recipe-15 transient fullscreen triangle for the post pass.

Prerequisites

Recipe 15 — Post-Processing & Bloom for the HDR framebuffer and the fullscreen-triangle pass, and recipe 10 for views. This is the bgfx twin of 3D recipe 23 — same maths, bgfx's framebuffer + view API instead of SDL_GPU's.

Walkthrough

A floating-point framebuffer

The backbuffer stores 8-bit colour clamped to [0,1]. To keep bright highlights, the scene renders into an RGBA16F target where values can exceed 1.0. The depth attachment rides along as write-only; only the colour needs to be sampleable afterwards. 

uint64_t rt = BGFX_TEXTURE_RT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
bgfx::TextureHandle att[2] = {
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::RGBA16F, rt),
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::D24S8, BGFX_TEXTURE_RT_WRITE_ONLY),
};
st->hdrFb = bgfx::createFrameBuffer(2, att, /*destroyTextures=*/true);

Clamping is not tone mapping

Clamping flattens every bright region to solid white. A tone-mapping curve compresses the high end smoothly, preserving the impression of brightness. This is the same Reinhard / ACES pair recipe 23 used, here in bgfx's shader dialect. 

# fs_tonemap.sc — exposure, then curve, then gamma
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_hdr, 0);
uniform vec4 u_tone;                 // x = exposure, y = mode (0 clamp,1 reinhard,2 aces)
vec3 reinhard(vec3 c) { return c / (1.0 + c); }
vec3 aces(vec3 x) {
    return clamp((x*(2.51*x+0.03)) / (x*(2.43*x+0.59)+0.14), 0.0, 1.0);
}
void main() {
    vec3 c = texture2D(s_hdr, v_texcoord0).rgb * u_tone.x;          // 1. exposure
    vec3 mapped = u_tone.y < 0.5 ? clamp(c, 0.0, 1.0)              // 2. curve
                : u_tone.y < 1.5 ? reinhard(c)
                :                  aces(c);
    mapped = pow(mapped, vec3_splat(1.0 / 2.2));                    // 3. gamma
    gl_FragColor = vec4(mapped, 1.0);
}
Order matters — and it's the last thing you do. Do all lighting and blending in linear HDR, then tone-map and gamma-encode once, just before the pixel reaches the screen. bgfx writes the swapchain in the second view, so this fullscreen pass is naturally that final step.

Complete listing

An over-bright cube drawn into an HDR framebuffer, then tone-mapped to the window. Press Space to cycle clamp → Reinhard → ACES, and ↑/↓ for exposure. Native-handle / loadProgram helpers identical to recipe 03 (elided). Shaders: vs_scene / fs_scene, vs_fullscreen / fs_tonemap. 

/*
 * bgfx/25-hdr-tone-mapping.cpp — HDR RGBA16F scene target tone-mapped
 * (clamp/Reinhard/ACES) with exposure + gamma in a fullscreen pass.
 * Native-handle + loadProgram helpers identical to recipe 03 (elided).
 * Shaders: vs_scene/fs_scene, vs_fullscreen/fs_tonemap.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);

struct PCVertex   { float px, py, pz; float r, g, b, a; };
struct PosUvVertex { float x, y; float u, v; };

// Over-bright cube: several channels well above 1.0 so tone mapping has work to do.
static const PCVertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 4.0f,1.0f,0.4f,1 },{  0.5f, 0.5f,-0.5f, 4.0f,1.0f,0.4f,1 },{  0.5f, 0.5f, 0.5f, 4.0f,1.0f,0.4f,1 },{  0.5f,-0.5f, 0.5f, 4.0f,1.0f,0.4f,1 },
    { -0.5f,-0.5f, 0.5f, 0.3f,2.5f,3.5f,1 },{ -0.5f, 0.5f, 0.5f, 0.3f,2.5f,3.5f,1 },{ -0.5f, 0.5f,-0.5f, 0.3f,2.5f,3.5f,1 },{ -0.5f,-0.5f,-0.5f, 0.3f,2.5f,3.5f,1 },
    { -0.5f, 0.5f,-0.5f, 3.0f,3.0f,0.6f,1 },{ -0.5f, 0.5f, 0.5f, 3.0f,3.0f,0.6f,1 },{  0.5f, 0.5f, 0.5f, 3.0f,3.0f,0.6f,1 },{  0.5f, 0.5f,-0.5f, 3.0f,3.0f,0.6f,1 },
    { -0.5f,-0.5f, 0.5f, 0.2f,0.4f,0.5f,1 },{ -0.5f,-0.5f,-0.5f, 0.2f,0.4f,0.5f,1 },{  0.5f,-0.5f,-0.5f, 0.2f,0.4f,0.5f,1 },{  0.5f,-0.5f, 0.5f, 0.2f,0.4f,0.5f,1 },
    { -0.5f,-0.5f, 0.5f, 0.4f,0.8f,4.0f,1 },{  0.5f,-0.5f, 0.5f, 0.4f,0.8f,4.0f,1 },{  0.5f, 0.5f, 0.5f, 0.4f,0.8f,4.0f,1 },{ -0.5f, 0.5f, 0.5f, 0.4f,0.8f,4.0f,1 },
    {  0.5f,-0.5f,-0.5f, 3.5f,0.5f,0.5f,1 },{ -0.5f,-0.5f,-0.5f, 3.5f,0.5f,0.5f,1 },{ -0.5f, 0.5f,-0.5f, 3.5f,0.5f,0.5f,1 },{  0.5f, 0.5f,-0.5f, 3.5f,0.5f,0.5f,1 },
};
static const uint16_t kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,

struct AppState {
    SDL_Window *window = nullptr; int width = 1280, height = 720, fbW = 0, fbH = 0;
    bgfx::VertexLayout layout, quadLayout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::FrameBufferHandle  hdrFb = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle scene = BGFX_INVALID_HANDLE, tonemap = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle s_hdr = BGFX_INVALID_HANDLE, u_tone = BGFX_INVALID_HANDLE;
    float exposure = 1.0f; int mode = 2;              // start on ACES
    Uint64 startNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

static void makeHDR(AppState *st, int w, int h)
{
    if (bgfx::isValid(st->hdrFb)) bgfx::destroy(st->hdrFb);
    uint64_t rt = BGFX_TEXTURE_RT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
    bgfx::TextureHandle att[2] = {
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA16F, rt),
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::D24S8, BGFX_TEXTURE_RT_WRITE_ONLY),
    };
    st->hdrFb = bgfx::createFrameBuffer(2, att, true);
    st->fbW = w; st->fbH = h;
}

// Clip-space fullscreen triangle (recipe 15); V flipped on top-left-origin backends.
static void screenTri(AppState *st)
{
    if (bgfx::getAvailTransientVertexBuffer(3, st->quadLayout) != 3) return;
    bgfx::TransientVertexBuffer tvb;
    bgfx::allocTransientVertexBuffer(&tvb, 3, st->quadLayout);
    PosUvVertex *v = (PosUvVertex *)tvb.data;
    bool flip = !bgfx::getCaps()->originBottomLeft;
    float vb = flip ? 1.0f : 0.0f, vt = flip ? -1.0f : 2.0f;
    v[0] = { -1.0f, -1.0f, 0.0f, vb };
    v[1] = {  3.0f, -1.0f, 2.0f, vb };
    v[2] = { -1.0f,  3.0f, 0.0f, vt };
    bgfx::setVertexBuffer(0, &tvb);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx HDR", "1.0", "com.example.sdl3.bgfx.hdr");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — HDR & Tone Mapping", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Color0,   4, bgfx::AttribType::Float).end();
    st->quadLayout.begin()
        .add(bgfx::Attrib::Position,  2, bgfx::AttribType::Float)
        .add(bgfx::Attrib::TexCoord0, 2, bgfx::AttribType::Float).end();
    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kIdx, sizeof(kIdx)));
    st->scene   = loadProgram("vs_scene",      "fs_scene");
    st->tonemap = loadProgram("vs_fullscreen", "fs_tonemap");
    st->s_hdr  = bgfx::createUniform("s_hdr",  bgfx::UniformType::Sampler);
    st->u_tone = bgfx::createUniform("u_tone", bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->scene) || !bgfx::isValid(st->tonemap)) return fail("loadProgram");

    makeHDR(st, st->width, st->height);
    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE) st->mode = (st->mode + 1) % 3;
        if (e->key.scancode == SDL_SCANCODE_UP)   st->exposure = SDL_min(8.0f, st->exposure * 1.25f);
        if (e->key.scancode == SDL_SCANCODE_DOWN) st->exposure = SDL_max(0.05f, st->exposure / 1.25f);
    }
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
        makeHDR(st, st->width, st->height);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    // --- view 0: scene into the HDR framebuffer ---
    bgfx::setViewFrameBuffer(0, st->hdrFb);
    bgfx::setViewRect(0, 0, 0, (uint16_t)st->fbW, (uint16_t)st->fbH);
    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x00000200, 1.0f, 0);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 0, -3.2f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 60.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, view, proj);

    float mtx[16], rx[16], ry[16];
    bx::mtxRotateX(rx, t * 0.4f); bx::mtxRotateY(ry, t * 0.6f); bx::mtxMul(mtx, rx, ry);
    bgfx::setTransform(mtx);
    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(0, st->scene);

    // --- view 1: tone-map the HDR target to the backbuffer ---
    bgfx::setViewFrameBuffer(1, BGFX_INVALID_HANDLE);
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    bgfx::setViewClear(1, BGFX_CLEAR_COLOR, 0x000000ff, 1.0f, 0);
    float tone[4] = { st->exposure, (float)st->mode, 0, 0 };
    bgfx::setUniform(st->u_tone, tone);
    bgfx::setTexture(0, st->s_hdr, bgfx::getTexture(st->hdrFb, 0));
    screenTri(st);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);
    bgfx::submit(1, st->tonemap);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (bgfx::isValid(st->u_tone))  bgfx::destroy(st->u_tone);
    if (bgfx::isValid(st->s_hdr))   bgfx::destroy(st->s_hdr);
    if (bgfx::isValid(st->hdrFb))   bgfx::destroy(st->hdrFb);
    if (bgfx::isValid(st->ibh))     bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))     bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->tonemap)) bgfx::destroy(st->tonemap);
    if (bgfx::isValid(st->scene))   bgfx::destroy(st->scene);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# vs_scene.sc / fs_scene.sc — pass the over-bright vertex colour straight through
# (vs)
$input  a_position, a_color0
$output v_color0
#include "bgfx_shader.sh"
void main() {
    gl_Position = mul(u_modelViewProj, vec4(a_position, 1.0));
    v_color0    = a_color0;          // raw HDR colour, may exceed 1.0
}
# (fs)
$input v_color0
#include "bgfx_shader.sh"
void main() { gl_FragColor = v_color0; }

# vs_fullscreen.sc / fs_tonemap.sc — the tone-map pass
# (vs)
$input  a_position, a_texcoord0
$output v_texcoord0
#include "bgfx_shader.sh"
void main() {
    gl_Position = vec4(a_position, 0.0, 1.0);
    v_texcoord0 = a_texcoord0;
}
# (fs)
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_hdr, 0);
uniform vec4 u_tone;                 // x = exposure, y = mode
vec3 reinhard(vec3 c) { return c / (1.0 + c); }
vec3 aces(vec3 x) {
    return clamp((x*(2.51*x+0.03)) / (x*(2.43*x+0.59)+0.14), 0.0, 1.0);
}
void main() {
    vec3 c = texture2D(s_hdr, v_texcoord0).rgb * u_tone.x;
    vec3 mapped = u_tone.y < 0.5 ? clamp(c, 0.0, 1.0)
                : u_tone.y < 1.5 ? reinhard(c)
                :                  aces(c);
    mapped = pow(mapped, vec3_splat(1.0 / 2.2));
    gl_FragColor = vec4(mapped, 1.0);
}

Try it yourself

  1. Remove the gamma line and watch the picture turn muddy — proof the 1/2.2 encode is doing real work.
  2. Chain recipe 15's bloom before this pass: composite scene + bloom into the HDR target, then tone-map once at the end.
  3. Add the Uncharted-2 (Hable) operator as a fourth mode and compare its shoulder against ACES.
  4. Auto-exposure: downsample the HDR target to 1×1 for average luminance (bgfx blit + mips) and drift exposure toward a target.
