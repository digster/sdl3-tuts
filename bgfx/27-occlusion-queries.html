Recipe 27 — Occlusion Queries

Ask the GPU what it can actually see. An occlusion query counts how many pixels of a draw survived the depth test; setCondition then lets the next frame skip drawing anything the query found hidden — a bgfx-native feature with no SDL_GPU equivalent.

What you'll learn

  • bgfx::createOcclusionQuery and submitting a draw tagged with one.
  • Reading last frame's result with bgfx::getResult (Visible / Invisible / NoResult).
  • Conditional rendering with setCondition — draw only if the query says visible.
  • Sequential view mode so the occluder is drawn before the things it hides, and the BGFX_CAPS_OCCLUSION_QUERY gate.

Prerequisites

Recipe 08 — Depth & Render State for the depth test the query rides on, and recipe 02 for the dbgTextPrintf overlay used to report the counts.

Walkthrough

What a query measures

An occlusion query wraps a draw and reports how many of its fragments passed the depth test. If a cube is entirely behind a wall that was drawn first, zero fragments survive — the query comes back Invisible. Results have a frame or two of latency (the GPU runs ahead of the CPU), so you read the result of a previous submission. 

// Create one query per testable object, once.
for (int i = 0; i < NUM_OBJ; ++i) st->query[i] = bgfx::createOcclusionQuery();

// Tag a draw with its query — this both renders and measures it.
bgfx::setTransform(model);
bgfx::setVertexBuffer(0, st->vbh);
bgfx::setIndexBuffer(st->ibh);
bgfx::setState(BGFX_STATE_DEFAULT);
bgfx::submit(0, st->prog, st->query[i]);            // submit-with-query overload

Draw the occluder first

For the query to mean "hidden behind the wall", the wall's depth must already be in the buffer when the cube is tested. bgfx sorts draws within a view by default, so force submission order with setViewMode(0, Sequential) and submit the wall before the objects — the same ordering trick the stencil recipe used. 

bgfx::setViewMode(0, bgfx::ViewMode::Sequential);   // preserve submit order
// ... draw the occluder wall (writes depth) ...
// ... then draw each object tagged with its query ...

Read results, or gate a draw

Two ways to use the result. getResult hands you last frame's outcome to tally or drive your own logic. setCondition goes further: it makes a subsequent submit execute on the GPU only if the query was visible — true GPU-side conditional rendering, so the skipped work never happens. 

int32_t px = 0;
bgfx::OcclusionQueryResult::Enum r = bgfx::getResult(st->query[i], &px);
if (r == bgfx::OcclusionQueryResult::Visible)   ++visible;
if (r == bgfx::OcclusionQueryResult::Invisible) ++hidden;

// Conditional: this marker draw runs only if the object was visible last frame.
bgfx::setCondition(st->query[i], true);
bgfx::submit(0, st->prog);
Check the cap. Occlusion queries need bgfx::getCaps()->supported & BGFX_CAPS_OCCLUSION_QUERY. Where they're unsupported, fall back to CPU tests (recipe 29's frustum culling) or just draw everything. Queries also cost a GPU round-trip, so they pay off only for expensive, frequently-hidden geometry.

Complete listing

A grid of cubes behind a big rotating wall. As the wall sweeps across them, the debug overlay reports how many cubes the GPU reports visible vs. hidden, and a bright marker floats over each visible cube — drawn through setCondition, so it vanishes the instant a cube slips behind the wall. Native-handle / loadProgram helpers identical to recipe 03 (elided). Shaders: vs_cube / fs_cube. 

/*
 * bgfx/27-occlusion-queries.cpp — per-object occlusion queries behind a moving
 * occluder; getResult tallies visibility, setCondition gates a marker draw.
 * Native-handle + loadProgram helpers identical to recipe 03 (elided).
 * Shaders: vs_cube/fs_cube.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);

struct PNVertex { float px, py, pz; float nx, ny, nz; };
static const PNVertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const uint16_t kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

static const int GRID = 6;
static const int NUM_OBJ = GRID * GRID;

struct AppState {
    SDL_Window *window = nullptr; int width = 1280, height = 720;
    bgfx::VertexLayout layout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle prog = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_color = BGFX_INVALID_HANDLE, u_lightDir = BGFX_INVALID_HANDLE;
    bgfx::OcclusionQueryHandle query[NUM_OBJ];
    bool haveOQ = false;
    Uint64 startNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

// Position of object i in the grid behind the wall.
static void obj_pos(int i, float *x, float *y, float *z) {
    int gx = i % GRID, gz = i / GRID;
    *x = (gx - GRID*0.5f + 0.5f) * 1.8f;
    *y = (gz - GRID*0.5f + 0.5f) * 1.8f;
    *z = 3.0f;                                  // all sit on a plane behind the wall
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Occlusion Queries", "1.0", "com.example.sdl3.bgfx.oq");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Occlusion Queries", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");
    bgfx::setDebug(BGFX_DEBUG_TEXT);                            // enable the dbgText overlay

    st->haveOQ = (bgfx::getCaps()->supported & BGFX_CAPS_OCCLUSION_QUERY) != 0;

    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Normal,   3, bgfx::AttribType::Float).end();
    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kIdx, sizeof(kIdx)));
    st->prog = loadProgram("vs_cube", "fs_cube");
    st->u_color    = bgfx::createUniform("u_color",    bgfx::UniformType::Vec4);
    st->u_lightDir = bgfx::createUniform("u_lightDir", bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->prog)) return fail("loadProgram");

    for (int i = 0; i < NUM_OBJ; ++i)
        st->query[i] = st->haveOQ ? bgfx::createOcclusionQuery() : BGFX_INVALID_HANDLE;

    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

static void setCube(AppState *st, const float *model, float r, float g, float b)
{
    float col[4] = { r, g, b, 1 };
    bgfx::setUniform(st->u_color, col);
    bgfx::setTransform(model);
    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    bgfx::setViewMode(0, bgfx::ViewMode::Sequential);          // draw order = submit order
    bgfx::setViewRect(0, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x141821ff, 1.0f, 0);
    const bx::Vec3 at = { 0, 0, 3.0f }, eye = { 0, 0, -9.0f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 55.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, view, proj);
    float ld[4] = { -0.4f, -0.5f, -0.7f, 0 };
    bgfx::setUniform(st->u_lightDir, ld);

    // --- 1) the occluder wall, drawn first so its depth is present ---
    float wall[16], ws[16], wr[16], wt[16];
    bx::mtxRotateY(wr, t * 0.5f);
    bx::mtxScale(ws, 5.0f, 8.0f, 0.4f);
    bx::mtxMul(wt, ws, wr);
    bx::mtxTranslate(wall, 0, 0, 0.0f);                        // wall sits in front of the grid
    float wallModel[16]; bx::mtxMul(wallModel, wt, wall);
    setCube(st, wallModel, 0.30f, 0.34f, 0.42f);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(0, st->prog);                                 // occluder: no query

    // --- 2) the grid objects, each tagged with its occlusion query ---
    int visible = 0, hidden = 0;
    for (int i = 0; i < NUM_OBJ; ++i) {
        if (st->haveOQ) {                                      // read last frame's result
            int32_t px = 0;
            bgfx::OcclusionQueryResult::Enum r = bgfx::getResult(st->query[i], &px);
            if (r == bgfx::OcclusionQueryResult::Visible)   ++visible;
            if (r == bgfx::OcclusionQueryResult::Invisible) ++hidden;
        }
        float x, y, z; obj_pos(i, &x, &y, &z);
        float m[16]; bx::mtxTranslate(m, x, y, z);
        setCube(st, m, 0.75f, 0.55f, 0.35f);
        bgfx::setState(BGFX_STATE_DEFAULT);
        if (st->haveOQ) bgfx::submit(0, st->prog, st->query[i]);
        else            bgfx::submit(0, st->prog);

        // --- 3) conditional marker: a small bright cube, drawn only if visible ---
        if (st->haveOQ) {
            float mk[16], ms[16], mtr[16];
            bx::mtxScale(ms, 0.35f, 0.35f, 0.35f);
            bx::mtxTranslate(mtr, x, y + 1.0f, z);
            bx::mtxMul(mk, ms, mtr);
            setCube(st, mk, 0.4f, 1.0f, 0.6f);
            bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);   // overlay: no depth write
            bgfx::setCondition(st->query[i], true);            // only if last frame was visible
            bgfx::submit(0, st->prog);
        }
    }

    bgfx::dbgTextClear();
    bgfx::dbgTextPrintf(1, 1, 0x0f, "Occlusion queries: %s", st->haveOQ ? "supported" : "UNSUPPORTED");
    bgfx::dbgTextPrintf(1, 2, 0x0a, "Visible: %d", visible);
    bgfx::dbgTextPrintf(1, 3, 0x0c, "Hidden:  %d", hidden);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    for (int i = 0; i < NUM_OBJ; ++i) if (bgfx::isValid(st->query[i])) bgfx::destroy(st->query[i]);
    if (bgfx::isValid(st->u_lightDir)) bgfx::destroy(st->u_lightDir);
    if (bgfx::isValid(st->u_color))    bgfx::destroy(st->u_color);
    if (bgfx::isValid(st->ibh))        bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))        bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->prog))       bgfx::destroy(st->prog);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# vs_cube.sc / fs_cube.sc — a flat-shaded cube with a uniform colour
# (vs)
$input  a_position, a_normal
$output v_normal
#include "bgfx_shader.sh"
void main() {
    gl_Position = mul(u_modelViewProj, vec4(a_position, 1.0));
    v_normal    = mul(u_model[0], vec4(a_normal, 0.0)).xyz;
}
# (fs)
$input v_normal
#include "bgfx_shader.sh"
uniform vec4 u_color;
uniform vec4 u_lightDir;
void main() {
    float diff = max(dot(normalize(v_normal), normalize(-u_lightDir.xyz)), 0.0);
    gl_FragColor = vec4(u_color.rgb * (0.3 + 0.75 * diff), 1.0);
}

Try it yourself

  1. Make the query a cheap proxy: draw an invisible bounding box (no colour/depth write) tagged with the query, then setCondition the real, expensive mesh — the standard GPU occlusion-culling pattern.
  2. Print the passing pixel count (px) per object and watch it fall to zero as a cube slides behind the wall.
  3. Handle NoResult explicitly: on the first frames (or a stall) the query isn't ready — decide whether to draw or skip by default.
  4. Compare with recipe 29's CPU frustum culling: queries catch occlusion a frustum test can't, but cost a GPU round-trip. Combine both.
