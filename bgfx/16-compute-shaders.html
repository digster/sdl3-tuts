Recipe 16 — Compute Shaders

Simulate on the GPU. A bgfx compute program steps a buffer of particles every frame, and the very same buffer is drawn as a point cloud — the data never touches the CPU.

What you'll learn

  • Authoring a bgfx compute shader with BUFFER_RW and NUM_THREADS.
  • A dynamic vertex buffer flagged for compute read/write and drawing.
  • bgfx::dispatch on one view, then drawing the result on the next.
  • Letting bgfx insert the compute→vertex barrier so the passes stay correct.

Prerequisites

Recipe 11 for buffers-as-data and the shaderc workflow from recipe 04 — compute shaders compile the same way, just with a compute profile.

Walkthrough

A buffer the GPU owns end to end

The particles live in a dynamic vertex buffer created with two compute flags so the kernel can read and write it, while its vertex layout still lets the render pass draw it. One buffer, simulated and drawn entirely on the GPU. 

layout.begin()
    .add(bgfx::Attrib::Position,  4, bgfx::AttribType::Float)   // xyz + life
    .add(bgfx::Attrib::TexCoord0, 4, bgfx::AttribType::Float)   // xyz velocity + _
    .end();
st->buffer = bgfx::createDynamicVertexBuffer(
    bgfx::copy(init.data(), uint32_t(init.size() * sizeof(Particle))), layout,
    BGFX_BUFFER_COMPUTE_READ | BGFX_BUFFER_COMPUTE_WRITE);

Each particle is two vec4s — position+life and velocity — so 32 bytes, matching the layout's stride.

The compute shader

bgfx compute shaders #include "bgfx_compute.sh", declare read/write buffers with BUFFER_RW, and set their workgroup size with NUM_THREADS. Each invocation handles one particle, keyed off gl_GlobalInvocationID. 

# cs_particles.sc — gravity fountain; respawn at the nozzle when life runs out
#include "bgfx_compute.sh"
BUFFER_RW(b_particles, vec4, 0);
uniform vec4 u_params;          // x = dt, y = count, z = time
NUM_THREADS(64, 1, 1)
void main() {
    uint id = gl_GlobalInvocationID.x;
    if (float(id) >= u_params.y) return;
    vec4 p = b_particles[id * 2u + 0u];     // xyz pos, w life
    vec4 v = b_particles[id * 2u + 1u];     // xyz velocity
    v.y   -= 1.5 * u_params.x;               // gravity
    p.xyz += v.xyz * u_params.x;
    p.w   -= u_params.x;                      // age
    if (p.w <= 0.0) {                         // reborn at the nozzle
        float r = fract(sin(float(id) * 12.9898 + u_params.z) * 43758.5453);
        float a = r * 6.2831;
        p = vec4(0.0, -1.6, 0.0, 1.5 + r);
        v = vec4(cos(a) * 1.1, 2.4 + r, sin(a) * 1.1, 0.0);
    }
    b_particles[id * 2u + 0u] = p;
    b_particles[id * 2u + 1u] = v;
}

Dispatch, then draw

Bind the buffer for the compute view with setBuffer(..., Access::ReadWrite) and dispatch one workgroup per 64 particles. Then, on the next view, draw the same handle as a point list. bgfx sees that a buffer written by compute on view 0 is read as vertices on view 1 and inserts the barrier for you. 

// view 0 — simulate
bgfx::setUniform(st->u_params, params);
bgfx::setBuffer(0, st->buffer, bgfx::Access::ReadWrite);
bgfx::dispatch(0, st->compute, (N + 63) / 64, 1, 1);

// view 1 — draw the same buffer
bgfx::setViewTransform(1, view, proj);
bgfx::setVertexBuffer(0, st->buffer);
bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A | BGFX_STATE_PT_POINTS);
bgfx::submit(1, st->render);
One buffer, two stages. Because the data is born, simulated and drawn without ever leaving GPU memory, this scales to millions of particles. It's also how bgfx does N-body, cloth and fluid demos.

Complete listing

A 65k-particle gravity fountain. The CPU only ticks the clock and counts workgroups; the GPU does the rest. Build cs_particles / vs_particle / fs_particle with shaderc (use the compute profile for the first). 

/*
 * bgfx/16-compute-shaders.cpp — GPU particle fountain simulated in compute,
 * drawn from the same dynamic vertex buffer.
 * Shaders: cs_particles (compute), vs_particle / fs_particle.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>
#include <vector>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);
static bgfx::ProgramHandle loadComputeProgram(const char *cs);

static const int N = 65536;
struct Particle { float pos[4]; float vel[4]; };   // pos.xyz+life, vel.xyz

struct AppState {
    SDL_Window *window = nullptr;
    int width = 1280, height = 720;
    bgfx::DynamicVertexBufferHandle buffer = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle compute = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle render  = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_params = BGFX_INVALID_HANDLE;
    Uint64 startNs = 0, lastNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Compute", "1.0", "com.example.sdl3.bgfx.compute");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");

    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Compute Shaders", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type              = bgfx::RendererType::Count;
    init.platformData.nwh  = nativeWindowHandle(st->window);
    init.platformData.ndt  = nativeDisplayHandle(st->window);
    init.resolution.width  = (uint32_t)st->width;
    init.resolution.height = (uint32_t)st->height;
    init.resolution.reset  = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    if (!(bgfx::getCaps()->supported & BGFX_CAPS_COMPUTE)) return fail("compute unsupported");

    bgfx::VertexLayout layout;
    layout.begin()
        .add(bgfx::Attrib::Position,  4, bgfx::AttribType::Float)
        .add(bgfx::Attrib::TexCoord0, 4, bgfx::AttribType::Float).end();

    // Seed the particles with staggered lifetimes so the fountain fills in.
    std::vector<Particle> seed(N);
    for (int i = 0; i < N; ++i) {
        float r = SDL_randf(), a = SDL_randf() * 6.2831f;
        seed[i] = { { 0, -1.6f, 0, SDL_randf() * 2.0f },
                    { SDL_cosf(a)*1.1f, 2.4f + r, SDL_sinf(a)*1.1f, 0 } };
    }
    st->buffer = bgfx::createDynamicVertexBuffer(
        bgfx::copy(seed.data(), uint32_t(seed.size()*sizeof(Particle))), layout,
        BGFX_BUFFER_COMPUTE_READ | BGFX_BUFFER_COMPUTE_WRITE);

    st->compute  = loadComputeProgram("cs_particles");
    st->render   = loadProgram("vs_particle", "fs_particle");
    st->u_params = bgfx::createUniform("u_params", bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->compute) || !bgfx::isValid(st->render)) return fail("loadProgram");

    st->startNs = st->lastNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->lastNs) / 1.0e9f; if (dt > 0.05f) dt = 0.05f;
    st->lastNs = now;
    float time = (now - st->startNs) / 1.0e9f;

    // --- view 0: simulate ---
    float params[4] = { dt, (float)N, time, 0 };
    bgfx::setUniform(st->u_params, params);
    bgfx::setBuffer(0, st->buffer, bgfx::Access::ReadWrite);
    bgfx::dispatch(0, st->compute, (N + 63) / 64, 1, 1);

    // --- view 1: draw the same buffer as points ---
    bgfx::setViewClear(1, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x05060aff, 1.0f, 0);
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 0.5f, -6 };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 60.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(1, view, proj);

    float mtx[16]; bx::mtxRotateY(mtx, time * 0.2f);
    bgfx::setTransform(mtx);
    bgfx::setVertexBuffer(0, st->buffer);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A | BGFX_STATE_PT_POINTS);
    bgfx::submit(1, st->render);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (bgfx::isValid(st->u_params)) bgfx::destroy(st->u_params);
    if (bgfx::isValid(st->buffer))   bgfx::destroy(st->buffer);
    if (bgfx::isValid(st->render))   bgfx::destroy(st->render);
    if (bgfx::isValid(st->compute))  bgfx::destroy(st->compute);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# vs_particle.sc / fs_particle.sc — colour points by remaining life.
# (vs)
$input  a_position, a_texcoord0
$output v_color0
#include "bgfx_shader.sh"
void main() {
    gl_Position = mul(u_modelViewProj, vec4(a_position.xyz, 1.0));
    float life  = a_position.w;
    v_color0    = vec4(0.6 + 0.4 * life, 0.45, 1.0 - 0.3 * life, 1.0);
}
# (fs)
$input v_color0
#include "bgfx_shader.sh"
void main() { gl_FragColor = v_color0; }

Try it yourself

  1. Add mouse attraction: feed the cursor's world position in u_params and steer velocities toward it.
  2. Push N to a million. Only the dispatch count changes; the fountain just gets denser.
  3. Expand points into camera-facing quads with an instanced draw so particles have real size, not one pixel.
  4. Double-buffer the particles (read one, write the other, swap) if you add neighbour interactions that must read a consistent previous state.
