Recipe 24 — Deferred Rendering

Build a G-buffer the bgfx way. A multi-render-target framebuffer captures albedo, normal and position in one geometry view; a second fullscreen view lights the screen from it — many lights, one rasterisation of the scene.

What you'll learn

  • A multi-attachment FrameBuffer from several RT textures (MRT) via createFrameBuffer.
  • Writing each G-buffer channel with gl_FragData[0..2].
  • Reading the attachments back with bgfx::getTexture(fb, i) for the lighting pass.
  • A fullscreen lighting view that loops many point lights over the G-buffer.

Prerequisites

Recipe 10 — Views & Render-to-Texture for framebuffers and the view model, and recipe 15 for the transient fullscreen triangle. This is the bgfx counterpart to 3D recipe 25.

Walkthrough

A framebuffer with several colour attachments

bgfx builds an MRT target by handing createFrameBuffer an array of RT textures plus a depth target. Albedo fits in RGBA8; normals and positions need RGBA16F so signed, unbounded values survive. 

uint64_t rt = BGFX_TEXTURE_RT | BGFX_SAMPLER_POINT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
bgfx::TextureHandle att[4] = {
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::RGBA8,    rt),  // albedo
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::RGBA16F,  rt),  // normal
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::RGBA16F,  rt),  // position
    bgfx::createTexture2D(w, h, false, 1, bgfx::TextureFormat::D24S8, BGFX_TEXTURE_RT_WRITE_ONLY),
};
st->gbuffer = bgfx::createFrameBuffer(4, att, /*destroyTextures=*/true);

One fragment shader, three outputs

The geometry program writes all three channels through gl_FragData. View 0 points at the G-buffer framebuffer, so a single draw of the scene fills albedo, normal and position at once. 

# fs_gbuffer.sc
$input v_wpos, v_normal
#include "bgfx_shader.sh"
uniform vec4 u_color;
void main() {
    gl_FragData[0] = u_color;                       // albedo
    gl_FragData[1] = vec4(normalize(v_normal), 1.0);// world normal (float target: raw)
    gl_FragData[2] = vec4(v_wpos, 1.0);             // world position
}

Light the screen from the attachments

View 1 targets the backbuffer. Bind the three attachments — fetched with bgfx::getTexture(st->gbuffer, i) — to three samplers, draw the fullscreen triangle, and loop the lights per pixel. 

bgfx::setViewFrameBuffer(1, BGFX_INVALID_HANDLE);          // back to the screen
bgfx::setTexture(0, st->s_albedo,   bgfx::getTexture(st->gbuffer, 0));
bgfx::setTexture(1, st->s_normal,   bgfx::getTexture(st->gbuffer, 1));
bgfx::setTexture(2, st->s_position, bgfx::getTexture(st->gbuffer, 2));
screenTri(st);
bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);  // no depth: fullscreen
bgfx::submit(1, st->light);
Same idea, two renderers. This mirrors SDL_GPU recipe 25 beat for beat — geometry into an MRT target, then a screen-space lighting pass. Only the API surface differs: bgfx's FrameBuffer + views versus SDL_GPU's colour-target arrays.

Complete listing

A ring of cubes over a floor, rasterised once into a three-target G-buffer and lit by sixteen orbiting point lights in a fullscreen pass. Native-handle / loadProgram helpers identical to recipe 03 (elided). Shaders: vs_gbuffer / fs_gbuffer, vs_fullscreen / fs_light. 

/*
 * bgfx/24-deferred-rendering.cpp — MRT G-buffer + a fullscreen lighting pass.
 * Native-handle + loadProgram helpers identical to recipe 03 (elided).
 * Shaders: vs_gbuffer/fs_gbuffer, vs_fullscreen/fs_light.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);

struct PNVertex { float px, py, pz; float nx, ny, nz; };
struct PosUvVertex { float x, y; float u, v; };

static const PNVertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const uint16_t kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

#define NUM_LIGHTS 16

struct AppState {
    SDL_Window *window = nullptr; int width = 1280, height = 720, fbW = 0, fbH = 0;
    bgfx::VertexLayout layout, quadLayout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::FrameBufferHandle  gbuffer = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle geom = BGFX_INVALID_HANDLE, light = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_color = BGFX_INVALID_HANDLE, u_eyePos = BGFX_INVALID_HANDLE, u_params = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_lightPosRange = BGFX_INVALID_HANDLE, u_lightColor = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle s_albedo = BGFX_INVALID_HANDLE, s_normal = BGFX_INVALID_HANDLE, s_position = BGFX_INVALID_HANDLE;
    Uint64 startNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

// (Re)build the three-target G-buffer + depth at the current size.
static void makeGBuffer(AppState *st, int w, int h)
{
    if (bgfx::isValid(st->gbuffer)) bgfx::destroy(st->gbuffer);   // also frees its textures
    uint64_t rt = BGFX_TEXTURE_RT | BGFX_SAMPLER_POINT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
    bgfx::TextureHandle att[4] = {
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA8,   rt),
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA16F, rt),
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA16F, rt),
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::D24S8, BGFX_TEXTURE_RT_WRITE_ONLY),
    };
    st->gbuffer = bgfx::createFrameBuffer(4, att, true);
    st->fbW = w; st->fbH = h;
}

// Allocate + bind a clip-space fullscreen triangle (V flipped on top-left-origin backends).
static void screenTri(AppState *st)
{
    if (bgfx::getAvailTransientVertexBuffer(3, st->quadLayout) != 3) return;
    bgfx::TransientVertexBuffer tvb;
    bgfx::allocTransientVertexBuffer(&tvb, 3, st->quadLayout);
    PosUvVertex *v = (PosUvVertex *)tvb.data;
    bool flip = !bgfx::getCaps()->originBottomLeft;
    float vb = flip ? 1.0f : 0.0f, vt = flip ? -1.0f : 2.0f;
    v[0] = { -1.0f, -1.0f, 0.0f, vb };
    v[1] = {  3.0f, -1.0f, 2.0f, vb };
    v[2] = { -1.0f,  3.0f, 0.0f, vt };
    bgfx::setVertexBuffer(0, &tvb);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Deferred", "1.0", "com.example.sdl3.bgfx.deferred");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Deferred Rendering", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Normal,   3, bgfx::AttribType::Float).end();
    st->quadLayout.begin()
        .add(bgfx::Attrib::Position,  2, bgfx::AttribType::Float)
        .add(bgfx::Attrib::TexCoord0, 2, bgfx::AttribType::Float).end();
    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kIdx, sizeof(kIdx)));
    st->geom  = loadProgram("vs_gbuffer",    "fs_gbuffer");
    st->light = loadProgram("vs_fullscreen", "fs_light");
    st->u_color         = bgfx::createUniform("u_color",         bgfx::UniformType::Vec4);
    st->u_eyePos        = bgfx::createUniform("u_eyePos",        bgfx::UniformType::Vec4);
    st->u_params        = bgfx::createUniform("u_params",        bgfx::UniformType::Vec4);
    st->u_lightPosRange = bgfx::createUniform("u_lightPosRange", bgfx::UniformType::Vec4, NUM_LIGHTS);
    st->u_lightColor    = bgfx::createUniform("u_lightColor",    bgfx::UniformType::Vec4, NUM_LIGHTS);
    st->s_albedo   = bgfx::createUniform("s_albedo",   bgfx::UniformType::Sampler);
    st->s_normal   = bgfx::createUniform("s_normal",   bgfx::UniformType::Sampler);
    st->s_position = bgfx::createUniform("s_position", bgfx::UniformType::Sampler);
    if (!bgfx::isValid(st->geom) || !bgfx::isValid(st->light)) return fail("loadProgram");

    makeGBuffer(st, st->width, st->height);
    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
        makeGBuffer(st, st->width, st->height);
    }
    return SDL_APP_CONTINUE;
}

static void drawCube(AppState *st, const float *model, float r, float g, float b)
{
    float col[4] = { r, g, b, 1.0f };
    bgfx::setUniform(st->u_color, col);
    bgfx::setTransform(model);
    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(0, st->geom);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    // --- view 0: geometry into the G-buffer ---
    bgfx::setViewFrameBuffer(0, st->gbuffer);
    bgfx::setViewRect(0, 0, 0, (uint16_t)st->fbW, (uint16_t)st->fbH);
    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x00000000, 1.0f, 0);   // normal=0 ⇒ background
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 3.5f, -9.0f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 55.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, view, proj);

    float floor[16], s[16], tr[16];
    bx::mtxScale(s, 12.0f, 0.2f, 12.0f);
    bx::mtxTranslate(tr, 0.0f, -1.2f, 0.0f);
    bx::mtxMul(floor, s, tr);
    drawCube(st, floor, 0.8f, 0.8f, 0.85f);
    for (int i = 0; i < 8; ++i) {
        float a = i / 8.0f * 6.2831853f;
        float rot[16], pos[16], model[16];
        bx::mtxRotateY(rot, t + a);
        bx::mtxTranslate(pos, 3.2f * bx::cos(a), -0.4f, 3.2f * bx::sin(a));
        bx::mtxMul(model, rot, pos);
        drawCube(st, model, 0.85f, 0.8f, 0.75f);
    }

    // --- view 1: fullscreen lighting from the G-buffer ---
    bgfx::setViewFrameBuffer(1, BGFX_INVALID_HANDLE);
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    bgfx::setViewClear(1, BGFX_CLEAR_COLOR, 0x05060aff, 1.0f, 0);

    float eyev[4] = { eye.x, eye.y, eye.z, 0 };
    bgfx::setUniform(st->u_eyePos, eyev);
    float params[4] = { (float)NUM_LIGHTS, 0, 0, 0 };
    bgfx::setUniform(st->u_params, params);
    float lpos[NUM_LIGHTS][4], lcol[NUM_LIGHTS][4];
    for (int i = 0; i < NUM_LIGHTS; ++i) {
        float a = t * 0.6f + i / (float)NUM_LIGHTS * 6.2831853f;
        lpos[i][0] = 3.4f * bx::cos(a);
        lpos[i][1] = 0.4f + bx::sin(t + i) * 0.6f;
        lpos[i][2] = 3.4f * bx::sin(a);
        lpos[i][3] = 4.5f;                                  // range
        lcol[i][0] = 0.5f + 0.5f * bx::sin(i * 1.3f);
        lcol[i][1] = 0.5f + 0.5f * bx::sin(i * 1.3f + 2.0f);
        lcol[i][2] = 0.5f + 0.5f * bx::sin(i * 1.3f + 4.0f);
        lcol[i][3] = 1.6f;                                  // intensity
    }
    bgfx::setUniform(st->u_lightPosRange, lpos, NUM_LIGHTS);
    bgfx::setUniform(st->u_lightColor,    lcol, NUM_LIGHTS);
    bgfx::setTexture(0, st->s_albedo,   bgfx::getTexture(st->gbuffer, 0));
    bgfx::setTexture(1, st->s_normal,   bgfx::getTexture(st->gbuffer, 1));
    bgfx::setTexture(2, st->s_position, bgfx::getTexture(st->gbuffer, 2));
    screenTri(st);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);
    bgfx::submit(1, st->light);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    bgfx::UniformHandle us[] = { st->s_position, st->s_normal, st->s_albedo, st->u_lightColor,
                                 st->u_lightPosRange, st->u_params, st->u_eyePos, st->u_color };
    for (bgfx::UniformHandle u : us) if (bgfx::isValid(u)) bgfx::destroy(u);
    if (bgfx::isValid(st->gbuffer)) bgfx::destroy(st->gbuffer);
    if (bgfx::isValid(st->ibh))     bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))     bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->light))   bgfx::destroy(st->light);
    if (bgfx::isValid(st->geom))    bgfx::destroy(st->geom);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# vs_gbuffer.sc / fs_gbuffer.sc — write albedo, normal, position
# (vs)
$input  a_position, a_normal
$output v_wpos, v_normal
#include "bgfx_shader.sh"
void main() {
    vec4 wpos   = mul(u_model[0], vec4(a_position, 1.0));
    v_wpos      = wpos.xyz;
    v_normal    = normalize(mul(u_model[0], vec4(a_normal, 0.0)).xyz);
    gl_Position = mul(u_viewProj, wpos);
}
# (fs)
$input v_wpos, v_normal
#include "bgfx_shader.sh"
uniform vec4 u_color;
void main() {
    gl_FragData[0] = u_color;                        // albedo
    gl_FragData[1] = vec4(normalize(v_normal), 1.0); // world-space normal
    gl_FragData[2] = vec4(v_wpos, 1.0);              // world-space position
}

# fs_light.sc — screen-space lighting from the G-buffer
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_albedo,   0);
SAMPLER2D(s_normal,   1);
SAMPLER2D(s_position, 2);
uniform vec4 u_eyePos;
uniform vec4 u_params;                 // x = light count
uniform vec4 u_lightPosRange[16];      // xyz position, w range
uniform vec4 u_lightColor[16];         // rgb colour, w intensity
void main() {
    vec3 albedo = texture2D(s_albedo, v_texcoord0).rgb;
    vec3 N = texture2D(s_normal, v_texcoord0).xyz;
    if (dot(N, N) < 0.5) { gl_FragColor = vec4(0.02, 0.02, 0.03, 1.0); return; }  // background
    N = normalize(N);
    vec3 P = texture2D(s_position, v_texcoord0).xyz;
    vec3 V = normalize(u_eyePos.xyz - P);
    vec3 lit = albedo * 0.06;
    int n = int(u_params.x);
    for (int i = 0; i < 16; ++i) {
        if (i >= n) break;
        vec3  toL  = u_lightPosRange[i].xyz - P;
        float dist = length(toL);
        vec3  L    = toL / max(dist, 1e-4);
        float att  = clamp(1.0 - dist / u_lightPosRange[i].w, 0.0, 1.0); att *= att;
        float diff = max(dot(N, L), 0.0);
        float spec = pow(max(dot(N, normalize(L + V)), 0.0), 32.0);
        lit += (albedo * diff + spec) * u_lightColor[i].xyz * u_lightColor[i].w * att;
    }
    gl_FragColor = vec4(lit, 1.0);
}

Try it yourself

  1. Inspect the G-buffer: point view 1 at one attachment and output it directly (albedo, normal*0.5+0.5, scaled position) to see what deferred stores.
  2. Add MRT clear colours with bgfx::setViewClearMrt-style palette clears so each attachment clears to its own value.
  3. Light volumes: draw a sphere per light into a stencil-masked region instead of a fullscreen triangle so only affected pixels shade.
  4. Reconstruct position from depth instead of storing it, dropping the third attachment to save bandwidth — the standard production optimisation.
