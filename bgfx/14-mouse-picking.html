Recipe 14 — Mouse Picking

Click a 3D object. Turn the 2D cursor into a world-space ray, test it against the scene, and light up whatever it hits — the maths behind every editor and strategy game.

What you'll learn

  • Unprojecting a screen pixel into a world-space ray through the inverse view-projection.
  • bgfx's matrix convention, and why mtx[12..14] being the translation tells you it's column-major.
  • A ray-vs-AABB slab test to find the nearest hit.
  • Highlighting the picked object with a per-submit tint uniform.

Prerequisites

Recipe 06 for view/projection and uniforms, and recipe 07 for the cube. New fragment shader fs_picking (a tint uniform); vs_cubes is reused.

Walkthrough

From pixel to ray

The cursor is a point on the screen, but in the world it's a whole ray shooting away from the camera. To find it, take two points in normalised device coordinates — one on the near plane, one on the far — and push both through the inverse of the view-projection matrix. The line between the two world-space points is your ray. 

// Pixel → NDC. bgfx NDC has Y up, so flip the cursor's Y.
float ndcx = 2.0f * mx / width  - 1.0f;
float ndcy = 1.0f - 2.0f * my / height;
// Near Z is -1 on OpenGL-style depth, 0 elsewhere.
float nearZ = bgfx::getCaps()->homogeneousDepth ? -1.0f : 0.0f;
bx::Vec3 o = unproject(invVP, ndcx, ndcy, nearZ);   // ray origin
bx::Vec3 f = unproject(invVP, ndcx, ndcy, 1.0f);    // far point
bx::Vec3 dir = bx::normalize(bx::sub(f, o));

bgfx is column-major

How do you transform a vector by a bgfx matrix in C++? Recall from recipe 11 that translation lives in mtx[12], [13], [14] — that's the fourth column, which means bgfx stores matrices column-major and transforms as M · v (matching the shader's mul(m, v)). So the product component r is Σ m[c*4 + r] · v[c], and we divide by w at the end: 

static bx::Vec3 unproject(const float *m, float x, float y, float z)
{
    float ox = m[0]*x + m[4]*y + m[8] *z + m[12];
    float oy = m[1]*x + m[5]*y + m[9] *z + m[13];
    float oz = m[2]*x + m[6]*y + m[10]*z + m[14];
    float ow = m[3]*x + m[7]*y + m[11]*z + m[15];
    return { ox/ow, oy/ow, oz/ow };
}
// viewProj = proj * view, so clip = mul(viewProj, world):
float viewProj[16], invVP[16];
bx::mtxMul(viewProj, proj, view);
bx::mtxInverse(invVP, viewProj);

Ray vs box

Each cube is an axis-aligned box. The slab test clips the ray against each axis's pair of planes and keeps the overlapping interval; if the interval survives all three axes and ends in front of the camera, it's a hit. Track the nearest t across all boxes — that's the one the user means. 

static bool ray_aabb(bx::Vec3 o, bx::Vec3 d, bx::Vec3 lo, bx::Vec3 hi, float &tHit)
{
    float tmin = -1e30f, tmax = 1e30f;
    const float od[3] = { o.x, o.y, o.z }, dd[3] = { d.x, d.y, d.z };
    const float lod[3] = { lo.x, lo.y, lo.z }, hid[3] = { hi.x, hi.y, hi.z };
    for (int a = 0; a < 3; ++a) {
        float inv = 1.0f / dd[a];
        float t0 = (lod[a] - od[a]) * inv, t1 = (hid[a] - od[a]) * inv;
        if (inv < 0) { float tmp = t0; t0 = t1; t1 = tmp; }
        tmin = bx::max(tmin, t0); tmax = bx::min(tmax, t1);
        if (tmax < tmin) return false;
    }
    tHit = tmin;
    return tmax >= 0;
}
Two ways to pick. This ray test runs on the CPU and scales to a few hundred objects easily. For thousands, the other classic is colour-ID picking: render each object in a unique flat colour to an offscreen view (recipe 10) and read back the pixel under the cursor.

Complete listing

A 5×5 grid of cubes. Move the mouse and the cube under the cursor lights up — that's the ray from your pixel hitting its box. Needs fs_picking: vs_cubes plus a u_tint multiply. 

/*
 * bgfx/14-mouse-picking.cpp — unproject the cursor to a ray, ray-AABB the grid.
 * Shaders: vs_cubes / fs_picking (vs_cubes + u_tint multiply).
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);

struct PosColorVertex { float x, y, z; uint32_t abgr; };
static PosColorVertex kCube[8] = {
    {-1, 1, 1, 0xff8fcaff},{ 1, 1, 1, 0xff8fcaff},{-1,-1, 1, 0xff5a86c0},{ 1,-1, 1, 0xff5a86c0},
    {-1, 1,-1, 0xff8fcaff},{ 1, 1,-1, 0xff8fcaff},{-1,-1,-1, 0xff5a86c0},{ 1,-1,-1, 0xff5a86c0},
};
static const uint16_t kCubeIdx[36] = {
    0,1,2, 1,3,2, 4,6,5, 5,6,7, 0,2,4, 4,2,6, 1,5,3, 5,7,3, 0,4,1, 4,5,1, 2,3,6, 6,3,7,
};

static const int GRID = 5;
static const float SP = 3.0f;        // grid spacing; cubes are ±1 so AABB = c±1

static bx::Vec3 unproject(const float *m, float x, float y, float z)
{
    float ox = m[0]*x + m[4]*y + m[8] *z + m[12];
    float oy = m[1]*x + m[5]*y + m[9] *z + m[13];
    float oz = m[2]*x + m[6]*y + m[10]*z + m[14];
    float ow = m[3]*x + m[7]*y + m[11]*z + m[15];
    return { ox/ow, oy/ow, oz/ow };
}
static bool ray_aabb(bx::Vec3 o, bx::Vec3 d, bx::Vec3 lo, bx::Vec3 hi, float &tHit)
{
    float tmin = -1e30f, tmax = 1e30f;
    const float od[3]={o.x,o.y,o.z}, dd[3]={d.x,d.y,d.z}, lod[3]={lo.x,lo.y,lo.z}, hid[3]={hi.x,hi.y,hi.z};
    for (int a = 0; a < 3; ++a) {
        float inv = 1.0f / dd[a];
        float t0 = (lod[a]-od[a])*inv, t1 = (hid[a]-od[a])*inv;
        if (inv < 0) { float t=t0; t0=t1; t1=t; }
        tmin = bx::max(tmin, t0); tmax = bx::min(tmax, t1);
        if (tmax < tmin) return false;
    }
    tHit = tmin; return tmax >= 0;
}

struct AppState {
    SDL_Window *window = nullptr;
    int width = 1280, height = 720;
    float mx = 0, my = 0;
    int   picked = -1;
    bgfx::VertexLayout layout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle program = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle  u_tint  = BGFX_INVALID_HANDLE;
    float view[16], proj[16];
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

static bx::Vec3 cube_center(int i) {
    int gx = i % GRID, gy = i / GRID;
    return { (gx - (GRID-1)*0.5f) * SP, (gy - (GRID-1)*0.5f) * SP, 0.0f };
}

// Re-cast the ray and update st->picked.
static void repick(AppState *st)
{
    float viewProj[16], invVP[16];
    bx::mtxMul(viewProj, st->proj, st->view);
    bx::mtxInverse(invVP, viewProj);

    float ndcx = 2.0f * st->mx / st->width  - 1.0f;
    float ndcy = 1.0f - 2.0f * st->my / st->height;
    float nearZ = bgfx::getCaps()->homogeneousDepth ? -1.0f : 0.0f;
    bx::Vec3 o = unproject(invVP, ndcx, ndcy, nearZ);
    bx::Vec3 f = unproject(invVP, ndcx, ndcy, 1.0f);
    bx::Vec3 dir = bx::normalize(bx::sub(f, o));

    st->picked = -1; float best = 1e30f;
    for (int i = 0; i < GRID*GRID; ++i) {
        bx::Vec3 c = cube_center(i);
        bx::Vec3 lo = { c.x-1, c.y-1, c.z-1 }, hi = { c.x+1, c.y+1, c.z+1 };
        float t;
        if (ray_aabb(o, dir, lo, hi, t) && t < best) { best = t; st->picked = i; }
    }
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Picking", "1.0", "com.example.sdl3.bgfx.pick");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");

    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Mouse Picking", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type              = bgfx::RendererType::Count;
    init.platformData.nwh  = nativeWindowHandle(st->window);
    init.platformData.ndt  = nativeDisplayHandle(st->window);
    init.resolution.width  = (uint32_t)st->width;
    init.resolution.height = (uint32_t)st->height;
    init.resolution.reset  = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x121a30ff, 1.0f, 0);
    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Color0,   4, bgfx::AttribType::Uint8, true).end();
    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kCubeIdx, sizeof(kCubeIdx)));
    st->program = loadProgram("vs_cubes", "fs_picking");
    st->u_tint  = bgfx::createUniform("u_tint", bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->program)) return fail("loadProgram");

    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_MOUSE_MOTION) { st->mx = e->motion.x; st->my = e->motion.y; }
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);

    bgfx::setViewRect(0, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 0, -22 };
    bx::mtxLookAt(st->view, eye, at);
    bx::mtxProj(st->proj, 60.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, st->view, st->proj);

    repick(st);   // ray-cast against the grid with the current cursor

    for (int i = 0; i < GRID*GRID; ++i) {
        bx::Vec3 c = cube_center(i);
        float mtx[16]; bx::mtxTranslate(mtx, c.x, c.y, c.z);
        bgfx::setTransform(mtx);
        float tint[4] = { 1, 1, 1, 1 };
        if (i == st->picked) { tint[0] = 2.2f; tint[1] = 2.0f; tint[2] = 0.8f; }  // glow
        bgfx::setUniform(st->u_tint, tint);
        bgfx::setVertexBuffer(0, st->vbh);
        bgfx::setIndexBuffer(st->ibh);
        bgfx::setState(BGFX_STATE_DEFAULT);
        bgfx::submit(0, st->program);
    }

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (bgfx::isValid(st->u_tint))  bgfx::destroy(st->u_tint);
    if (bgfx::isValid(st->ibh))     bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))     bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->program)) bgfx::destroy(st->program);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Pick on click instead of hover: cast the ray only on SDL_EVENT_MOUSE_BUTTON_DOWN and keep the selection until the next click.
  2. Print the hit distance t and the world hit point o + dir*t — useful for placing things where the user clicked.
  3. Spin each cube. The AABB stays axis-aligned and a hair loose, which is exactly why broad-phase picking uses boxes, not the rotated mesh.
  4. Swap to colour-ID picking against an offscreen view (recipe 10) and compare the two approaches' cost as the grid grows.
