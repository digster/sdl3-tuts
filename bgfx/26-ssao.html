Recipe 26 — Screen-Space Ambient Occlusion

Contact shadow from the G-buffer you already built. A hemisphere sample kernel plus tiled noise darkens creases and corners; a box blur hides the grain; a composite pass folds it into the lighting — the bgfx twin of 3D recipe 26, on recipe 24's deferred renderer.

What you'll learn

  • Writing a view-space G-buffer (position + normal) from bgfx's u_modelView.
  • A hemisphere kernel uniform array plus a 4×4 noise texture tiled with a REPEAT sampler.
  • Re-projecting samples with the scene projection to read occluder depth.
  • A four-view chain: geometry → SSAO → blur → lit composite.

Prerequisites

Recipe 24 — Deferred Rendering for the MRT framebuffer and fullscreen lighting view, and recipe 15 for the chained fullscreen passes. This is the bgfx counterpart to 3D recipe 26 — the same maths, one buffer laid out in view space.

Walkthrough

A view-space G-buffer

SSAO works in view space so it can re-project sample points with just the projection matrix. bgfx hands the geometry shader u_modelView (view · model) as a built-in, so writing view-space position and normal is a one-line change from recipe 24's world-space G-buffer. 

# vs_gbuffer.sc (view space)
$input  a_position, a_normal
$output v_vpos, v_vnormal
#include "bgfx_shader.sh"
void main() {
    vec4 vp    = mul(u_modelView, vec4(a_position, 1.0));
    v_vpos     = vp.xyz;                                  // view-space position
    v_vnormal  = mul(u_modelView, vec4(a_normal, 0.0)).xyz;
    gl_Position = mul(u_modelViewProj, vec4(a_position, 1.0));
}

Kernel and noise

A u_kernel[24] uniform array holds hemisphere points clustered near the origin; a 4×4 noise texture of random in-plane vectors, tiled with a REPEAT + POINT sampler, rotates the kernel per pixel so the banding breaks into fine grain a blur can erase. 

float kernel[24][4];
for (int i = 0; i < 24; ++i) {
    float sx=SDL_randf()*2-1, sy=SDL_randf()*2-1, sz=SDL_randf();   // +Z hemisphere
    float l = SDL_sqrtf(sx*sx+sy*sy+sz*sz); if (l<1e-5f) l=1;
    float s = 0.1f + 0.9f * (i/24.0f) * (i/24.0f);                  // bias toward centre
    kernel[i][0]=sx/l*s; kernel[i][1]=sy/l*s; kernel[i][2]=sz/l*s; kernel[i][3]=0;
}
bgfx::setUniform(st->u_kernel, kernel, 24);

The SSAO fragment shader

Build a TBN frame from the pixel's normal and the noise vector, place each kernel sample in view space, project it with the scene's projection matrix, and compare the occluder depth stored there. A range check ignores occluders too far to matter. 

# fs_ssao.sc (excerpt)
vec3 sp  = P + mul(TBN, u_kernel[i].xyz) * u_ssaoParams.z;    // view-space sample
vec4 off = mul(u_projMat, vec4(sp, 1.0));
off.xyz /= off.w;
vec2 uv  = off.xy * 0.5 + 0.5;
if (u_flipY.x > 0.5) uv.y = 1.0 - uv.y;                       // match the G-buffer orientation
float sd = texture2D(s_pos, uv).z;                            // occluder view-space z
float range = smoothstep(0.0, 1.0, u_ssaoParams.z / abs(P.z - sd));
occ += (sd >= sp.z + u_ssaoParams.w ? 1.0 : 0.0) * range;
V-orientation is handled; sign and bias still tune. The projected sample UV is flipped to match the G-buffer via a u_flipY uniform driven by bgfx::getCaps()->originBottomLeft, so AO no longer haloes on non-GL backends. The depth comparison still assumes a right-handed view space, so if AO inverts, flip the sd >= sp.z test or nudge radius/bias — the same knobs 3D recipe 26 carries.

struct PNVertex   { float px, py, pz; float nx, ny, nz; };
struct PosUvVertex { float x, y; float u, v; };

    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

// (Re)build the three offscreen framebuffers at the current size.
static void makeTargets(AppState *st, int w, int h)
{
    if (bgfx::isValid(st->gbuffer))  bgfx::destroy(st->gbuffer);
    if (bgfx::isValid(st->aoFb))     bgfx::destroy(st->aoFb);
    if (bgfx::isValid(st->aoBlurFb)) bgfx::destroy(st->aoBlurFb);
    uint64_t rt = BGFX_TEXTURE_RT | BGFX_SAMPLER_POINT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
    bgfx::TextureHandle g[4] = {
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA8,   rt),  // albedo
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA16F, rt),  // view normal
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::RGBA16F, rt),  // view position
        bgfx::createTexture2D((uint16_t)w, (uint16_t)h, false, 1, bgfx::TextureFormat::D24S8, BGFX_TEXTURE_RT_WRITE_ONLY),
    };
    st->gbuffer = bgfx::createFrameBuffer(4, g, true);
    bgfx::TextureHandle a0 = bgfx::createTexture2D((uint16_t)w,(uint16_t)h,false,1,bgfx::TextureFormat::RGBA8, rt);
    st->aoFb = bgfx::createFrameBuffer(1, &a0, true);
    bgfx::TextureHandle a1 = bgfx::createTexture2D((uint16_t)w,(uint16_t)h,false,1,bgfx::TextureFormat::RGBA8, rt);
    st->aoBlurFb = bgfx::createFrameBuffer(1, &a1, true);
    st->fbW = w; st->fbH = h;
}

static void screenTri(AppState *st)
{
    if (bgfx::getAvailTransientVertexBuffer(3, st->quadLayout) != 3) return;
    bgfx::TransientVertexBuffer tvb;
    bgfx::allocTransientVertexBuffer(&tvb, 3, st->quadLayout);
    PosUvVertex *v = (PosUvVertex *)tvb.data;
    bool flip = !bgfx::getCaps()->originBottomLeft;
    float vb = flip ? 1.0f : 0.0f, vt = flip ? -1.0f : 2.0f;
    v[0] = { -1.0f, -1.0f, 0.0f, vb };
    v[1] = {  3.0f, -1.0f, 2.0f, vb };
    v[2] = { -1.0f,  3.0f, 0.0f, vt };
    bgfx::setVertexBuffer(0, &tvb);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx SSAO", "1.0", "com.example.sdl3.bgfx.ssao");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — SSAO", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Normal,   3, bgfx::AttribType::Float).end();
    st->quadLayout.begin()
        .add(bgfx::Attrib::Position,  2, bgfx::AttribType::Float)
        .add(bgfx::Attrib::TexCoord0, 2, bgfx::AttribType::Float).end();
    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kIdx, sizeof(kIdx)));

    st->geom = loadProgram("vs_gbuffer",    "fs_gbuffer");
    st->ssao = loadProgram("vs_fullscreen", "fs_ssao");
    st->blur = loadProgram("vs_fullscreen", "fs_blur");
    st->comp = loadProgram("vs_fullscreen", "fs_composite");
    if (!bgfx::isValid(st->geom) || !bgfx::isValid(st->ssao)) return fail("loadProgram");

    st->u_albedo     = bgfx::createUniform("u_albedo",     bgfx::UniformType::Vec4);
    st->u_projMat    = bgfx::createUniform("u_projMat",    bgfx::UniformType::Mat4);
    st->u_kernel     = bgfx::createUniform("u_kernel",     bgfx::UniformType::Vec4, 24);
    st->u_ssaoParams = bgfx::createUniform("u_ssaoParams", bgfx::UniformType::Vec4);
    st->u_flipY      = bgfx::createUniform("u_flipY",      bgfx::UniformType::Vec4);
    st->u_texel      = bgfx::createUniform("u_texel",      bgfx::UniformType::Vec4);
    st->u_lightDir   = bgfx::createUniform("u_lightDir",   bgfx::UniformType::Vec4);
    st->u_compMode   = bgfx::createUniform("u_compMode",   bgfx::UniformType::Vec4);
    st->s_pos    = bgfx::createUniform("s_pos",    bgfx::UniformType::Sampler);
    st->s_normal = bgfx::createUniform("s_normal", bgfx::UniformType::Sampler);
    st->s_noise  = bgfx::createUniform("s_noise",  bgfx::UniformType::Sampler);
    st->s_albedo = bgfx::createUniform("s_albedo", bgfx::UniformType::Sampler);
    st->s_ao     = bgfx::createUniform("s_ao",     bgfx::UniformType::Sampler);

    // Hemisphere kernel, clustered near the origin.
    for (int i = 0; i < 24; ++i) {
        float sx=SDL_randf()*2-1, sy=SDL_randf()*2-1, sz=SDL_randf();
        float l=SDL_sqrtf(sx*sx+sy*sy+sz*sz); if (l<1e-5f) l=1;
        float s=0.1f+0.9f*(i/24.0f)*(i/24.0f);
        st->kernel[i][0]=sx/l*s; st->kernel[i][1]=sy/l*s; st->kernel[i][2]=sz/l*s; st->kernel[i][3]=0;
    }
    // 4x4 noise of random in-plane rotation vectors, encoded *0.5+0.5 in RGBA8.
    const bgfx::Memory *nm = bgfx::alloc(4*4*4);
    for (int i = 0; i < 16; ++i) {
        float x=SDL_randf()*2-1, y=SDL_randf()*2-1;
        nm->data[i*4+0]=(uint8_t)((x*0.5f+0.5f)*255); nm->data[i*4+1]=(uint8_t)((y*0.5f+0.5f)*255);
        nm->data[i*4+2]=128; nm->data[i*4+3]=255;
    }
    st->noise = bgfx::createTexture2D(4, 4, false, 1, bgfx::TextureFormat::RGBA8,
                                      BGFX_SAMPLER_POINT | BGFX_SAMPLER_U_REPEAT | BGFX_SAMPLER_V_REPEAT, nm);

    makeTargets(st, st->width, st->height);
    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE)  st->mode ^= 1;
    }
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
        makeTargets(st, st->width, st->height);
    }
    return SDL_APP_CONTINUE;
}

static void drawCube(AppState *st, const float *model, float r, float g, float b)
{
    float col[4] = { r, g, b, 1 };
    bgfx::setUniform(st->u_albedo, col);
    bgfx::setTransform(model);
    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(0, st->geom);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    // --- view 0: geometry into the view-space G-buffer ---
    bgfx::setViewFrameBuffer(0, st->gbuffer);
    bgfx::setViewRect(0, 0, 0, (uint16_t)st->fbW, (uint16_t)st->fbH);
    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x00000000, 1.0f, 0);   // pos.z = 0 ⇒ background
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 3.0f, -8.0f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 55.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, view, proj);

    float floor[16], s[16], tr[16];
    bx::mtxScale(s, 14.0f, 0.3f, 14.0f); bx::mtxTranslate(tr, 0, -1.6f, 0); bx::mtxMul(floor, s, tr);
    drawCube(st, floor, 0.85f, 0.85f, 0.88f);
    for (int i = 0; i < 6; ++i) {
        float a = i / 6.0f * 6.2831853f;
        float rot[16], pos[16], model[16];
        bx::mtxRotateY(rot, t + a);
        bx::mtxTranslate(pos, bx::cos(a)*2.4f, -1.0f + (i%2)*0.9f, bx::sin(a)*2.4f);
        bx::mtxMul(model, rot, pos);
        drawCube(st, model, 0.8f, 0.78f, 0.74f);
    }

    float texel[4] = { 1.0f/st->fbW, 1.0f/st->fbH, 0, 0 };

    // --- view 1: SSAO into aoFb ---
    bgfx::setViewFrameBuffer(1, st->aoFb);
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->fbW, (uint16_t)st->fbH);
    bgfx::setUniform(st->u_projMat, proj);
    bgfx::setUniform(st->u_kernel, st->kernel, 24);
    float ssaoParams[4] = { st->fbW / 4.0f, st->fbH / 4.0f, 0.6f, 0.025f };   // noiseScale.xy, radius, bias
    bgfx::setUniform(st->u_ssaoParams, ssaoParams);
    // Match the projected sample UV to how screenTri orients the G-buffer sampling.
    float flip[4] = { bgfx::getCaps()->originBottomLeft ? 0.0f : 1.0f, 0, 0, 0 };
    bgfx::setUniform(st->u_flipY, flip);
    bgfx::setTexture(0, st->s_pos,    bgfx::getTexture(st->gbuffer, 2));
    bgfx::setTexture(1, st->s_normal, bgfx::getTexture(st->gbuffer, 1));
    bgfx::setTexture(2, st->s_noise,  st->noise);
    screenTri(st);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);
    bgfx::submit(1, st->ssao);

    // --- view 2: blur the AO into aoBlurFb ---
    bgfx::setViewFrameBuffer(2, st->aoBlurFb);
    bgfx::setViewRect(2, 0, 0, (uint16_t)st->fbW, (uint16_t)st->fbH);
    bgfx::setUniform(st->u_texel, texel);
    bgfx::setTexture(0, st->s_ao, bgfx::getTexture(st->aoFb, 0));
    screenTri(st);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);
    bgfx::submit(2, st->blur);

    // --- view 3: composite to the backbuffer ---
    bgfx::setViewFrameBuffer(3, BGFX_INVALID_HANDLE);
    bgfx::setViewRect(3, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    bgfx::setViewClear(3, BGFX_CLEAR_COLOR, 0x05060aff, 1.0f, 0);
    float ld[4] = { 0.4f, 0.7f, 0.6f, 0 };                   // view-space light dir
    float cm[4] = { (float)st->mode, 0, 0, 0 };
    bgfx::setUniform(st->u_lightDir, ld);
    bgfx::setUniform(st->u_compMode, cm);
    bgfx::setTexture(0, st->s_albedo, bgfx::getTexture(st->gbuffer, 0));
    bgfx::setTexture(1, st->s_normal, bgfx::getTexture(st->gbuffer, 1));
    bgfx::setTexture(2, st->s_ao,     bgfx::getTexture(st->aoBlurFb, 0));
    screenTri(st);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A);
    bgfx::submit(3, st->comp);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    bgfx::UniformHandle us[] = { st->s_ao, st->s_albedo, st->s_noise, st->s_normal, st->s_pos,
        st->u_compMode, st->u_lightDir, st->u_texel, st->u_flipY, st->u_ssaoParams, st->u_kernel, st->u_projMat, st->u_albedo };
    for (bgfx::UniformHandle u : us) if (bgfx::isValid(u)) bgfx::destroy(u);
    if (bgfx::isValid(st->noise))    bgfx::destroy(st->noise);
    if (bgfx::isValid(st->gbuffer))  bgfx::destroy(st->gbuffer);
    if (bgfx::isValid(st->aoFb))     bgfx::destroy(st->aoFb);
    if (bgfx::isValid(st->aoBlurFb)) bgfx::destroy(st->aoBlurFb);
    if (bgfx::isValid(st->ibh))      bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))      bgfx::destroy(st->vbh);
    bgfx::ProgramHandle ps[] = { st->comp, st->blur, st->ssao, st->geom };
    for (bgfx::ProgramHandle p : ps) if (bgfx::isValid(p)) bgfx::destroy(p);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# vs_gbuffer.sc / fs_gbuffer.sc — view-space position + normal + albedo
# (vs)
$input  a_position, a_normal
$output v_vpos, v_vnormal
#include "bgfx_shader.sh"
void main() {
    vec4 vp     = mul(u_modelView, vec4(a_position, 1.0));
    v_vpos      = vp.xyz;
    v_vnormal   = mul(u_modelView, vec4(a_normal, 0.0)).xyz;
    gl_Position = mul(u_modelViewProj, vec4(a_position, 1.0));
}
# (fs)
$input v_vpos, v_vnormal
#include "bgfx_shader.sh"
uniform vec4 u_albedo;
void main() {
    gl_FragData[0] = u_albedo;                        // albedo
    gl_FragData[1] = vec4(normalize(v_vnormal), 1.0); // view-space normal
    gl_FragData[2] = vec4(v_vpos, 1.0);               // view-space position
}

# fs_ssao.sc — hemisphere kernel occlusion in view space
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_pos,    0);
SAMPLER2D(s_normal, 1);
SAMPLER2D(s_noise,  2);
uniform mat4 u_projMat;
uniform vec4 u_kernel[24];
uniform vec4 u_ssaoParams;                 // xy noiseScale, z radius, w bias
uniform vec4 u_flipY;                      // x > 0.5 : flip the projected sample V
void main() {
    vec3 P = texture2D(s_pos, v_texcoord0).xyz;
    if (P.z == 0.0) { gl_FragColor = vec4(1.0); return; }   // background
    vec3 N  = normalize(texture2D(s_normal, v_texcoord0).xyz);
    vec3 rv = normalize(texture2D(s_noise, v_texcoord0 * u_ssaoParams.xy).xyz * 2.0 - 1.0);
    vec3 T  = normalize(rv - N * dot(rv, N));
    vec3 B  = cross(N, T);
    mat3 TBN = mtxFromCols(T, B, N);
    float occ = 0.0;
    for (int i = 0; i < 24; ++i) {
        vec3 sp  = P + mul(TBN, u_kernel[i].xyz) * u_ssaoParams.z;
        vec4 off = mul(u_projMat, vec4(sp, 1.0));
        off.xyz /= off.w;
        vec2 uv  = off.xy * 0.5 + 0.5;
        if (u_flipY.x > 0.5) uv.y = 1.0 - uv.y;   // match screenTri's G-buffer orientation
        float sd = texture2D(s_pos, uv).z;
        float range = smoothstep(0.0, 1.0, u_ssaoParams.z / abs(P.z - sd));
        occ += (sd >= sp.z + u_ssaoParams.w ? 1.0 : 0.0) * range;
    }
    float ao = 1.0 - occ / 24.0;
    gl_FragColor = vec4(ao, ao, ao, 1.0);
}

# fs_blur.sc / fs_composite.sc
# (blur — 4x4 box that hides the noise tiling)
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_ao, 0);
uniform vec4 u_texel;                       // xy = 1/size
void main() {
    float sum = 0.0;
    for (int x = -2; x < 2; ++x)
        for (int y = -2; y < 2; ++y)
            sum += texture2D(s_ao, v_texcoord0 + vec2(x, y) * u_texel.xy).r;
    gl_FragColor = vec4_splat(sum / 16.0);
}
# (composite — occlusion darkens the ambient term)
$input v_texcoord0
#include "bgfx_shader.sh"
SAMPLER2D(s_albedo, 0);
SAMPLER2D(s_normal, 1);
SAMPLER2D(s_ao,     2);
uniform vec4 u_lightDir;                     // view-space light dir
uniform vec4 u_compMode;                     // x > 0.5 : raw-AO view
void main() {
    float ao = texture2D(s_ao, v_texcoord0).r;
    if (u_compMode.x > 0.5) { gl_FragColor = vec4_splat(ao); return; }
    vec3 N = texture2D(s_normal, v_texcoord0).xyz;
    if (dot(N, N) < 0.5) { gl_FragColor = vec4(0.02, 0.02, 0.03, 1.0); return; }
    vec3 albedo = texture2D(s_albedo, v_texcoord0).rgb;
    float diff = max(dot(normalize(N), normalize(u_lightDir.xyz)), 0.0);
    vec3 col = albedo * 0.35 * ao + albedo * diff * 0.8;
    gl_FragColor = vec4(col, 1.0);
}

Try it yourself

  1. Press Space to view the blurred AO, then bypass the blur view and watch the 4×4 noise grain reappear.
  2. Bind radius and bias in u_ssaoParams to keys and sweep them — too large occludes distant geometry, too small vanishes.
  3. Drop the kernel to 8 samples with a wider blur — often cheaper for a similar look.
  4. Fold AO into recipe 24's full deferred lighting instead of the single directional light here.
