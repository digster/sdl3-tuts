Recipe 22 — Compute Particles

Grow the compute fountain into a system: a flow field swirls 65k particles, additive blending makes them glow, a speed-driven colour ramp gives them heat, and the mouse becomes an attractor you can stir the field with.

What you'll learn

  • Advecting particles through a procedural flow field instead of plain gravity.
  • An interactive attractor — feeding the mouse position into the compute kernel.
  • Additive blending so overlapping particles accumulate into glow.
  • Colouring points by speed and life for a hot-to-cold look — all on the GPU.

Prerequisites

Recipe 16 — Compute Shaders built the single-force fountain and the dispatch-then-draw plumbing; this recipe keeps that exact mechanism and changes what the kernel does. Recipe 20 covers the dynamic buffer the particles live in.

Walkthrough

From one force to a field

Recipe 16 pushed every particle with the same downward gravity. A flow field instead gives a different velocity at every point in space. A cheap curl-like field from a few sines costs nothing and produces organic, turbulent motion — no texture, no noise table. 

# cs_flow.sc — velocity from position
vec3 flow(vec3 p) {
    return vec3(sin(p.y*1.3 + p.z*0.7),
                sin(p.z*1.1 + p.x*0.9),
                sin(p.x*1.2 + p.y*0.8));
}

An attractor from the mouse

Feed an attractor position and strength in through a uniform. A linear pull toward it (no normalize, so there's no singularity at the centre) lets you stir the whole swarm by moving the mouse with the button held. 

uniform vec4 u_params;     // x = dt, y = count, z = time, w = attract strength
uniform vec4 u_attract;    // xyz = attractor position
// ...inside main():
vec3 force  = flow(p.xyz) * 1.2;
force      += (u_attract.xyz - p.xyz) * u_params.w;   // pull toward the mouse
v.xyz       = v.xyz * 0.96 + force * u_params.x;       // damping keeps it from exploding
p.xyz      += v.xyz * u_params.x;

Additive blending makes glow

Drawing the points with BGFX_STATE_BLEND_ADD and depth-write off means overlapping particles sum their colour. Dense regions bloom to white while sparse wisps stay dim — the cheapest possible glow, and the reason particle effects almost always blend additively. 

bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A
             | BGFX_STATE_PT_POINTS | BGFX_STATE_BLEND_ADD);
The look is all in the kernel + blend. The C++ side is nearly identical to recipe 16 — same dynamic buffer, same dispatch-then-draw across two views. Everything that makes this feel alive lives in the compute shader's forces and the additive draw state.

Complete listing

65,536 particles advected through a flow field, glowing additively, hot where they move fast. Hold the left mouse button and move to drag an attractor through the swarm. Native-handle / loadProgram helpers identical to recipe 03 (elided). Shaders: cs_flow (compute), vs_particle / fs_particle. 

/*
 * bgfx/22-compute-particles.cpp — a flow-field particle system with an interactive
 * attractor, drawn additively. Mechanism (dynamic buffer, dispatch-then-draw) from
 * recipe 16. Shaders: cs_flow (compute), vs_particle / fs_particle.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>
#include <vector>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);
static bgfx::ProgramHandle loadComputeProgram(const char *cs);

static const int N = 65536;
struct Particle { float pos[4]; float vel[4]; };   // pos.xyz + life, vel.xyz

struct AppState {
    SDL_Window *window = nullptr; int width = 1280, height = 720;
    bgfx::DynamicVertexBufferHandle buffer = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle compute = BGFX_INVALID_HANDLE, render = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_params = BGFX_INVALID_HANDLE, u_attract = BGFX_INVALID_HANDLE;
    float mouseX = 0, mouseY = 0; bool attracting = false;
    Uint64 startNs = 0, lastNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Compute Particles", "1.0", "com.example.sdl3.bgfx.particles");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Compute Particles", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");
    if (!(bgfx::getCaps()->supported & BGFX_CAPS_COMPUTE)) return fail("compute unsupported");

    bgfx::VertexLayout layout;
    layout.begin()
        .add(bgfx::Attrib::Position,  4, bgfx::AttribType::Float)
        .add(bgfx::Attrib::TexCoord0, 4, bgfx::AttribType::Float).end();

    std::vector<Particle> seed(N);
    for (int i = 0; i < N; ++i)
        seed[i] = { { (SDL_randf()-0.5f)*6, (SDL_randf()-0.5f)*4, (SDL_randf()-0.5f)*6, SDL_randf()*2.0f },
                    { 0, 0, 0, 0 } };
    st->buffer = bgfx::createDynamicVertexBuffer(
        bgfx::copy(seed.data(), uint32_t(seed.size()*sizeof(Particle))), layout,
        BGFX_BUFFER_COMPUTE_READ | BGFX_BUFFER_COMPUTE_WRITE);

    st->compute  = loadComputeProgram("cs_flow");
    st->render   = loadProgram("vs_particle", "fs_particle");
    st->u_params = bgfx::createUniform("u_params",  bgfx::UniformType::Vec4);
    st->u_attract= bgfx::createUniform("u_attract", bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->compute) || !bgfx::isValid(st->render)) return fail("loadProgram");

    st->startNs = st->lastNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_MOUSE_MOTION) { st->mouseX = e->motion.x; st->mouseY = e->motion.y; }
    if (e->type == SDL_EVENT_MOUSE_BUTTON_DOWN && e->button.button == SDL_BUTTON_LEFT) st->attracting = true;
    if (e->type == SDL_EVENT_MOUSE_BUTTON_UP   && e->button.button == SDL_BUTTON_LEFT) st->attracting = false;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->lastNs) / 1.0e9f; if (dt > 0.05f) dt = 0.05f;
    st->lastNs = now;
    float time = (now - st->startNs) / 1.0e9f;

    // --- view 0: simulate ---
    // Map the mouse (window pixels) to a point on the z = 0 plane, roughly framing the swarm.
    int ww = st->width > 0 ? st->width : 1, wh = st->height > 0 ? st->height : 1;
    float ax = ((float)st->mouseX / ww * 2.0f - 1.0f) * 6.0f;
    float ay = -((float)st->mouseY / wh * 2.0f - 1.0f) * 4.0f;
    float attract[4] = { ax, ay, 0.0f, 0.0f };
    float params[4]  = { dt, (float)N, time, st->attracting ? 3.0f : 0.0f };
    bgfx::setUniform(st->u_params,  params);
    bgfx::setUniform(st->u_attract, attract);
    bgfx::setBuffer(0, st->buffer, bgfx::Access::ReadWrite);
    bgfx::dispatch(0, st->compute, (N + 63) / 64, 1, 1);

    // --- view 1: draw the same buffer additively ---
    bgfx::setViewClear(1, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x03040aff, 1.0f, 0);
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 0, -11.0f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 60.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(1, view, proj);

    float mtx[16]; bx::mtxIdentity(mtx);
    bgfx::setTransform(mtx);
    bgfx::setVertexBuffer(0, st->buffer);
    bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A
                 | BGFX_STATE_PT_POINTS | BGFX_STATE_BLEND_ADD);
    bgfx::submit(1, st->render);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (bgfx::isValid(st->u_attract)) bgfx::destroy(st->u_attract);
    if (bgfx::isValid(st->u_params))  bgfx::destroy(st->u_params);
    if (bgfx::isValid(st->buffer))    bgfx::destroy(st->buffer);
    if (bgfx::isValid(st->render))    bgfx::destroy(st->render);
    if (bgfx::isValid(st->compute))   bgfx::destroy(st->compute);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# cs_flow.sc — advect through a flow field + mouse attractor, respawn on death
#include "bgfx_compute.sh"
BUFFER_RW(b_particles, vec4, 0);
uniform vec4 u_params;      // x = dt, y = count, z = time, w = attract strength
uniform vec4 u_attract;     // xyz = attractor position
NUM_THREADS(64, 1, 1)

vec3 flow(vec3 p) {
    return vec3(sin(p.y*1.3 + p.z*0.7),
                sin(p.z*1.1 + p.x*0.9),
                sin(p.x*1.2 + p.y*0.8));
}
void main() {
    uint id = gl_GlobalInvocationID.x;
    if (float(id) >= u_params.y) return;
    vec4 p = b_particles[id*2u + 0u];        // xyz pos, w life
    vec4 v = b_particles[id*2u + 1u];        // xyz velocity

    vec3 force = flow(p.xyz) * 1.2;
    force += (u_attract.xyz - p.xyz) * u_params.w;   // linear pull, no singularity
    v.xyz = v.xyz * 0.96 + force * u_params.x;        // damping
    p.xyz += v.xyz * u_params.x;
    p.w   -= u_params.x;

    if (p.w <= 0.0) {                          // respawn in the seed volume
        float r  = fract(sin(float(id)*12.9898 + u_params.z) * 43758.5453);
        float r2 = fract(sin(float(id)*78.233  + u_params.z) * 43758.5453);
        float r3 = fract(sin(float(id)*37.719  + u_params.z) * 43758.5453);
        p = vec4((r-0.5)*6.0, (r2-0.5)*4.0, (r3-0.5)*6.0, 1.5 + 2.0*r);
        v = vec4_splat(0.0);
    }
    b_particles[id*2u + 0u] = p;
    b_particles[id*2u + 1u] = v;
}

# vs_particle.sc / fs_particle.sc — colour by speed and life
# (vs)
$input  a_position, a_texcoord0
$output v_color0
#include "bgfx_shader.sh"
void main() {
    gl_Position = mul(u_modelViewProj, vec4(a_position.xyz, 1.0));
    float life = clamp(a_position.w, 0.0, 1.0);
    float spd  = length(a_texcoord0.xyz);
    vec3 cold  = vec3(0.10, 0.30, 0.90);
    vec3 hot   = vec3(1.00, 0.60, 0.20);
    v_color0   = vec4(mix(cold, hot, clamp(spd*0.4, 0.0, 1.0)) * (0.25 + 0.75*life), 1.0);
}
# (fs)
$input v_color0
#include "bgfx_shader.sh"
void main() { gl_FragColor = v_color0; }

Try it yourself

  1. Make it true curl noise: sample a 3D noise texture's gradient and take its curl for a divergence-free field that never bunches up.
  2. Add a repulsor on the right mouse button (negate the attractor strength) and push particles away.
  3. Trade points for soft sprites: expand each particle into a camera-facing quad (recipe 11 instancing) and sample a radial-falloff texture for round, soft blobs.
  4. Drive the colour ramp from acceleration instead of speed — particles flash where the field changes direction sharply.
