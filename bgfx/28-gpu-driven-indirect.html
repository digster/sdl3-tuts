Recipe 28 — GPU-Driven Indirect Rendering

The bgfx track's finale. A compute shader culls a huge instance set, compacts the survivors, and writes the draw call itself into an indirect buffer. The CPU issues one submit and never learns how many instances survived — the GPU drives its own rendering.

What you'll learn

  • Compute-side culling with an atomic append into a compacted instance buffer.
  • Writing draw arguments from a shader with drawIndexedIndirect into a createIndirectBuffer.
  • Issuing the draw with bgfx::submit(view, prog, indirectHandle, …) — count decided on the GPU.
  • The compute→draw barrier bgfx inserts for you, and the BGFX_CAPS_DRAW_INDIRECT gate.

Prerequisites

Recipe 16 — Compute Shaders for dispatch and compute-readable/writable buffers, recipe 11 for instanced drawing, and 3D recipe 29 for the culling idea — moved here from the CPU onto the GPU.

Walkthrough

The indirect buffer holds the draw call

bgfx::createIndirectBuffer(n) reserves storage for n indexed-draw argument sets — index count, instance count, offsets. A compute shader fills it with the drawIndexedIndirect helper, and bgfx::submit(view, program, indirectHandle, start, num) executes those draws. The CPU issues the submit blind; the GPU supplies the numbers. 

st->indirect = bgfx::createIndirectBuffer(1);        // room for one indexed draw
// ... compute fills it ...
bgfx::setInstanceDataBuffer(st->instOut, 0, MAX_INST);
bgfx::submit(1, st->render, st->indirect, 0, 1);      // one indirect draw

Cull and compact in compute

One thread per candidate instance tests visibility. Survivors claim a slot in the output buffer with an atomic add on a counter and write their compacted instance data there — the same stream-compaction pattern real GPU-driven engines use. 

# cs_cull.sc
#include "bgfx_compute.sh"
BUFFER_RO(instanceIn,  vec4, 0);      // all candidates: xyz position, w hue
BUFFER_WR(instanceOut, vec4, 1);      // compacted survivors
BUFFER_RW(drawCount,   uint, 2);      // single atomic counter
uniform vec4 u_cullTotal;             // x = candidate count
uniform vec4 u_cullSphere;            // xyz centre, w radius
NUM_THREADS(64, 1, 1)
void main() {
    uint id = gl_GlobalInvocationID.x;
    if (id >= uint(u_cullTotal.x)) return;
    vec4 inst = instanceIn[id];
    if (length(inst.xyz - u_cullSphere.xyz) > u_cullSphere.w) return;  // culled
    uint slot = atomicFetchAndAdd(drawCount[0], 1u);                   // claim a slot
    instanceOut[slot] = inst;                                          // compact
}

Write the draw args, reset the counter

A one-thread pass reads the survivor count and writes the indexed-draw arguments, then zeroes the counter so next frame's cull starts clean. That single reset is what keeps the whole thing to two dispatches and no CPU readback. 

# cs_genargs.sc
#include "bgfx_compute.sh"
BUFFER_RW(drawCount,      uint,  0);
BUFFER_WR(indirectBuffer, uvec4, 1);
uniform vec4 u_drawParams;            // x = index count of the mesh
NUM_THREADS(1, 1, 1)
void main() {
    uint count = drawCount[0];
    drawIndexedIndirect(indirectBuffer, 0u, uint(u_drawParams.x), count, 0u, 0u, 0u);
    drawCount[0] = 0u;                 // prime for next frame
}
bgfx inserts the barriers. The counter and buffers are written by compute and read by the next dispatch or the draw; bgfx sees the dependency and synchronises for you — exactly as it did for the compute particles in recipe 16. Needs bgfx::getCaps()->supported & BGFX_CAPS_DRAW_INDIRECT.

Complete listing

Thousands of cubes fill a volume, but only those inside a roaming sphere are culled-in, compacted and drawn — the sphere's contents pop in and out as it moves, all decided on the GPU in one indirect draw. Native-handle / loadProgram / loadComputeProgram helpers identical to recipes 03 and 16 (elided). Shaders: cs_cull, cs_genargs, vs_render / fs_render. 

/*
 * bgfx/28-gpu-driven-indirect.cpp — compute culls + compacts instances and writes
 * an indirect draw; the CPU issues one submit with a GPU-decided instance count.
 * Native-handle + loadProgram/loadComputeProgram helpers as recipes 03/16 (elided).
 * Shaders: cs_cull, cs_genargs, vs_render/fs_render.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>
#include <vector>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);
static bgfx::ProgramHandle loadComputeProgram(const char *cs);

struct PNVertex { float px, py, pz; float nx, ny, nz; };
static const PNVertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const uint16_t kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

static const int SIDE = 20;                         // SIDE^3 candidate instances
static const int MAX_INST = SIDE * SIDE * SIDE;     // 8000

struct AppState {
    SDL_Window *window = nullptr; int width = 1280, height = 720;
    bgfx::VertexLayout meshLayout, instLayout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::VertexBufferHandle instIn = BGFX_INVALID_HANDLE;           // all candidates (compute-read)
    bgfx::DynamicVertexBufferHandle instOut = BGFX_INVALID_HANDLE;   // compacted (compute-write + instance data)
    bgfx::DynamicIndexBufferHandle  drawCount = BGFX_INVALID_HANDLE; // single atomic counter
    bgfx::IndirectBufferHandle indirect = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle cull = BGFX_INVALID_HANDLE, genargs = BGFX_INVALID_HANDLE, render = BGFX_INVALID_HANDLE;
    bgfx::UniformHandle u_cullTotal = BGFX_INVALID_HANDLE, u_cullSphere = BGFX_INVALID_HANDLE, u_drawParams = BGFX_INVALID_HANDLE, u_lightDir = BGFX_INVALID_HANDLE;
    bool ok = false; Uint64 startNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx GPU-Driven", "1.0", "com.example.sdl3.bgfx.gdr");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");
    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — GPU-Driven Indirect Rendering", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type = bgfx::RendererType::Count;
    init.platformData.nwh = nativeWindowHandle(st->window);
    init.platformData.ndt = nativeDisplayHandle(st->window);
    init.resolution.width = (uint32_t)st->width; init.resolution.height = (uint32_t)st->height;
    init.resolution.reset = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");
    bgfx::setDebug(BGFX_DEBUG_TEXT);

    uint32_t caps = bgfx::getCaps()->supported;
    st->ok = (caps & BGFX_CAPS_COMPUTE) && (caps & BGFX_CAPS_DRAW_INDIRECT);
    if (!st->ok) SDL_Log("GPU-driven rendering needs BGFX_CAPS_COMPUTE + BGFX_CAPS_DRAW_INDIRECT");

    st->meshLayout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Normal,   3, bgfx::AttribType::Float).end();
    // One vec4 per instance (xyz position, w hue), read in compute and fed as instance data.
    st->instLayout.begin().add(bgfx::Attrib::TexCoord7, 4, bgfx::AttribType::Float).end();

    st->vbh = bgfx::createVertexBuffer(bgfx::makeRef(kCube, sizeof(kCube)), st->meshLayout);
    st->ibh = bgfx::createIndexBuffer (bgfx::makeRef(kIdx, sizeof(kIdx)));

    // Candidate instances: a SIDE^3 lattice, compute-readable.
    std::vector<float> cand(MAX_INST * 4);
    for (int i = 0; i < MAX_INST; ++i) {
        int x = i % SIDE, y = (i / SIDE) % SIDE, z = i / (SIDE*SIDE);
        cand[i*4+0] = (x - SIDE*0.5f) * 1.6f;
        cand[i*4+1] = (y - SIDE*0.5f) * 1.6f;
        cand[i*4+2] = (z - SIDE*0.5f) * 1.6f;
        cand[i*4+3] = (float)i / MAX_INST;                 // hue
    }
    st->instIn = bgfx::createVertexBuffer(
        bgfx::copy(cand.data(), (uint32_t)(cand.size()*sizeof(float))), st->instLayout,
        BGFX_BUFFER_COMPUTE_READ);
    st->instOut = bgfx::createDynamicVertexBuffer(MAX_INST, st->instLayout,
        BGFX_BUFFER_COMPUTE_WRITE);
    st->drawCount = bgfx::createDynamicIndexBuffer(1,
        BGFX_BUFFER_COMPUTE_READ_WRITE | BGFX_BUFFER_INDEX32);   // starts zeroed
    st->indirect = bgfx::createIndirectBuffer(1);

    st->cull    = loadComputeProgram("cs_cull");
    st->genargs = loadComputeProgram("cs_genargs");
    st->render  = loadProgram("vs_render", "fs_render");
    st->u_cullTotal  = bgfx::createUniform("u_cullTotal",  bgfx::UniformType::Vec4);
    st->u_cullSphere = bgfx::createUniform("u_cullSphere", bgfx::UniformType::Vec4);
    st->u_drawParams = bgfx::createUniform("u_drawParams", bgfx::UniformType::Vec4);
    st->u_lightDir   = bgfx::createUniform("u_lightDir",   bgfx::UniformType::Vec4);
    if (!bgfx::isValid(st->cull) || !bgfx::isValid(st->render)) return fail("loadProgram");

    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    if (!st->ok) {                                             // graceful fallback message
        bgfx::setViewRect(0, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
        bgfx::setViewClear(0, BGFX_CLEAR_COLOR, 0x201010ff, 1.0f, 0);
        bgfx::touch(0);
        bgfx::dbgTextClear();
        bgfx::dbgTextPrintf(1, 1, 0x0c, "BGFX_CAPS_DRAW_INDIRECT / COMPUTE unsupported on this backend");
        bgfx::frame();
        return SDL_APP_CONTINUE;
    }

    // A roaming cull sphere so the visible set changes every frame.
    float cx = SDL_cosf(t*0.5f) * 8.0f, cy = SDL_sinf(t*0.7f) * 6.0f, cz = SDL_sinf(t*0.4f) * 8.0f;
    float total[4] = { (float)MAX_INST, 0, 0, 0 };
    float sphere[4] = { cx, cy, cz, 7.0f };
    float draw[4]  = { 36.0f, 0, 0, 0 };                      // index count of the cube

    // --- Pass A (view 0, compute): cull + compact ---
    bgfx::setUniform(st->u_cullTotal,  total);
    bgfx::setUniform(st->u_cullSphere, sphere);
    bgfx::setBuffer(0, st->instIn,    bgfx::Access::Read);
    bgfx::setBuffer(1, st->instOut,   bgfx::Access::Write);
    bgfx::setBuffer(2, st->drawCount, bgfx::Access::ReadWrite);
    bgfx::dispatch(0, st->cull, (MAX_INST + 63) / 64, 1, 1);

    // --- Pass B (view 0, compute): write the indirect draw args ---
    bgfx::setUniform(st->u_drawParams, draw);
    bgfx::setBuffer(0, st->drawCount, bgfx::Access::ReadWrite);
    bgfx::setBuffer(1, st->indirect,  bgfx::Access::Write);
    bgfx::dispatch(0, st->genargs, 1, 1, 1);

    // --- Render (view 1): one indirect instanced draw ---
    bgfx::setViewRect(1, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    bgfx::setViewClear(1, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x0a0c12ff, 1.0f, 0);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 6.0f, -30.0f };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 60.0f, float(st->width)/float(st->height), 0.1f, 200.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(1, view, proj);
    float ld[4] = { -0.4f, -0.7f, -0.5f, 0 };
    bgfx::setUniform(st->u_lightDir, ld);

    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
    bgfx::setInstanceDataBuffer(st->instOut, 0, MAX_INST);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(1, st->render, st->indirect, 0, 1);           // GPU decides the instance count

    bgfx::dbgTextClear();
    bgfx::dbgTextPrintf(1, 1, 0x0f, "GPU-driven: %d candidates, culled + drawn on the GPU", MAX_INST);
    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    bgfx::UniformHandle us[] = { st->u_lightDir, st->u_drawParams, st->u_cullSphere, st->u_cullTotal };
    for (bgfx::UniformHandle u : us) if (bgfx::isValid(u)) bgfx::destroy(u);
    if (bgfx::isValid(st->indirect))  bgfx::destroy(st->indirect);
    if (bgfx::isValid(st->drawCount)) bgfx::destroy(st->drawCount);
    if (bgfx::isValid(st->instOut))   bgfx::destroy(st->instOut);
    if (bgfx::isValid(st->instIn))    bgfx::destroy(st->instIn);
    if (bgfx::isValid(st->ibh))       bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh))       bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->render))    bgfx::destroy(st->render);
    if (bgfx::isValid(st->genargs))   bgfx::destroy(st->genargs);
    if (bgfx::isValid(st->cull))      bgfx::destroy(st->cull);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

# cs_cull.sc — one thread per candidate; keep those inside the sphere, compact them
#include "bgfx_compute.sh"
BUFFER_RO(instanceIn,  vec4, 0);
BUFFER_WR(instanceOut, vec4, 1);
BUFFER_RW(drawCount,   uint, 2);
uniform vec4 u_cullTotal;             // x = candidate count
uniform vec4 u_cullSphere;            // xyz centre, w radius
NUM_THREADS(64, 1, 1)
void main() {
    uint id = gl_GlobalInvocationID.x;
    if (id >= uint(u_cullTotal.x)) return;
    vec4 inst = instanceIn[id];
    if (length(inst.xyz - u_cullSphere.xyz) > u_cullSphere.w) return;   // outside: cull
    uint slot = atomicFetchAndAdd(drawCount[0], 1u);                    // claim output slot
    instanceOut[slot] = inst;
}

# cs_genargs.sc — turn the survivor count into an indexed indirect draw
#include "bgfx_compute.sh"
BUFFER_RW(drawCount,      uint,  0);
BUFFER_WR(indirectBuffer, uvec4, 1);
uniform vec4 u_drawParams;            // x = index count of the mesh
NUM_THREADS(1, 1, 1)
void main() {
    uint count = drawCount[0];
    // (buffer, index, numIndices, numInstances, startIndex, startVertex, startInstance)
    drawIndexedIndirect(indirectBuffer, 0u, uint(u_drawParams.x), count, 0u, 0u, 0u);
    drawCount[0] = 0u;                // reset for next frame
}

# vs_render.sc / fs_render.sc — instanced cube coloured by its hue
# (vs)
$input  a_position, a_normal, i_data0
$output v_normal, v_color
#include "bgfx_shader.sh"
void main() {
    vec3 world  = a_position + i_data0.xyz;          // per-instance offset
    gl_Position = mul(u_viewProj, vec4(world, 1.0));
    v_normal    = a_normal;
    float h = i_data0.w * 6.2831;
    v_color = vec3(0.5+0.5*sin(h), 0.5+0.5*sin(h+2.09), 0.5+0.5*sin(h+4.19));
}
# (fs)
$input v_normal, v_color
#include "bgfx_shader.sh"
uniform vec4 u_lightDir;
void main() {
    float diff = max(dot(normalize(v_normal), normalize(-u_lightDir.xyz)), 0.0);
    gl_FragColor = vec4(v_color * (0.3 + 0.75 * diff), 1.0);
}

Try it yourself

  1. Swap the sphere test for the recipe-29 frustum planes (pass them as uniforms) so the GPU does true frustum culling.
  2. Read the count back: bgfx::createDynamicIndexBuffer with a readback, or blit the counter to a CPU-visible buffer, and print the live survivor count.
  3. Per-instance LOD: pick one of several meshes per instance and emit several indirect draws (one per LOD) from cs_genargs.
  4. Push SIDE to 40 (64 000 candidates). The CPU cost doesn't move — it still issues exactly one submit.
