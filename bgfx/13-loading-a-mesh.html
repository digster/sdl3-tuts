Recipe 13 — Loading a Mesh

Stop hand-typing vertices. Parse a real model — vertices and faces — into bgfx vertex and index buffers, and draw something that isn't a cube.

What you'll learn

  • The shape of a Wavefront .obj file: v lines and f lines.
  • Turning parsed positions and faces into createVertexBuffer / createIndexBuffer data.
  • Why bgfx::copy (not makeRef) is the safe choice for data you build at runtime.
  • Deriving a per-vertex normal so the model reads as 3D, not a silhouette.

Prerequisites

Recipe 05 for static buffers and the PosColorVertex layout, and recipe 07 for the MVP setup we reuse. Shaders vs_cubes / fs_cubes are unchanged from recipe 07.

Walkthrough

What's in an .obj

Wavefront OBJ is the "hello world" of model formats — plain text, one element per line. Two line types matter for a basic mesh: v x y z declares a vertex position, and f a b c declares a triangular face referencing those vertices by 1-based index. (Real files add vn normals and vt UVs, and faces can read a/b/c; we keep to positions here.) 

# a fragment of the embedded icosahedron
v -1 1.618 0
v  1 1.618 0
...
f 1 12 6
f 1 6 2

A tiny parser

Walk the text line by line. Push every v into a position list; for every f, push three indices (minus one, to go 0-based). In a real app you'd SDL_LoadFile the .obj from disk first — here the model is an embedded string so the recipe stays asset-free, but the parser is the same either way. 

static void parse_obj(const char *text,
                     std::vector<PosColorVertex> &verts, std::vector<uint16_t> &idx)
{
    std::vector<float[3]> pos;            // collected v lines
    const char *p = text; char line[128];
    while (*p) {
        int n = 0; while (p[n] && p[n] != '\n') ++n;
        SDL_strlcpy(line, p, (size_t)SDL_min(n + 1, (int)sizeof(line)));
        if (line[0] == 'v' && line[1] == ' ') {
            float v[3]; SDL_sscanf(line + 2, "%f %f %f", &v[0], &v[1], &v[2]);
            pos.push_back({ v[0], v[1], v[2] });
        } else if (line[0] == 'f' && line[1] == ' ') {
            int a, b, c; SDL_sscanf(line + 2, "%d %d %d", &a, &b, &c);
            idx.push_back((uint16_t)(a - 1));
            idx.push_back((uint16_t)(b - 1));
            idx.push_back((uint16_t)(c - 1));
        }
        p += n; if (*p == '\n') ++p;
    }
    // Per-vertex normal = normalised position (fine for a centred convex shell),
    // packed into the colour so we can reuse vs_cubes/fs_cubes.
    for (auto &v : pos) {
        float l = SDL_sqrtf(v[0]*v[0] + v[1]*v[1] + v[2]*v[2]); if (l == 0) l = 1;
        uint8_t r = (uint8_t)((v[0]/l*0.5f + 0.5f)*255);
        uint8_t g = (uint8_t)((v[1]/l*0.5f + 0.5f)*255);
        uint8_t b = (uint8_t)((v[2]/l*0.5f + 0.5f)*255);
        verts.push_back({ v[0], v[1], v[2], 0xff000000u | (b<<16) | (g<<8) | r });
    }
}

copy, not makeRef

Recipe 05 used bgfx::makeRef because the data was a static const array that outlives every frame. parse_obj fills local vectors that vanish when init returns — makeRef would dangle. bgfx::copy duplicates the bytes into bgfx-owned memory, so the source can die immediately. 

st->vbh = bgfx::createVertexBuffer(
    bgfx::copy(verts.data(), uint32_t(verts.size() * sizeof(PosColorVertex))), st->layout);
st->ibh = bgfx::createIndexBuffer(
    bgfx::copy(idx.data(), uint32_t(idx.size() * sizeof(uint16_t))));
Rule of thumb. makeRef for data you guarantee stays put (globals, long-lived pools); copy for anything transient. When unsure, copy — a one-time duplication is cheap insurance against a crash.

Complete listing

An icosahedron embedded as an OBJ string, parsed at startup into bgfx buffers and spun in front of the camera. Swap the string for an SDL_LoadFile'd model and nothing else changes. 

/*
 * bgfx/13-loading-a-mesh.cpp — parse an embedded .obj into bgfx buffers.
 * Shaders: vs_cubes / fs_cubes (recipe 07).
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <bgfx/bgfx.h>
#include <bgfx/platform.h>
#include <bx/math.h>
#include <cstdint>
#include <vector>

static void *nativeWindowHandle(SDL_Window *);
static void *nativeDisplayHandle(SDL_Window *);
static bgfx::ProgramHandle loadProgram(const char *vs, const char *fs);

struct PosColorVertex { float x, y, z; uint32_t abgr; };

static const char *kObj = R"obj(
v -1 1.618 0
v 1 1.618 0
v -1 -1.618 0
v 1 -1.618 0
v 0 -1 1.618
v 0 1 1.618
v 0 -1 -1.618
v 0 1 -1.618
v 1.618 0 -1
v 1.618 0 1
v -1.618 0 -1
v -1.618 0 1
f 1 12 6
f 1 6 2
f 1 2 8
f 1 8 11
f 1 11 12
f 2 6 10
f 6 12 5
f 12 11 3
f 11 8 7
f 8 2 9
f 4 10 5
f 4 5 3
f 4 3 7
f 4 7 9
f 4 9 10
f 5 10 6
f 3 5 12
f 7 3 11
f 9 7 8
f 10 9 2
)obj";

static void parse_obj(const char *text, std::vector<PosColorVertex> &verts, std::vector<uint16_t> &idx)
{
    std::vector<bx::Vec3> pos;
    const char *p = text; char line[128];
    while (*p) {
        int n = 0; while (p[n] && p[n] != '\n') ++n;
        SDL_strlcpy(line, p, (size_t)SDL_min(n + 1, (int)sizeof(line)));
        if (line[0] == 'v' && line[1] == ' ') {
            float v[3]; SDL_sscanf(line + 2, "%f %f %f", &v[0], &v[1], &v[2]);
            pos.push_back({ v[0], v[1], v[2] });
        } else if (line[0] == 'f' && line[1] == ' ') {
            int a, b, c; SDL_sscanf(line + 2, "%d %d %d", &a, &b, &c);
            idx.push_back((uint16_t)(a - 1)); idx.push_back((uint16_t)(b - 1)); idx.push_back((uint16_t)(c - 1));
        }
        p += n; if (*p == '\n') ++p;
    }
    for (auto &v : pos) {
        float l = SDL_sqrtf(v.x*v.x + v.y*v.y + v.z*v.z); if (l == 0) l = 1;
        uint8_t r = (uint8_t)((v.x/l*0.5f + 0.5f)*255);
        uint8_t g = (uint8_t)((v.y/l*0.5f + 0.5f)*255);
        uint8_t b = (uint8_t)((v.z/l*0.5f + 0.5f)*255);
        verts.push_back({ v.x, v.y, v.z, 0xff000000u | ((uint32_t)b<<16) | ((uint32_t)g<<8) | r });
    }
}

struct AppState {
    SDL_Window *window = nullptr;
    int width = 1280, height = 720;
    bgfx::VertexLayout layout;
    bgfx::VertexBufferHandle vbh = BGFX_INVALID_HANDLE;
    bgfx::IndexBufferHandle  ibh = BGFX_INVALID_HANDLE;
    bgfx::ProgramHandle program = BGFX_INVALID_HANDLE;
    uint32_t indexCount = 0;
    Uint64 startNs = 0;
};

static SDL_AppResult fail(const char *what) { SDL_Log("%s: %s", what, SDL_GetError()); return SDL_APP_FAILURE; }

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("bgfx Mesh", "1.0", "com.example.sdl3.bgfx.mesh");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("SDL_Init");

    AppState *st = new AppState(); *appstate = st;
    st->window = SDL_CreateWindow("bgfx — Loading a Mesh", st->width, st->height,
                                  SDL_WINDOW_RESIZABLE | SDL_WINDOW_HIGH_PIXEL_DENSITY);
    if (!st->window) return fail("SDL_CreateWindow");
    SDL_GetWindowSizeInPixels(st->window, &st->width, &st->height);

    bgfx::renderFrame();
    bgfx::Init init;
    init.type              = bgfx::RendererType::Count;
    init.platformData.nwh  = nativeWindowHandle(st->window);
    init.platformData.ndt  = nativeDisplayHandle(st->window);
    init.resolution.width  = (uint32_t)st->width;
    init.resolution.height = (uint32_t)st->height;
    init.resolution.reset  = BGFX_RESET_VSYNC;
    if (!bgfx::init(init)) return fail("bgfx::init");

    bgfx::setViewClear(0, BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH, 0x121a30ff, 1.0f, 0);

    st->layout.begin()
        .add(bgfx::Attrib::Position, 3, bgfx::AttribType::Float)
        .add(bgfx::Attrib::Color0,   4, bgfx::AttribType::Uint8, true).end();

    std::vector<PosColorVertex> verts; std::vector<uint16_t> idx;
    parse_obj(kObj, verts, idx);
    st->indexCount = (uint32_t)idx.size();
    st->vbh = bgfx::createVertexBuffer(bgfx::copy(verts.data(), uint32_t(verts.size()*sizeof(PosColorVertex))), st->layout);
    st->ibh = bgfx::createIndexBuffer (bgfx::copy(idx.data(),   uint32_t(idx.size()*sizeof(uint16_t))));
    st->program = loadProgram("vs_cubes", "fs_cubes");
    if (!bgfx::isValid(st->program)) return fail("loadProgram");

    SDL_Log("loaded mesh: %zu verts, %zu tris", verts.size(), idx.size()/3);
    st->startNs = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_WINDOW_PIXEL_SIZE_CHANGED) {
        st->width = e->window.data1; st->height = e->window.data2;
        bgfx::reset((uint32_t)st->width, (uint32_t)st->height, BGFX_RESET_VSYNC);
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (float)(SDL_GetTicksNS() - st->startNs) / 1.0e9f;

    bgfx::setViewRect(0, 0, 0, (uint16_t)st->width, (uint16_t)st->height);
    const bx::Vec3 at = { 0, 0, 0 }, eye = { 0, 0, -6 };
    float view[16], proj[16];
    bx::mtxLookAt(view, eye, at);
    bx::mtxProj(proj, 60.0f, float(st->width)/float(st->height), 0.1f, 100.0f, bgfx::getCaps()->homogeneousDepth);
    bgfx::setViewTransform(0, view, proj);

    float mtx[16];
    bx::mtxRotateXY(mtx, t * 0.5f, t * 0.7f);
    bgfx::setTransform(mtx);
    bgfx::setVertexBuffer(0, st->vbh);
    bgfx::setIndexBuffer(st->ibh);
    bgfx::setState(BGFX_STATE_DEFAULT);
    bgfx::submit(0, st->program);

    bgfx::frame();
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (bgfx::isValid(st->ibh)) bgfx::destroy(st->ibh);
    if (bgfx::isValid(st->vbh)) bgfx::destroy(st->vbh);
    if (bgfx::isValid(st->program)) bgfx::destroy(st->program);
    bgfx::shutdown();
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Load from disk: SDL_LoadFile("model.obj", &len), hand the buffer to parse_obj, free it after the buffers are built.
  2. Handle real faces. Split each f token on / to ignore the UV/normal indices, and triangulate quads (f a b c d → two triangles).
  3. Read the file's own vn normals instead of faking them, add a Normal attribute to the layout, and light the model with recipe 12's shader.
  4. Centre and scale arbitrary models: scan the positions for their bounding box, then offset and normalise so anything fits the view.
