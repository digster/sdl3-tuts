Recipe 24 — Environment Reflections

Turn a surface into a mirror. Reflect the view direction about the surface normal, sample the same cubemap that fills the sky, and a chrome cube mirrors its surroundings — with a Fresnel rim so grazing angles glint.

What you'll learn

  • The reflection vector R = reflect(I, N) and what I and N must be.
  • Sampling a cubemap by a direction to fetch the environment colour along R.
  • Reusing one cubemap for both the skybox and the reflections so they match.
  • A cheap Fresnel term that brightens reflections at glancing angles.

Prerequisites

Recipe 14 — Skybox & Cubemaps for the cubemap and the sky pipeline, and recipe 09 for per-fragment world-space normals. This recipe joins those two ideas.

Walkthrough

The reflection vector

For a perfect mirror, the colour at a surface point is whatever the environment shows along the reflected view ray. With I the incident direction (from the camera to the surface) and N the world-space normal, the GPU's built-in reflect gives the bounce direction. 

float3 I = normalize(in.world - u.cam_pos.xyz);   // camera -> surface
float3 N = normalize(in.normal);
float3 R = reflect(I, N);                          // mirror direction
float3 env = cube.sample(smp, R).rgb;             // look it up in the environment

Same cubemap, two uses

The trick that makes it convincing: the reflections sample the same cubemap drawn as the sky. Whatever surrounds the object is exactly what it mirrors, so the chrome and the backdrop are always consistent — no separate reflection probe needed for a static environment.

Fresnel: mirrors glint at the edge

Real reflective surfaces reflect more at grazing angles than head-on (Fresnel). A cheap Schlick-style approximation — one minus the view/normal alignment, raised to a power — lets you bias reflectivity and add an edge glint without a full BRDF. 

float fres = pow(1.0 - max(dot(N, -I), 0.0), 3.0);   // ~0 head-on, ~1 at the rim
float3 base = float3(0.05, 0.06, 0.08);              // dark tint where it faces you
float3 col  = mix(base, env, clamp(0.5 + fres, 0.0, 1.0)) + fres * 0.25;
Static vs dynamic. A cubemap captures a fixed environment, so the cube can't reflect itself or other moving objects. For those you'd render the scene into a cubemap each frame (a dynamic reflection probe) — the same sampling maths, just with a freshly rendered cubemap.

Complete listing

A spinning chrome cube sits inside a synthesised cubemap sky. The camera orbits; the cube mirrors the coloured faces around it, and the Fresnel rim keeps its silhouette reading against the background.

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_rotate_x(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[5]=c;r.m[6]=s;r.m[9]=-s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}
// Column-major look-at, matching the perspective convention (camera looks down -Z).
static Mat4 mat4_look_at(float ex,float ey,float ez, float cx,float cy,float cz, float ux,float uy,float uz){
    float fx=cx-ex, fy=cy-ey, fz=cz-ez; float fl=1.0f/SDL_sqrtf(fx*fx+fy*fy+fz*fz); fx*=fl; fy*=fl; fz*=fl;
    float sx=fy*uz-fz*uy, sy=fz*ux-fx*uz, sz=fx*uy-fy*ux; float sl=1.0f/SDL_sqrtf(sx*sx+sy*sy+sz*sz); sx*=sl; sy*=sl; sz*=sl;
    float vx=sy*fz-sz*fy, vy=sz*fx-sx*fz, vz=sx*fy-sy*fx;
    Mat4 r=mat4_identity();
    r.m[0]=sx; r.m[4]=sy; r.m[8]=sz;
    r.m[1]=vx; r.m[5]=vy; r.m[9]=vz;
    r.m[2]=-fx; r.m[6]=-fy; r.m[10]=-fz;
    r.m[12]=-(sx*ex+sy*ey+sz*ez); r.m[13]=-(vx*ex+vy*ey+vz*ez); r.m[14]=(fx*ex+fy*ey+fz*ez);
    return r;
}

static const char *kReflVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 world; float3 normal; };
struct VUBO  { float4x4 mvp; float4x4 model; };
vertex VSOut reflVS(VSIn in [[stage_in]], constant VUBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp   * float4(in.pos, 1.0);
    o.world    = (u.model * float4(in.pos, 1.0)).xyz;
    o.normal   = (u.model * float4(in.nrm, 0.0)).xyz;
    return o;
}
)msl";
static const char *kReflFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 world; float3 normal; };
struct FUBO  { float4 cam_pos; };
fragment float4 reflFS(VSOut in [[stage_in]],
                       texturecube<float> cube [[texture(0)]], sampler smp [[sampler(0)]],
                       constant FUBO &u [[buffer(0)]]) {
    float3 I = normalize(in.world - u.cam_pos.xyz);
    float3 N = normalize(in.normal);
    float3 R = reflect(I, N);
    float3 env = cube.sample(smp, R).rgb;
    float fres = pow(1.0 - max(dot(N, -I), 0.0), 3.0);
    float3 base = float3(0.05, 0.06, 0.08);
    return float4(mix(base, env, clamp(0.5 + fres, 0.0, 1.0)) + fres * 0.25, 1.0);
}
)msl";

static const char *kSkyVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; };

vertex VSOut skyVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o; o.dir = in.pos;

}
)msl";

                      texturecube<float> sky [[texture(0)]], sampler smp [[sampler(0)]]) {
    return sky.sample(smp, normalize(in.dir));
}
)msl";

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}

// Synthesise a distinctly-coloured environment so reflections are easy to read.
static void fill_face(int face, Uint8 *px)
{
    // A base hue per face (+X -X +Y -Y +Z -Z), brightened toward the centre.
    const Uint8 hue[6][3] = {
        {220,80,80}, {80,220,120}, {120,160,235}, {60,50,45}, {235,200,90}, {180,90,210}
    };
    for (int y = 0; y < FACE; ++y)
    for (int x = 0; x < FACE; ++x) {
        float cx = (x + 0.5f) / FACE - 0.5f, cy = (y + 0.5f) / FACE - 0.5f;
        float glow = 1.0f - SDL_min(1.0f, (SDL_fabsf(cx) + SDL_fabsf(cy)) * 1.6f);
        Uint8 *p = px + (y*FACE + x)*4;
        p[0] = (Uint8)SDL_min(255.0f, hue[face][0] * (0.5f + 0.7f*glow));
        p[1] = (Uint8)SDL_min(255.0f, hue[face][1] * (0.5f + 0.7f*glow));
        p[2] = (Uint8)SDL_min(255.0f, hue[face][2] * (0.5f + 0.7f*glow));
        p[3] = 255;
    }
}
static void upload_cubemap(AppState *st)
{
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=FACE*FACE*4*6;
    SDL_GPUTransferBuffer *tb = SDL_CreateGPUTransferBuffer(st->device, &tbi);
    Uint8 *m = (Uint8*)SDL_MapGPUTransferBuffer(st->device, tb, false);
    for (int f = 0; f < 6; ++f) fill_face(f, m + f*FACE*FACE*4);
    SDL_UnmapGPUTransferBuffer(st->device, tb);
    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUCopyPass *cp = SDL_BeginGPUCopyPass(cmd);
    for (int f = 0; f < 6; ++f) {
        SDL_GPUTextureTransferInfo src{}; src.transfer_buffer=tb; src.offset=f*FACE*FACE*4;
        SDL_GPUTextureRegion r{}; r.texture=st->cubemap; r.layer=(Uint32)f; r.w=FACE; r.h=FACE; r.d=1;
        SDL_UploadToGPUTexture(cp, &src, &r, false);
    }
    SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(st->device, tb);
}
static void make_depth(AppState *st, Uint32 w, Uint32 h){
    if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->depth=SDL_CreateGPUTexture(st->device,&ti); st->fb_w=w; st->fb_h=h;
}

// Build a graphics pipeline; pos+normal layout if `with_normal`, else pos only.
static SDL_GPUGraphicsPipeline *make_pipeline(AppState *st, SDL_GPUShader *vs, SDL_GPUShader *fs,
                                              bool with_normal, SDL_GPUTextureFormat fmt,
                                              SDL_GPUCompareOp depth_op, bool depth_write){
    SDL_GPUVertexBufferDescription vb{}; vb.slot=0;
    vb.pitch = with_normal ? sizeof(Vertex) : (Uint32)(3*sizeof(float));
    vb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute a[2]{};
    a[0].location=0; a[0].buffer_slot=0; a[0].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; a[0].offset=0;
    a[1].location=1; a[1].buffer_slot=0; a[1].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; a[1].offset=(Uint32)(3*sizeof(float));
    SDL_GPUVertexInputState vis{}; vis.vertex_buffer_descriptions=&vb; vis.num_vertex_buffers=1;
    vis.vertex_attributes=a; vis.num_vertex_attributes = with_normal ? 2 : 1;
    SDL_GPUColorTargetDescription ct{}; ct.format=fmt;
    SDL_GPUGraphicsPipelineCreateInfo p{};
    p.vertex_shader=vs; p.fragment_shader=fs; p.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; p.vertex_input_state=vis;
    p.target_info.color_target_descriptions=&ct; p.target_info.num_color_targets=1;
    p.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; p.target_info.has_depth_stencil_target=true;
    p.depth_stencil_state.enable_depth_test=true; p.depth_stencil_state.enable_depth_write=depth_write; p.depth_stencil_state.compare_op=depth_op;
    p.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    return SDL_CreateGPUGraphicsPipeline(st->device, &p);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Environment Reflections", "1.0", "com.example.sdl3.reflect");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Environment Reflections", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    SDL_GPUBufferCreateInfo vbi{}; vbi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; vbi.size=sizeof(kCube); st->vbuf=SDL_CreateGPUBuffer(st->device,&vbi);
    SDL_GPUBufferCreateInfo ibi{}; ibi.usage=SDL_GPU_BUFFERUSAGE_INDEX;  ibi.size=sizeof(kIdx);  st->ibuf=SDL_CreateGPUBuffer(st->device,&ibi);
    SDL_GPUBufferCreateInfo bvi{}; bvi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; bvi.size=sizeof(kBox);  st->boxv=SDL_CreateGPUBuffer(st->device,&bvi);
    SDL_GPUBufferCreateInfo bii{}; bii.usage=SDL_GPU_BUFFERUSAGE_INDEX;  bii.size=sizeof(kBoxIdx); st->boxi=SDL_CreateGPUBuffer(st->device,&bii);
    upload_buffer(st->device, st->vbuf, kCube, sizeof(kCube));
    upload_buffer(st->device, st->ibuf, kIdx,  sizeof(kIdx));
    upload_buffer(st->device, st->boxv, kBox,  sizeof(kBox));
    upload_buffer(st->device, st->boxi, kBoxIdx, sizeof(kBoxIdx));

    SDL_GPUTextureCreateInfo ci{}; ci.type=SDL_GPU_TEXTURETYPE_CUBE; ci.format=SDL_GPU_TEXTUREFORMAT_R8G8B8A8_UNORM;
    ci.usage=SDL_GPU_TEXTUREUSAGE_SAMPLER; ci.width=FACE; ci.height=FACE; ci.layer_count_or_depth=6; ci.num_levels=1; ci.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->cubemap = SDL_CreateGPUTexture(st->device, &ci);
    upload_cubemap(st);

    SDL_GPUSamplerCreateInfo si{}; si.min_filter=SDL_GPU_FILTER_LINEAR; si.mag_filter=SDL_GPU_FILTER_LINEAR;
    si.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_w=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->sampler = SDL_CreateGPUSampler(st->device, &si);

    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    SDL_GPUShader *rvs=make_shader(st->device,kReflVS,SDL_GPU_SHADERSTAGE_VERTEX,  "reflVS",1,0);
    SDL_GPUShader *rfs=make_shader(st->device,kReflFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"reflFS",1,1);
    st->refl = make_pipeline(st, rvs, rfs, true, swap, SDL_GPU_COMPAREOP_LESS, true);
    SDL_ReleaseGPUShader(st->device,rvs); SDL_ReleaseGPUShader(st->device,rfs);

    SDL_GPUShader *svs=make_shader(st->device,kSkyVS,SDL_GPU_SHADERSTAGE_VERTEX,  "skyVS",1,0);
    SDL_GPUShader *sfs=make_shader(st->device,kSkyFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"skyFS",0,1);
    st->sky = make_pipeline(st, svs, sfs, false, swap, SDL_GPU_COMPAREOP_LESS_OR_EQUAL, false);
    SDL_ReleaseGPUShader(st->device,svs); SDL_ReleaseGPUShader(st->device,sfs);
    if (!st->refl || !st->sky) return fail("pipelines");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) make_depth(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;

        float ang = t * 0.3f, dist = 4.0f;
        float ex = SDL_sinf(ang)*dist, ey = 1.2f, ez = SDL_cosf(ang)*dist;   // orbiting camera
        Mat4 proj = mat4_perspective(60.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);
        Mat4 view = mat4_look_at(ex,ey,ez, 0,0,0, 0,1,0);
        Mat4 model = mat4_mul(mat4_rotate_y(t*0.8f), mat4_rotate_x(t*0.5f));

        VUBO vub{}; vub.mvp = mat4_mul(proj, mat4_mul(view, model)); vub.model = model;
        FUBO fub{}; fub.cam_pos[0]=ex; fub.cam_pos[1]=ey; fub.cam_pos[2]=ez; fub.cam_pos[3]=1;
        // Sky view = orientation only (camera at origin), so it stays put around the camera.
        Mat4 sky_view = mat4_look_at(0,0,0, -ex,-ey,-ez, 0,1,0);
        SkyUBO sub{}; sub.vp = mat4_mul(proj, sky_view);

        SDL_GPUColorTargetInfo cti{}; cti.texture=swap; cti.clear_color=SDL_FColor{0.02f,0.02f,0.03f,1};
        cti.load_op=SDL_GPU_LOADOP_CLEAR; cti.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo dti{}; dti.texture=st->depth; dti.clear_depth=1.0f;
        dti.load_op=SDL_GPU_LOADOP_CLEAR; dti.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *pass = SDL_BeginGPURenderPass(cmd, &cti, 1, &dti);

        // --- reflective cube first (writes depth) ---
        SDL_BindGPUGraphicsPipeline(pass, st->refl);
        SDL_GPUBufferBinding vb{}; vb.buffer=st->vbuf; SDL_BindGPUVertexBuffers(pass,0,&vb,1);
        SDL_GPUBufferBinding ib{}; ib.buffer=st->ibuf; SDL_BindGPUIndexBuffer(pass,&ib, SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_GPUTextureSamplerBinding cb{}; cb.texture=st->cubemap; cb.sampler=st->sampler;
        SDL_BindGPUFragmentSamplers(pass, 0, &cb, 1);
        SDL_PushGPUVertexUniformData(cmd, 0, &vub, sizeof(vub));
        SDL_PushGPUFragmentUniformData(cmd, 0, &fub, sizeof(fub));
        SDL_DrawGPUIndexedPrimitives(pass, 36, 1, 0, 0, 0);

        // --- sky last (LEQUAL, fills only untouched pixels) ---
        SDL_BindGPUGraphicsPipeline(pass, st->sky);
        SDL_GPUBufferBinding bvb{}; bvb.buffer=st->boxv; SDL_BindGPUVertexBuffers(pass,0,&bvb,1);
        SDL_GPUBufferBinding bib{}; bib.buffer=st->boxi; SDL_BindGPUIndexBuffer(pass,&bib, SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_BindGPUFragmentSamplers(pass, 0, &cb, 1);
        SDL_PushGPUVertexUniformData(cmd, 0, &sub, sizeof(sub));
        SDL_DrawGPUIndexedPrimitives(pass, 36, 1, 0, 0, 0);

        SDL_EndGPURenderPass(pass);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->sampler) SDL_ReleaseGPUSampler(st->device, st->sampler);
        if (st->cubemap) SDL_ReleaseGPUTexture(st->device, st->cubemap);
        if (st->depth)   SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->boxi) SDL_ReleaseGPUBuffer(st->device, st->boxi);
        if (st->boxv) SDL_ReleaseGPUBuffer(st->device, st->boxv);
        if (st->ibuf) SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf) SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        if (st->sky)  SDL_ReleaseGPUGraphicsPipeline(st->device, st->sky);
        if (st->refl) SDL_ReleaseGPUGraphicsPipeline(st->device, st->refl);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Add roughness: sample a blurred (lower-resolution) version of the cubemap and mix toward it for a brushed-metal look.
  2. Refraction instead of reflection: swap reflect for refract(I, N, 0.66) and the cube turns to glass.
  3. Blend reflection with a Phong base colour by a reflectivity uniform so the surface reads as painted metal rather than pure chrome.
  4. Replace the cube with a sphere (subdivide a cube or generate a UV-sphere) — curved surfaces show off environment reflections far better than flat faces.
