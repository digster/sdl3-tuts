Recipe 22 — Point & Spot Lights

Light a scene with many dynamic lights in a single forward pass. Pack a uniform array of point and spot lights, give each one distance attenuation and a cone, and loop over them in the fragment shader.

What you'll learn

  • Passing an array of lights as a uniform block, float4-packed to avoid alignment traps.
  • Distance attenuation so a point light fades with range.
  • Turning a point light into a spotlight with a direction and a cosine cutoff.
  • Accumulating every light's diffuse + specular contribution in one loop.

Prerequisites

Recipe 09 — Phong Lighting (the single light) and recipe 10 for drawing several transforms in a pass. This recipe generalises the one light into a loop over many.

Walkthrough

One light, packed for the GPU

A point light needs a position and a colour; a spotlight adds a direction and a cone angle; range controls attenuation. Pack it all into three float4s — never a bare float3 in a uniform array, because MSL pads float3 to 16 bytes and the C++ side rarely agrees. With float4 the two layouts are byte-identical. 

struct Light {
    float pos_range[4];    // xyz = world position, w = range
    float color_int[4];    // rgb = colour,        w = intensity
    float dir_cone[4];     // xyz = spot direction, w = cos(cutoff); w < -0.5 ⇒ point light
};
struct FUBO { float view_pos[4]; float count[4]; Light lights[MAX_LIGHTS]; };

Attenuation and the cone

Each light's reach falls off with distance. A simple, clamp-friendly model is (1 - dist/range)² — zero past the range, smooth within it. A spotlight multiplies that by a cone factor: how aligned the surface-to- light direction is with the spotlight's axis, softened with smoothstep so the edge isn't hard. 

float att  = clamp(1.0 - dist / range, 0.0, 1.0);
att *= att;                                   // quadratic-ish falloff
float spot = 1.0;
if (cone > -0.5) {                            // this light is a spotlight
    float aligned = dot(-L, normalize(spot_dir));
    spot = smoothstep(cone, cone + 0.05, aligned);
}

Loop over every light

The fragment shader starts from a small ambient term, then adds each light's Lambertian diffuse and Blinn specular, weighted by its colour, intensity, attenuation and cone. Dozens of lights cost only dozens of loop iterations per pixel — the essence of forward shading. 

float3 result = base * 0.08;                  // ambient
for (int i = 0; i < int(u.count.x); ++i) {
    /* ...compute L, dist, att, spot... */
    float  diff = max(dot(N, L), 0.0);
    float  spec = pow(max(dot(N, normalize(L + V)), 0.0), 48.0);
    result += (base * diff + spec) * lightColor * intensity * att * spot;
}
Forward shading's ceiling. Every light touches every lit fragment, so cost is lights × fragments. That's fine for a handful of lights; when it isn't, the answer is deferred shading (recipe 25), which decouples the two.

Complete listing

A ground plane and two cubes lit by three coloured point lights orbiting at different radii plus one white spotlight aimed down from above. The light pools slide across the floor as they move. 

/*
 * 22-point-and-spot-lights.cpp — a forward pass with an array of point/spot lights.
 * Builds on the Phong cube of recipe 09 and the multi-object draw of recipe 10.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <cstring>
#include <cstddef>

#define MAX_LIGHTS 8

struct Vertex { float x, y, z, nx, ny, nz; };
static const Vertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const Uint16 kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_scale(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[0]=x;r.m[5]=y;r.m[10]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}

static const char *kVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 world; float3 normal; float4 base; };
struct VUBO  { float4x4 mvp; float4x4 model; float4 base; };
vertex VSOut vertexMain(VSIn in [[stage_in]], constant VUBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp   * float4(in.pos, 1.0);
    o.world    = (u.model * float4(in.pos, 1.0)).xyz;
    o.normal   = (u.model * float4(in.nrm, 0.0)).xyz;
    o.base     = u.base;
    return o;
}
)msl";

static const char *kFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 world; float3 normal; float4 base; };
struct Light { float4 pos_range; float4 color_int; float4 dir_cone; };
struct FUBO  { float4 view_pos; float4 count; Light lights[8]; };
fragment float4 fragmentMain(VSOut in [[stage_in]], constant FUBO &u [[buffer(0)]]) {
    float3 base = in.base.rgb;
    float3 N = normalize(in.normal);
    float3 V = normalize(u.view_pos.xyz - in.world);
    float3 result = base * 0.08;                         // ambient
    int n = int(u.count.x);
    for (int i = 0; i < n; ++i) {
        float3 toL  = u.lights[i].pos_range.xyz - in.world;
        float  dist = length(toL);
        float3 L    = toL / max(dist, 1e-4);
        float  att  = clamp(1.0 - dist / u.lights[i].pos_range.w, 0.0, 1.0);
        att *= att;
        float  spot = 1.0;
        float  cone = u.lights[i].dir_cone.w;
        if (cone > -0.5) {
            float aligned = dot(-L, normalize(u.lights[i].dir_cone.xyz));
            spot = smoothstep(cone, cone + 0.05, aligned);
        }
        float  diff = max(dot(N, L), 0.0);
        float  spec = pow(max(dot(N, normalize(L + V)), 0.0), 48.0);
        result += (base * diff + spec) * u.lights[i].color_int.rgb
                  * u.lights[i].color_int.w * att * spot;
    }
    return float4(result, 1.0);
}
)msl";

struct VUBO  { Mat4 mvp; Mat4 model; float base[4]; };
struct Light { float pos_range[4]; float color_int[4]; float dir_cone[4]; };
struct FUBO  { float view_pos[4]; float count[4]; Light lights[MAX_LIGHTS]; };

struct AppState {
    SDL_Window *window=nullptr; SDL_GPUDevice *device=nullptr;
    SDL_GPUGraphicsPipeline *pipeline=nullptr;
    SDL_GPUBuffer *vbuf=nullptr, *ibuf=nullptr;
    SDL_GPUTexture *depth=nullptr; Uint32 fb_w=0, fb_h=0;
    Uint64 start_ns=0;
};

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo;
    return SDL_CreateGPUShader(d,&i);
}
static void upload(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static void make_depth(AppState *st, Uint32 w, Uint32 h){
    if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->depth=SDL_CreateGPUTexture(st->device,&ti); st->fb_w=w; st->fb_h=h;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Point and Spot Lights", "1.0", "com.example.sdl3.lights");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Point and Spot Lights", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    SDL_GPUBufferCreateInfo vbi{}; vbi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; vbi.size=sizeof(kCube); st->vbuf=SDL_CreateGPUBuffer(st->device,&vbi);
    SDL_GPUBufferCreateInfo ibi{}; ibi.usage=SDL_GPU_BUFFERUSAGE_INDEX;  ibi.size=sizeof(kIdx);  st->ibuf=SDL_CreateGPUBuffer(st->device,&ibi);
    upload(st->device, st->vbuf, kCube, sizeof(kCube));
    upload(st->device, st->ibuf, kIdx,  sizeof(kIdx));

    SDL_GPUShader *vs = make_shader(st->device, kVS, SDL_GPU_SHADERSTAGE_VERTEX,   "vertexMain",   1);
    SDL_GPUShader *fs = make_shader(st->device, kFS, SDL_GPU_SHADERSTAGE_FRAGMENT, "fragmentMain", 1);

    SDL_GPUVertexBufferDescription vb{}; vb.slot=0; vb.pitch=sizeof(Vertex); vb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute attrs[2]{};
    attrs[0].location=0; attrs[0].buffer_slot=0; attrs[0].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; attrs[0].offset=offsetof(Vertex,x);
    attrs[1].location=1; attrs[1].buffer_slot=0; attrs[1].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; attrs[1].offset=offsetof(Vertex,nx);
    SDL_GPUVertexInputState vis{}; vis.vertex_buffer_descriptions=&vb; vis.num_vertex_buffers=1; vis.vertex_attributes=attrs; vis.num_vertex_attributes=2;

    SDL_GPUColorTargetDescription ct{}; ct.format=SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    SDL_GPUGraphicsPipelineCreateInfo pci{};
    pci.vertex_shader=vs; pci.fragment_shader=fs; pci.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; pci.vertex_input_state=vis;
    pci.target_info.color_target_descriptions=&ct; pci.target_info.num_color_targets=1;
    pci.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; pci.target_info.has_depth_stencil_target=true;
    pci.depth_stencil_state.enable_depth_test=true; pci.depth_stencil_state.enable_depth_write=true; pci.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    pci.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->pipeline = SDL_CreateGPUGraphicsPipeline(st->device, &pci);
    SDL_ReleaseGPUShader(st->device, vs); SDL_ReleaseGPUShader(st->device, fs);
    if (!st->pipeline) return fail("pipeline");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

// One draw: push the per-object vertex UBO, then draw the 36-index cube.
static void draw_cube(SDL_GPUCommandBuffer *cmd, SDL_GPURenderPass *pass,
                      const Mat4 &vp, const Mat4 &model, float r, float g, float b)
{
    VUBO v{}; v.mvp = mat4_mul(vp, model); v.model = model;
    v.base[0]=r; v.base[1]=g; v.base[2]=b; v.base[3]=1;
    SDL_PushGPUVertexUniformData(cmd, 0, &v, sizeof(v));
    SDL_DrawGPUIndexedPrimitives(pass, 36, 1, 0, 0, 0);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) make_depth(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;
        Mat4 view = mat4_mul(mat4_translate(0,-1.5f,-8.0f), mat4_rotate_y(t*0.15f));
        Mat4 proj = mat4_perspective(55.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);
        Mat4 vp   = mat4_mul(proj, view);

        // ---- assemble the light array ----
        FUBO fub{};
        fub.view_pos[0]=0; fub.view_pos[1]=1.5f; fub.view_pos[2]=8.0f;
        fub.count[0]=4;
        // three orbiting point lights (cone = -1 ⇒ point)
        const float pc[3][3] = { {1.0f,0.3f,0.2f}, {0.2f,0.5f,1.0f}, {0.3f,1.0f,0.4f} };
        for (int i = 0; i < 3; ++i) {
            float a = t * (0.7f + 0.2f*i) + i * 2.094f;
            fub.lights[i].pos_range[0] = SDL_cosf(a) * (2.0f + i);
            fub.lights[i].pos_range[1] = 0.6f;
            fub.lights[i].pos_range[2] = SDL_sinf(a) * (2.0f + i);
            fub.lights[i].pos_range[3] = 7.0f;                       // range
            fub.lights[i].color_int[0]=pc[i][0]; fub.lights[i].color_int[1]=pc[i][1]; fub.lights[i].color_int[2]=pc[i][2];
            fub.lights[i].color_int[3]=1.4f;                         // intensity
            fub.lights[i].dir_cone[3] = -1.0f;                       // point light
        }
        // one spotlight aimed straight down from above
        fub.lights[3].pos_range[0]=0; fub.lights[3].pos_range[1]=4.0f; fub.lights[3].pos_range[2]=0; fub.lights[3].pos_range[3]=10.0f;
        fub.lights[3].color_int[0]=1.0f; fub.lights[3].color_int[1]=1.0f; fub.lights[3].color_int[2]=0.95f; fub.lights[3].color_int[3]=2.0f;
        fub.lights[3].dir_cone[0]=0; fub.lights[3].dir_cone[1]=-1.0f; fub.lights[3].dir_cone[2]=0;
        fub.lights[3].dir_cone[3]=SDL_cosf(22.0f*SDL_PI_F/180.0f);   // cone cutoff

        SDL_GPUColorTargetInfo cti{}; cti.texture=swap; cti.clear_color=SDL_FColor{0.02f,0.02f,0.03f,1};
        cti.load_op=SDL_GPU_LOADOP_CLEAR; cti.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo dti{}; dti.texture=st->depth; dti.clear_depth=1.0f;
        dti.load_op=SDL_GPU_LOADOP_CLEAR; dti.store_op=SDL_GPU_STOREOP_DONT_CARE;

        SDL_GPURenderPass *pass = SDL_BeginGPURenderPass(cmd, &cti, 1, &dti);
        SDL_BindGPUGraphicsPipeline(pass, st->pipeline);
        SDL_GPUBufferBinding vbind{}; vbind.buffer=st->vbuf; SDL_BindGPUVertexBuffers(pass,0,&vbind,1);
        SDL_GPUBufferBinding ibind{}; ibind.buffer=st->ibuf; SDL_BindGPUIndexBuffer(pass,&ibind, SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUFragmentUniformData(cmd, 0, &fub, sizeof(fub));   // same lights for every object

        draw_cube(cmd, pass, vp, mat4_mul(mat4_translate(0,-1.1f,0), mat4_scale(10,0.2f,10)), 0.8f,0.8f,0.85f); // floor
        draw_cube(cmd, pass, vp, mat4_mul(mat4_translate(-1.4f,-0.5f,0), mat4_rotate_y(t*0.6f)), 0.9f,0.6f,0.4f);
        draw_cube(cmd, pass, vp, mat4_mul(mat4_translate( 1.4f,-0.5f,0), mat4_rotate_y(-t*0.5f)), 0.5f,0.7f,0.9f);

        SDL_EndGPURenderPass(pass);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->ibuf)  SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf)  SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        if (st->pipeline) SDL_ReleaseGPUGraphicsPipeline(st->device, st->pipeline);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Raise count to 8 and scatter more coloured point lights — watch the per-pixel cost grow with light count, the defining trait of forward shading.
  2. Animate the spotlight: sweep its dir_cone in a circle so the cone sweeps the floor like a searchlight.
  3. Add an inner/outer cone: pass two cosines and smoothstep(outer, inner, aligned) for a softer penumbra.
  4. Visualise each light: draw a tiny emissive cube at every light position so you can see where the pools come from.
