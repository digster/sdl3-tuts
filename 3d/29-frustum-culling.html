Recipe 29 — Frustum Culling

Don't draw what the camera can't see. Extract the six planes of the view frustum straight from the view-projection matrix, test each object's bounding box against them, and re-upload only the survivors to the instance buffer — the CPU side of scaling to big scenes.

What you'll learn

  • Extracting the six frustum planes from a view-projection matrix (Gribb–Hartmann).
  • The fast AABB-vs-plane "positive vertex" visibility test.
  • Rebuilding a visible-instance list each frame and re-uploading it.
  • Drawing the survivors with one instanced call, and reading the cull rate live.

Prerequisites

Recipe 18 — Instanced Rendering for the per-instance vertex buffer and single instanced draw, and recipe 05 for the matrix layout the plane extraction relies on.

Walkthrough

Six planes from one matrix

A point is inside clip space when −w ≤ x ≤ w (and the same for y, z). Since the clip coordinate is a row of the view-projection matrix dotted with the world point, rearranging each inequality gives a world-space plane as a sum or difference of matrix rows — the Gribb–Hartmann method. No inverse, no per-object work. 

struct Plane { float a, b, c, d; };   // inside if a*x + b*y + c*z + d >= 0

// rows of the column-major view-proj: row_i = (m[i], m[4+i], m[8+i], m[12+i])
static void extract_planes(const float *m, Plane p[6]) {
    auto row = [&](int i){ return Plane{ m[i], m[4+i], m[8+i], m[12+i] }; };
    Plane r0 = row(0), r1 = row(1), r2 = row(2), r3 = row(3);
    auto add = [](Plane a, Plane b, float s){ return Plane{ a.a+s*b.a, a.b+s*b.b, a.c+s*b.c, a.d+s*b.d }; };
    p[0] = add(r3, r0, +1);   // left    : w + x >= 0
    p[1] = add(r3, r0, -1);   // right   : w - x >= 0
    p[2] = add(r3, r1, +1);   // bottom  : w + y >= 0
    p[3] = add(r3, r1, -1);   // top     : w - y >= 0
    p[4] = r2;                // near    : z >= 0   (zero-to-one depth)
    p[5] = add(r3, r2, -1);   // far     : w - z >= 0
    for (int i = 0; i < 6; ++i) {                         // normalize for correct distances
        float l = SDL_sqrtf(p[i].a*p[i].a + p[i].b*p[i].b + p[i].c*p[i].c);
        p[i].a/=l; p[i].b/=l; p[i].c/=l; p[i].d/=l;
    }
}

The positive-vertex test

A box is outside a plane only if even its most-positive corner (the AABB vertex furthest along the plane normal) is behind the plane. Test that one corner per plane; if any plane rejects it, the box is invisible and we skip it. This over-accepts boxes straddling a corner, which is fine — the goal is a cheap conservative reject, not an exact test. 

static bool aabb_visible(const Plane p[6], float cx, float cy, float cz, float h) {
    for (int i = 0; i < 6; ++i) {
        // pick the corner furthest along this plane's normal
        float px = cx + (p[i].a >= 0 ? h : -h);
        float py = cy + (p[i].b >= 0 ? h : -h);
        float pz = cz + (p[i].c >= 0 ? h : -h);
        if (p[i].a*px + p[i].b*py + p[i].c*pz + p[i].d < 0) return false;   // fully outside
    }
    return true;
}

Rebuild the instance list

Each frame, walk every object, keep the visible ones, and copy their per-instance records into a dynamic vertex buffer. The draw then issues a single instanced call whose instance count is the number that survived — the GPU never even sees the culled objects. 

visible.clear();
for (const Object &o : world)
    if (aabb_visible(planes, o.x, o.y, o.z, 0.5f)) visible.push_back(o.inst);
upload_instances(dev, st->inst_buf, visible.data(), visible.size());   // cycle = true
SDL_DrawGPUIndexedPrimitives(pass, 36, (Uint32)visible.size(), 0, 0, 0);
Cull from a fixed camera, view from another. The demo extracts planes from a narrow culling frustum but renders from a pulled-back camera, so you can watch cubes wink out exactly at the culling frustum's edges. In a real engine both matrices are the same; here they're split only to make the culling visible.

Complete listing

A large grid of cubes. The scene is viewed from a high, pulled-back camera, but culled against a narrower frustum that slowly rotates — so you can see the visible wedge sweep through the field. The window title reports how many of the total cubes survived each frame. 

/*
 * 29-frustum-culling.cpp — Gribb-Hartmann plane extraction + AABB tests build a
 * per-frame visible-instance list, drawn with one instanced call (recipe 18).
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <cstring>
#include <cstddef>
#include <vector>

static const int GRID = 24;                          // GRID*GRID cubes
static const int MAX_INST = GRID * GRID;

struct Vertex   { float x, y, z, nx, ny, nz; };
struct Instance { float ox, oy, oz; float r, g, b, a; };
struct Object   { float x, y, z; Instance inst; };

static const Vertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const Uint16 kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}
static Mat4 mat4_look_at(float ex,float ey,float ez, float cx,float cy,float cz, float ux,float uy,float uz){
    float fx=cx-ex, fy=cy-ey, fz=cz-ez; float fl=SDL_sqrtf(fx*fx+fy*fy+fz*fz); fx/=fl; fy/=fl; fz/=fl;
    float sx=fy*uz-fz*uy, sy=fz*ux-fx*uz, sz=fx*uy-fy*ux; float sl=SDL_sqrtf(sx*sx+sy*sy+sz*sz); sx/=sl; sy/=sl; sz/=sl;
    float ux2=sy*fz-sz*fy, uy2=sz*fx-sx*fz, uz2=sx*fy-sy*fx;
    Mat4 r=mat4_identity();
    r.m[0]=sx; r.m[4]=sy; r.m[8]=sz;  r.m[1]=ux2; r.m[5]=uy2; r.m[9]=uz2;
    r.m[2]=-fx; r.m[6]=-fy; r.m[10]=-fz;
    r.m[12]=-(sx*ex+sy*ey+sz*ez); r.m[13]=-(ux2*ex+uy2*ey+uz2*ez); r.m[14]=(fx*ex+fy*ey+fz*ez);
    return r;
}

struct Plane { float a, b, c, d; };
static void extract_planes(const float *m, Plane p[6]) {
    auto row = [&](int i){ return Plane{ m[i], m[4+i], m[8+i], m[12+i] }; };
    Plane r0 = row(0), r1 = row(1), r2 = row(2), r3 = row(3);
    auto add = [](Plane a, Plane b, float s){ return Plane{ a.a+s*b.a, a.b+s*b.b, a.c+s*b.c, a.d+s*b.d }; };
    p[0] = add(r3, r0, +1); p[1] = add(r3, r0, -1);
    p[2] = add(r3, r1, +1); p[3] = add(r3, r1, -1);
    p[4] = r2;              p[5] = add(r3, r2, -1);
    for (int i = 0; i < 6; ++i) {
        float l = SDL_sqrtf(p[i].a*p[i].a + p[i].b*p[i].b + p[i].c*p[i].c);
        if (l > 1e-6f) { p[i].a/=l; p[i].b/=l; p[i].c/=l; p[i].d/=l; }
    }
}
static bool aabb_visible(const Plane p[6], float cx, float cy, float cz, float h) {
    for (int i = 0; i < 6; ++i) {
        float px = cx + (p[i].a >= 0 ? h : -h);
        float py = cy + (p[i].b >= 0 ? h : -h);
        float pz = cz + (p[i].c >= 0 ? h : -h);
        if (p[i].a*px + p[i].b*py + p[i].c*pz + p[i].d < 0) return false;
    }
    return true;
}

static const char *kVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]];
               float3 ioffset [[attribute(2)]]; float4 icolor [[attribute(3)]]; };
struct VSOut { float4 position [[position]]; float3 wn; float4 color; };
struct UBO   { float4x4 viewProj; };
vertex VSOut vs(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.viewProj * float4(in.pos + in.ioffset, 1.0);
    o.wn = in.nrm; o.color = in.icolor;
    return o;
}
)msl";
static const char *kFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 wn; float4 color; };
fragment float4 fs(VSOut in [[stage_in]]) {
    float diff = max(dot(normalize(in.wn), normalize(float3(0.4, 0.8, 0.3))), 0.0);
    return float4(in.color.rgb * (0.3 + 0.75 * diff), 1.0);
}
)msl";

struct AppState {
    SDL_Window *window=nullptr; SDL_GPUDevice *device=nullptr;
    SDL_GPUGraphicsPipeline *pipeline=nullptr;
    SDL_GPUBuffer *vbuf=nullptr, *ibuf=nullptr, *inst=nullptr;
    SDL_GPUTexture *depth=nullptr;
    std::vector<Object> world;
    std::vector<Instance> visible;
    Uint32 fb_w=0, fb_h=0; Uint64 start_ns=0; Uint64 title_ns=0;
};

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size, bool cycle){
    if (size == 0) return;
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,cycle); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,cycle); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static SDL_GPUBuffer *make_buf(SDL_GPUDevice *d, SDL_GPUBufferUsageFlags u, const void *src, Uint32 size){
    SDL_GPUBufferCreateInfo bi{}; bi.usage=u; bi.size=size; SDL_GPUBuffer *b=SDL_CreateGPUBuffer(d,&bi);
    upload_buffer(d,b,src,size,false); return b;
}
static SDL_GPUTexture *create_depth(SDL_GPUDevice *dev, Uint32 w, Uint32 h){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Frustum Culling", "1.0", "com.example.sdl3.frustum");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Frustum Culling", 900, 640, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    // Build the static world; each cube keeps its centre for the AABB test.
    st->world.reserve(MAX_INST);
    for (int z = 0; z < GRID; ++z)
        for (int x = 0; x < GRID; ++x) {
            float ox = (x - GRID*0.5f) * 2.0f, oz = (z - GRID*0.5f) * 2.0f;
            float h = (float)(x + z) / (2 * GRID);
            Instance in{ ox, 0.0f, oz,
                         0.5f + 0.5f*SDL_sinf(h*6.283f),
                         0.5f + 0.5f*SDL_sinf(h*6.283f + 2.094f),
                         0.5f + 0.5f*SDL_sinf(h*6.283f + 4.188f), 1.0f };
            st->world.push_back({ ox, 0.0f, oz, in });
        }

    st->vbuf = make_buf(st->device, SDL_GPU_BUFFERUSAGE_VERTEX, kCube, sizeof(kCube));
    st->ibuf = make_buf(st->device, SDL_GPU_BUFFERUSAGE_INDEX,  kIdx,  sizeof(kIdx));
    SDL_GPUBufferCreateInfo ii{}; ii.usage=SDL_GPU_BUFFERUSAGE_VERTEX; ii.size=sizeof(Instance)*MAX_INST;
    st->inst = SDL_CreateGPUBuffer(st->device, &ii);   // dynamic: rewritten each frame

    SDL_GPUShader *vs=make_shader(st->device,kVS,SDL_GPU_SHADERSTAGE_VERTEX,  "vs",1,0);
    SDL_GPUShader *fs=make_shader(st->device,kFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"fs",0,0);
    SDL_GPUVertexBufferDescription vb[2]{};
    vb[0].slot=0; vb[0].pitch=sizeof(Vertex);   vb[0].input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    vb[1].slot=1; vb[1].pitch=sizeof(Instance); vb[1].input_rate=SDL_GPU_VERTEXINPUTRATE_INSTANCE;
    SDL_GPUVertexAttribute a[4]{};
    a[0]={0,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,x)};
    a[1]={1,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,nx)};
    a[2]={2,1,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Instance,ox)};
    a[3]={3,1,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT4, offsetof(Instance,r)};
    SDL_GPUVertexInputState vi{}; vi.vertex_buffer_descriptions=vb; vi.num_vertex_buffers=2; vi.vertex_attributes=a; vi.num_vertex_attributes=4;
    SDL_GPUColorTargetDescription ct{}; ct.format=SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    SDL_GPUGraphicsPipelineCreateInfo pc{};
    pc.vertex_shader=vs; pc.fragment_shader=fs; pc.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; pc.vertex_input_state=vi;
    pc.target_info.color_target_descriptions=&ct; pc.target_info.num_color_targets=1;
    pc.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; pc.target_info.has_depth_stencil_target=true;
    pc.depth_stencil_state.enable_depth_test=true; pc.depth_stencil_state.enable_depth_write=true; pc.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    pc.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->pipeline=SDL_CreateGPUGraphicsPipeline(st->device,&pc);
    SDL_ReleaseGPUShader(st->device,vs); SDL_ReleaseGPUShader(st->device,fs);
    if(!st->pipeline) return fail("pipeline");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) {
            if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
            st->depth = create_depth(st->device, sw, sh); st->fb_w=sw; st->fb_h=sh;
        }
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;

        // The CULLING frustum: a narrow rotating camera near the field.
        float cull_x = SDL_cosf(t*0.4f) * 4.0f, cull_z = SDL_sinf(t*0.4f) * 4.0f;
        Mat4 cullView = mat4_look_at(cull_x, 6.0f, cull_z,
                                     SDL_cosf(t*0.4f)*20.0f, 0.0f, SDL_sinf(t*0.4f)*20.0f, 0,1,0);
        Mat4 cullProj = mat4_perspective(45.0f*SDL_PI_F/180.0f, aspect, 1.0f, 40.0f);
        Mat4 cullVP   = mat4_mul(cullProj, cullView);
        Plane planes[6]; extract_planes(cullVP.m, planes);

        // The VIEWING camera: high and pulled back so we watch the cull happen.
        Mat4 view = mat4_look_at(0, 46.0f, 0.1f, 0, 0, 0, 0, 0, -1);
        Mat4 proj = mat4_perspective(60.0f*SDL_PI_F/180.0f, aspect, 0.1f, 200.0f);
        Mat4 viewProj = mat4_mul(proj, view);

        // Rebuild the visible-instance list and re-upload it.
        st->visible.clear();
        for (const Object &o : st->world)
            if (aabb_visible(planes, o.x, o.y, o.z, 0.5f)) st->visible.push_back(o.inst);
        upload_buffer(st->device, st->inst, st->visible.data(),
                      (Uint32)(st->visible.size()*sizeof(Instance)), true);

        if (SDL_GetTicksNS() - st->title_ns > 100'000'000ull) {   // update title ~10 Hz
            char title[96];
            SDL_snprintf(title, sizeof(title), "Frustum Culling — %zu / %d visible",
                         st->visible.size(), MAX_INST);
            SDL_SetWindowTitle(st->window, title);
            st->title_ns = SDL_GetTicksNS();
        }

        SDL_GPUColorTargetInfo c{}; c.texture=swap; c.clear_color=SDL_FColor{0.05f,0.06f,0.09f,1};
        c.load_op=SDL_GPU_LOADOP_CLEAR; c.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo d{}; d.texture=st->depth; d.clear_depth=1.0f;
        d.load_op=SDL_GPU_LOADOP_CLEAR; d.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p = SDL_BeginGPURenderPass(cmd, &c, 1, &d);
        if (!st->visible.empty()) {
            SDL_BindGPUGraphicsPipeline(p, st->pipeline);
            SDL_GPUBufferBinding vbn[2]{}; vbn[0].buffer=st->vbuf; vbn[1].buffer=st->inst;
            SDL_BindGPUVertexBuffers(p, 0, vbn, 2);
            SDL_GPUBufferBinding ib{}; ib.buffer=st->ibuf; SDL_BindGPUIndexBuffer(p,&ib, SDL_GPU_INDEXELEMENTSIZE_16BIT);
            SDL_PushGPUVertexUniformData(cmd, 0, &viewProj, sizeof(viewProj));
            SDL_DrawGPUIndexedPrimitives(p, 36, (Uint32)st->visible.size(), 0, 0, 0);
        }
        SDL_EndGPURenderPass(p);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->inst)  SDL_ReleaseGPUBuffer(st->device, st->inst);
        if (st->ibuf)  SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf)  SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        if (st->pipeline) SDL_ReleaseGPUGraphicsPipeline(st->device, st->pipeline);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Use the culling frustum as the view: cull and render from one camera, and the cubes just vanish at the screen edge as intended.
  2. Switch to a bounding sphere test — one dot product and a radius per plane, cheaper than the corner selection.
  3. Log the visible count vs. total each second; at this grid size a narrow frustum typically keeps only a small fraction on screen.
  4. Add a coarse grid or BVH so whole clusters can be rejected with one test before you ever touch individual objects.
