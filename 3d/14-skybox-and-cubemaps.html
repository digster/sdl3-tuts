Recipe 14 — Skybox & Cubemaps

Wrap the world in a sky. A six-faced cubemap, sampled by view direction, drawn behind everything with one depth trick so it costs almost nothing.

What you'll learn

  • What a cubemap is: one texture, six square faces, addressed by a 3D direction.
  • Creating an SDL_GPU_TEXTURETYPE_CUBE texture and uploading its six layers.
  • The skybox vertex trick — position.xyww forces depth to the far plane.
  • Drawing the sky last with LESS_OR_EQUAL so it only fills empty pixels.

Prerequisites

Recipe 08 — Textures & Samplers for the upload/sampler machinery, the cube from recipe 06, and recipe 07 for the depth buffer the trick relies on.

Walkthrough

Six faces, one texture

A cubemap is six square images — +X, −X, +Y, −Y, +Z, −Z — that together tile the inside of a cube. You don't sample it with a 2D UV; you sample it with a direction vector, and the hardware figures out which face that ray hits and where. Point a ray any which way and you get the sky in that direction. 

SDL_GPUTextureCreateInfo ti{};
ti.type   = SDL_GPU_TEXTURETYPE_CUBE;          // the magic flag
ti.format = SDL_GPU_TEXTUREFORMAT_R8G8B8A8_UNORM;
ti.usage  = SDL_GPU_TEXTUREUSAGE_SAMPLER;
ti.width  = FACE; ti.height = FACE;
ti.layer_count_or_depth = 6;                   // six layers = six faces
ti.num_levels = 1; ti.sample_count = SDL_GPU_SAMPLECOUNT_1;
st->sky = SDL_CreateGPUTexture(st->device, &ti);

Uploading the layers

Each face is a layer. We stage all six in one transfer buffer, then copy each into its layer with a SDL_GPUTextureRegion whose layer is the face index. Here we synthesise the faces in code — a gradient sky — so there's no image file to ship. 

SDL_GPUTextureRegion r{};
r.texture = st->sky; r.layer = face;           // which of the six
r.w = FACE; r.h = FACE; r.d = 1;
SDL_GPUTextureTransferInfo src{};
src.transfer_buffer = tb;
src.offset          = face * FACE * FACE * 4;  // this face's bytes in the staging buffer
SDL_UploadToGPUTexture(copy, &src, &r, false);

The skybox vertex trick

We draw a unit cube centred on the camera, using its object-space positions as sample directions. Two details make it a sky: the view matrix has its translation stripped (the sky never moves as you walk), and the output z is forced equal to w so that after the perspective divide the depth is exactly 1.0 — the far plane. 

static const char *kSkyVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; };
struct VSOut { float4 position [[position]]; float3 dir; };
struct UBO   { float4x4 vp; };                 // view (no translation) * proj
vertex VSOut skyVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o;
    o.dir = in.pos;
    float4 p = u.vp * float4(in.pos, 1.0);
    o.position = p.xyww;                        // z = w  =>  depth = 1.0
    return o;
}
)msl";
static const char *kSkyFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 dir; };
fragment float4 skyFS(VSOut in [[stage_in]],
                      texturecube<float> sky [[texture(0)]],
                      sampler smp [[sampler(0)]]) {
    return sky.sample(smp, normalize(in.dir));
}
)msl";

The sky pipeline tests depth with SDL_GPU_COMPAREOP_LESS_OR_EQUAL and does not write depth, with culling off (we're inside the cube). Draw the real scene first; the sky then paints only the pixels nothing else claimed.

Why draw the sky last? If you drew it first it would shade every pixel, then the scene would overdraw most of them — wasted work. Drawing it last with LESS_OR_EQUAL lets early-Z reject the covered pixels, so the expensive sky shader runs only on background.

Complete listing

A spinning cube floating in a synthesised gradient sky. The view orbits slowly; the sky stays put around the camera while the cube turns in front of it.

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_rotate_x(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[5]=c;r.m[6]=s;r.m[9]=-s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}

static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}

// Synthesise a gradient sky into a face (RGBA8). face: 0+X 1-X 2+Y 3-Y 4+Z 5-Z.
static void fill_face(int face, Uint8 *px)
{
    for (int y = 0; y < FACE; ++y)
    for (int x = 0; x < FACE; ++x) {
        float v = (float)y / (FACE - 1);              // 0 top .. 1 bottom
        Uint8 r, g, b;
        if (face == 2) { r=90;  g=150; b=235; }                        // up: sky blue
        else if (face == 3) { r=60; g=48; b=40; }                      // down: ground
        else { r=(Uint8)(150-90*(1-v)); g=(Uint8)(180-60*(1-v)); b=(Uint8)(235-40*(1-v)); } // horizon
        Uint8 *p = px + (y*FACE + x)*4; p[0]=r; p[1]=g; p[2]=b; p[3]=255;
    }
}

static void upload_cubemap(AppState *st)
{
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD;
    tbi.size = FACE*FACE*4*6;
    SDL_GPUTransferBuffer *tb = SDL_CreateGPUTransferBuffer(st->device, &tbi);
    Uint8 *m = (Uint8*)SDL_MapGPUTransferBuffer(st->device, tb, false);
    for (int f = 0; f < 6; ++f) fill_face(f, m + f*FACE*FACE*4);
    SDL_UnmapGPUTransferBuffer(st->device, tb);

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUCopyPass *cp = SDL_BeginGPUCopyPass(cmd);
    for (int f = 0; f < 6; ++f) {
        SDL_GPUTextureTransferInfo src{}; src.transfer_buffer=tb; src.offset=f*FACE*FACE*4;
        SDL_GPUTextureRegion r{}; r.texture=st->sky; r.layer=(Uint32)f; r.w=FACE; r.h=FACE; r.d=1;
        SDL_UploadToGPUTexture(cp, &src, &r, false);
    }
    SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(st->device, tb);
}

static void make_depth(AppState *st, Uint32 w, Uint32 h)
{
    if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->depth = SDL_CreateGPUTexture(st->device, &ti); st->fb_w=w; st->fb_h=h;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Skybox & Cubemaps", "1.0", "com.example.sdl3.skybox");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Skybox & Cubemaps", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    SDL_GPUBufferCreateInfo vbi{}; vbi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; vbi.size=sizeof(kCube); st->vbuf=SDL_CreateGPUBuffer(st->device,&vbi);
    SDL_GPUBufferCreateInfo ibi{}; ibi.usage=SDL_GPU_BUFFERUSAGE_INDEX;  ibi.size=sizeof(kIdx);  st->ibuf=SDL_CreateGPUBuffer(st->device,&ibi);
    SDL_GPUBufferCreateInfo bvi{}; bvi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; bvi.size=sizeof(kBox);  st->boxv=SDL_CreateGPUBuffer(st->device,&bvi);
    SDL_GPUBufferCreateInfo bii{}; bii.usage=SDL_GPU_BUFFERUSAGE_INDEX;  bii.size=sizeof(kBoxIdx);st->boxi=SDL_CreateGPUBuffer(st->device,&bii);
    upload_buffer(st->device, st->vbuf, kCube, sizeof(kCube));
    upload_buffer(st->device, st->ibuf, kIdx,  sizeof(kIdx));
    upload_buffer(st->device, st->boxv, kBox,  sizeof(kBox));
    upload_buffer(st->device, st->boxi, kBoxIdx, sizeof(kBoxIdx));

    SDL_GPUTextureCreateInfo ci{}; ci.type=SDL_GPU_TEXTURETYPE_CUBE; ci.format=SDL_GPU_TEXTUREFORMAT_R8G8B8A8_UNORM;
    ci.usage=SDL_GPU_TEXTUREUSAGE_SAMPLER; ci.width=FACE; ci.height=FACE; ci.layer_count_or_depth=6; ci.num_levels=1; ci.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->sky = SDL_CreateGPUTexture(st->device, &ci);
    upload_cubemap(st);

    SDL_GPUSamplerCreateInfo si{}; si.min_filter=SDL_GPU_FILTER_LINEAR; si.mag_filter=SDL_GPU_FILTER_LINEAR;
    si.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_w=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->sampler = SDL_CreateGPUSampler(st->device, &si);

    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);

    // --- scene pipeline (depth LESS, write on) ---
    SDL_GPUShader *svs=make_shader(st->device,kSceneVS,SDL_GPU_SHADERSTAGE_VERTEX,"sceneVS",1,0);
    SDL_GPUShader *sfs=make_shader(st->device,kSceneFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"sceneFS",0,0);
    SDL_GPUVertexBufferDescription svb{}; svb.slot=0; svb.pitch=sizeof(Vertex); svb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute sa[2]{};
    sa[0]={0,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,x)};
    sa[1]={1,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT4, offsetof(Vertex,r)};
    SDL_GPUVertexInputState sv{}; sv.vertex_buffer_descriptions=&svb; sv.num_vertex_buffers=1; sv.vertex_attributes=sa; sv.num_vertex_attributes=2;
    SDL_GPUColorTargetDescription sct{}; sct.format=swap;
    SDL_GPUGraphicsPipelineCreateInfo sp{};
    sp.vertex_shader=svs; sp.fragment_shader=sfs; sp.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; sp.vertex_input_state=sv;
    sp.target_info.color_target_descriptions=&sct; sp.target_info.num_color_targets=1;
    sp.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; sp.target_info.has_depth_stencil_target=true;
    sp.depth_stencil_state.enable_depth_test=true; sp.depth_stencil_state.enable_depth_write=true; sp.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    sp.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->scene=SDL_CreateGPUGraphicsPipeline(st->device,&sp);
    SDL_ReleaseGPUShader(st->device,svs); SDL_ReleaseGPUShader(st->device,sfs);
    if(!st->scene) return fail("scene pipeline");

    // --- sky pipeline (depth LESS_OR_EQUAL, no write, no cull) ---
    SDL_GPUShader *kvs=make_shader(st->device,kSkyVS,SDL_GPU_SHADERSTAGE_VERTEX,"skyVS",1,0);
    SDL_GPUShader *kfs=make_shader(st->device,kSkyFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"skyFS",0,1);
    SDL_GPUVertexBufferDescription kvb{}; kvb.slot=0; kvb.pitch=sizeof(float)*3; kvb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute ka{0,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3,0};
    SDL_GPUVertexInputState kv{}; kv.vertex_buffer_descriptions=&kvb; kv.num_vertex_buffers=1; kv.vertex_attributes=&ka; kv.num_vertex_attributes=1;
    SDL_GPUColorTargetDescription kct{}; kct.format=swap;
    SDL_GPUGraphicsPipelineCreateInfo kp{};
    kp.vertex_shader=kvs; kp.fragment_shader=kfs; kp.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; kp.vertex_input_state=kv;
    kp.target_info.color_target_descriptions=&kct; kp.target_info.num_color_targets=1;
    kp.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; kp.target_info.has_depth_stencil_target=true;
    kp.depth_stencil_state.enable_depth_test=true; kp.depth_stencil_state.enable_depth_write=false; kp.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS_OR_EQUAL;
    kp.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->skybox=SDL_CreateGPUGraphicsPipeline(st->device,&kp);
    SDL_ReleaseGPUShader(st->device,kvs); SDL_ReleaseGPUShader(st->device,kfs);
    if(!st->skybox) return fail("sky pipeline");

    st->start_ns=SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) make_depth(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;

        // An orbiting view: rotate around the cube, looking inward.
        Mat4 view = mat4_mul(mat4_translate(0,0,-3.0f), mat4_rotate_y(t*0.3f));
        Mat4 proj = mat4_perspective(60.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);
        Mat4 model = mat4_mul(mat4_rotate_y(t*0.8f), mat4_rotate_x(t*0.5f));
        Mat4 mvp = mat4_mul(proj, mat4_mul(view, model));

        // Sky uses the SAME rotation but no translation.
        Mat4 viewNoT = view; viewNoT.m[12]=viewNoT.m[13]=viewNoT.m[14]=0;
        Mat4 skyVP = mat4_mul(proj, viewNoT);

        SDL_GPUColorTargetInfo c{}; c.texture=swap; c.clear_color=SDL_FColor{0,0,0,1};
        c.load_op=SDL_GPU_LOADOP_CLEAR; c.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo d{}; d.texture=st->depth; d.clear_depth=1.0f;
        d.load_op=SDL_GPU_LOADOP_CLEAR; d.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p = SDL_BeginGPURenderPass(cmd, &c, 1, &d);

        // Scene first.
        SDL_BindGPUGraphicsPipeline(p, st->scene);
        SDL_GPUBufferBinding vb{}; vb.buffer=st->vbuf; SDL_BindGPUVertexBuffers(p,0,&vb,1);
        SDL_GPUBufferBinding ib{}; ib.buffer=st->ibuf; SDL_BindGPUIndexBuffer(p,&ib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&mvp,sizeof(mvp));
        SDL_DrawGPUIndexedPrimitives(p,36,1,0,0,0);

        // Sky last, only where the depth buffer is still 1.0.
        SDL_BindGPUGraphicsPipeline(p, st->skybox);
        SDL_GPUBufferBinding bvb{}; bvb.buffer=st->boxv; SDL_BindGPUVertexBuffers(p,0,&bvb,1);
        SDL_GPUBufferBinding bib{}; bib.buffer=st->boxi; SDL_BindGPUIndexBuffer(p,&bib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_GPUTextureSamplerBinding ts{}; ts.texture=st->sky; ts.sampler=st->sampler;
        SDL_BindGPUFragmentSamplers(p,0,&ts,1);
        SDL_PushGPUVertexUniformData(cmd,0,&skyVP,sizeof(skyVP));
        SDL_DrawGPUIndexedPrimitives(p,36,1,0,0,0);

        SDL_EndGPURenderPass(p);
    }

    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->sampler) SDL_ReleaseGPUSampler(st->device, st->sampler);
        if (st->sky)     SDL_ReleaseGPUTexture(st->device, st->sky);
        if (st->depth)   SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->boxi) SDL_ReleaseGPUBuffer(st->device, st->boxi);
        if (st->boxv) SDL_ReleaseGPUBuffer(st->device, st->boxv);
        if (st->ibuf) SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf) SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        if (st->skybox) SDL_ReleaseGPUGraphicsPipeline(st->device, st->skybox);
        if (st->scene)  SDL_ReleaseGPUGraphicsPipeline(st->device, st->scene);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Paint recognisable faces (a number or a different solid colour per side) in fill_face so you can see which face a direction lands on as the view orbits.
  2. Load six real images instead of synthesising — one SDL_LoadBMP per face, uploaded into its layer.
  3. Reflect the sky off the cube: in the scene fragment shader, sample the cubemap with the reflection of the view direction about the surface normal.
  4. Wire in the FPS camera from recipe 11 — the sky should stay rock-steady around you as you turn.
