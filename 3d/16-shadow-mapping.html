Recipe 16 — Shadow Mapping

Real cast shadows. Render the scene's depth from the light's point of view, then in the main pass ask: is this pixel further from the light than the nearest thing the light could see? If so, it's in shadow.

What you'll learn

  • The two-pass shadow-map algorithm: a depth pass from the light, then a lit pass that samples it.
  • Rendering depth-only into a sampleable depth texture (no colour target).
  • Transforming a world position into light space and comparing depths.
  • The depth bias that kills "shadow acne", and where shadows stop at the map's edge.

Prerequisites

Recipe 07 for depth buffers, recipe 09 for diffuse lighting, and recipe 12 for the idea of rendering into a texture and sampling it later.

Walkthrough

The core idea

A shadow map is a depth buffer rendered from the light. It records, for every direction the light looks, the distance to the closest surface. In the main pass you take each fragment's world position, re-project it into the light's view, and compare: if the fragment is further away than the depth the map stored for that direction, something else was closer to the light — so this fragment is occluded, i.e. in shadow.

Pass 1 — depth from the light

We need a depth texture we can later sample, so it gets both usage flags. The pass has no colour target at all — just the depth — and a trivial void fragment shader. 

SDL_GPUTextureCreateInfo ti{};
ti.type   = SDL_GPU_TEXTURETYPE_2D;
ti.format = SDL_GPU_TEXTUREFORMAT_D32_FLOAT;
ti.usage  = SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET | SDL_GPU_TEXTUREUSAGE_SAMPLER;
ti.width  = SHADOW; ti.height = SHADOW;
ti.layer_count_or_depth = 1; ti.num_levels = 1; ti.sample_count = SDL_GPU_SAMPLECOUNT_1;
st->shadow = SDL_CreateGPUTexture(st->device, &ti);

The light's view-projection is a lookAt from the light position toward the scene, times an orthographic projection (directional lights are parallel, so no perspective).

Pass 2 — compare in the lit shader

static const char *kMainFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 wn; float4 sc; };
struct FBO   { float4 lightDir; float4 baseColor; };
fragment float4 mainFS(VSOut in [[stage_in]], constant FBO &u [[buffer(0)]],
                       depth2d<float> shadowMap [[texture(0)]], sampler smp [[sampler(0)]]) {
    float3 N    = normalize(in.wn);
    float  diff = max(dot(N, normalize(-u.lightDir.xyz)), 0.0);

    float3 proj = in.sc.xyz / in.sc.w;            // light-clip → NDC
    float2 uv   = proj.xy * 0.5 + 0.5; uv.y = 1.0 - uv.y;
    float  closest = shadowMap.sample(smp, uv);   // nearest depth the light saw
    float  bias    = 0.0025;
    float  lit     = (proj.z - bias > closest) ? 0.3 : 1.0;
    if (uv.x < 0 || uv.x > 1 || uv.y < 0 || uv.y > 1) lit = 1.0;   // outside the map

    float3 col = u.baseColor.rgb * (0.15 + 0.85 * diff * lit);
    return float4(col, 1.0);
}
)msl";
Shadow acne. Without the bias, a surface shadows itself — limited depth precision makes it compare as "further than itself". Too much bias and shadows detach ("peter-panning"). The small constant here is the simplest fix; slope-scaled bias is the next step.

Complete listing

A cube hovering over a ground plane, lit by a light that orbits overhead. Watch the cube's shadow sweep across the floor — that's pass 1's depth buffer being sampled in pass 2. 

/*
 * 16-shadow-mapping.cpp — depth pass from the light, then a shadowed lit pass.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <cstring>
#include <cstddef>

static const int SHADOW = 1024;

struct Vertex { float px,py,pz, nx,ny,nz; };
// Unit cube, 24 verts with per-face normals.
static const Vertex kCube[24] = {
    { 0.5f,-0.5f,-0.5f, 1,0,0},{ 0.5f, 0.5f,-0.5f, 1,0,0},{ 0.5f, 0.5f, 0.5f, 1,0,0},{ 0.5f,-0.5f, 0.5f, 1,0,0},
    {-0.5f,-0.5f, 0.5f,-1,0,0},{-0.5f, 0.5f, 0.5f,-1,0,0},{-0.5f, 0.5f,-0.5f,-1,0,0},{-0.5f,-0.5f,-0.5f,-1,0,0},
    {-0.5f, 0.5f,-0.5f, 0,1,0},{-0.5f, 0.5f, 0.5f, 0,1,0},{ 0.5f, 0.5f, 0.5f, 0,1,0},{ 0.5f, 0.5f,-0.5f, 0,1,0},
    {-0.5f,-0.5f, 0.5f, 0,-1,0},{-0.5f,-0.5f,-0.5f,0,-1,0},{ 0.5f,-0.5f,-0.5f,0,-1,0},{ 0.5f,-0.5f, 0.5f,0,-1,0},
    {-0.5f,-0.5f, 0.5f, 0,0,1},{ 0.5f,-0.5f, 0.5f, 0,0,1},{ 0.5f, 0.5f, 0.5f, 0,0,1},{-0.5f, 0.5f, 0.5f, 0,0,1},
    { 0.5f,-0.5f,-0.5f, 0,0,-1},{-0.5f,-0.5f,-0.5f,0,0,-1},{-0.5f, 0.5f,-0.5f,0,0,-1},{ 0.5f, 0.5f,-0.5f,0,0,-1},
};
static const Uint16 kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};
// Ground: a big quad in the XZ plane (y = -1), normal +Y.
static const Vertex kGround[4] = {
    {-5,-1,-5, 0,1,0},{ 5,-1,-5, 0,1,0},{ 5,-1, 5, 0,1,0},{-5,-1, 5, 0,1,0},
};
static const Uint16 kGroundIdx[6] = { 0,2,1, 0,3,2 };

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_ortho(float l,float r,float b,float t,float n,float f){ Mat4 m{}; m.m[0]=2/(r-l); m.m[5]=2/(t-b); m.m[10]=1/(n-f); m.m[12]=-(r+l)/(r-l); m.m[13]=-(t+b)/(t-b); m.m[14]=n/(n-f); m.m[15]=1; return m; }
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}
static Mat4 mat4_look_at(float ex,float ey,float ez, float cx,float cy,float cz, float ux,float uy,float uz){
    float fx=cx-ex, fy=cy-ey, fz=cz-ez; float fl=SDL_sqrtf(fx*fx+fy*fy+fz*fz); fx/=fl; fy/=fl; fz/=fl;
    float sx=fy*uz-fz*uy, sy=fz*ux-fx*uz, sz=fx*uy-fy*ux; float sl=SDL_sqrtf(sx*sx+sy*sy+sz*sz); sx/=sl; sy/=sl; sz/=sl;
    float ux2=sy*fz-sz*fy, uy2=sz*fx-sx*fz, uz2=sx*fy-sy*fx;
    Mat4 r=mat4_identity();
    r.m[0]=sx; r.m[4]=sy; r.m[8]=sz;
    r.m[1]=ux2; r.m[5]=uy2; r.m[9]=uz2;
    r.m[2]=-fx; r.m[6]=-fy; r.m[10]=-fz;
    r.m[12]=-(sx*ex+sy*ey+sz*ez); r.m[13]=-(ux2*ex+uy2*ey+uz2*ez); r.m[14]=(fx*ex+fy*ey+fz*ez);
    return r;
}

static const char *kShadowVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct UBO  { float4x4 lightVP; float4x4 model; };
vertex float4 shadowVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    return u.lightVP * (u.model * float4(in.pos, 1.0));
}
)msl";
static const char *kShadowFS = R"msl(
#include <metal_stdlib>
using namespace metal;
fragment void shadowFS() {}
)msl";

static const char *kMainVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 wn; float4 sc; };
struct UBO   { float4x4 mvp; float4x4 lightVP; float4x4 model; };
vertex VSOut mainVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o;
    float4 world = u.model * float4(in.pos, 1.0);
    o.position = u.mvp * float4(in.pos, 1.0);
    o.wn = (u.model * float4(in.nrm, 0.0)).xyz;
    o.sc = u.lightVP * world;
    return o;
}
)msl";
static const char *kMainFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 wn; float4 sc; };
struct FBO   { float4 lightDir; float4 baseColor; };
fragment float4 mainFS(VSOut in [[stage_in]], constant FBO &u [[buffer(0)]],
                       depth2d<float> shadowMap [[texture(0)]], sampler smp [[sampler(0)]]) {
    float3 N    = normalize(in.wn);
    float  diff = max(dot(N, normalize(-u.lightDir.xyz)), 0.0);
    float3 proj = in.sc.xyz / in.sc.w;
    float2 uv   = proj.xy * 0.5 + 0.5; uv.y = 1.0 - uv.y;
    float  closest = shadowMap.sample(smp, uv);
    float  lit  = (proj.z - 0.0025 > closest) ? 0.3 : 1.0;
    if (uv.x < 0 || uv.x > 1 || uv.y < 0 || uv.y > 1) lit = 1.0;
    return float4(u.baseColor.rgb * (0.15 + 0.85 * diff * lit), 1.0);
}
)msl";

struct AppState {
    SDL_Window *window=nullptr; SDL_GPUDevice *device=nullptr;
    SDL_GPUGraphicsPipeline *shadowPipe=nullptr, *mainPipe=nullptr;
    SDL_GPUBuffer *cv=nullptr, *ci=nullptr, *gv=nullptr, *gi=nullptr;
    SDL_GPUTexture *shadow=nullptr, *depth=nullptr; SDL_GPUSampler *sampler=nullptr;
    Uint32 fb_w=0, fb_h=0; Uint64 start_ns=0;
};

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static SDL_GPUBuffer *make_buf(SDL_GPUDevice *d, SDL_GPUBufferUsageFlags u, const void *src, Uint32 size){
    SDL_GPUBufferCreateInfo bi{}; bi.usage=u; bi.size=size; SDL_GPUBuffer *b=SDL_CreateGPUBuffer(d,&bi);
    upload_buffer(d,b,src,size); return b;
}
static void make_depth(AppState *st, Uint32 w, Uint32 h){
    if (st->depth) SDL_ReleaseGPUTexture(st->device, st->depth);
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D; ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT;
    ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET; ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->depth=SDL_CreateGPUTexture(st->device,&ti); st->fb_w=w; st->fb_h=h;
}

// Build a depth-only pipeline (shadow) or the lit pipeline (main).
static SDL_GPUGraphicsPipeline *build_pipeline(AppState *st, bool depthOnly, SDL_GPUTextureFormat swap)
{
    const char *vsrc = depthOnly ? kShadowVS : kMainVS;
    const char *fsrc = depthOnly ? kShadowFS : kMainFS;
    SDL_GPUShader *vs=make_shader(st->device,vsrc,SDL_GPU_SHADERSTAGE_VERTEX, depthOnly?"shadowVS":"mainVS", 1, 0);
    SDL_GPUShader *fs=make_shader(st->device,fsrc,SDL_GPU_SHADERSTAGE_FRAGMENT, depthOnly?"shadowFS":"mainFS", depthOnly?0u:1u, depthOnly?0u:1u);

    SDL_GPUVertexBufferDescription vb{}; vb.slot=0; vb.pitch=sizeof(Vertex); vb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute a[2]{};
    a[0]={0,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,px)};
    a[1]={1,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,nx)};
    SDL_GPUVertexInputState vi{}; vi.vertex_buffer_descriptions=&vb; vi.num_vertex_buffers=1; vi.vertex_attributes=a; vi.num_vertex_attributes=2;

    SDL_GPUGraphicsPipelineCreateInfo pc{};
    pc.vertex_shader=vs; pc.fragment_shader=fs; pc.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; pc.vertex_input_state=vi;
    SDL_GPUColorTargetDescription ct{}; ct.format=swap;
    if (!depthOnly) { pc.target_info.color_target_descriptions=&ct; pc.target_info.num_color_targets=1; }
    pc.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; pc.target_info.has_depth_stencil_target=true;
    pc.depth_stencil_state.enable_depth_test=true; pc.depth_stencil_state.enable_depth_write=true; pc.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    pc.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    SDL_GPUGraphicsPipeline *p=SDL_CreateGPUGraphicsPipeline(st->device,&pc);
    SDL_ReleaseGPUShader(st->device,vs); SDL_ReleaseGPUShader(st->device,fs);
    return p;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Shadow Mapping", "1.0", "com.example.sdl3.shadow");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Shadow Mapping", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    st->cv = make_buf(st->device, SDL_GPU_BUFFERUSAGE_VERTEX, kCube,   sizeof(kCube));
    st->ci = make_buf(st->device, SDL_GPU_BUFFERUSAGE_INDEX,  kIdx,    sizeof(kIdx));
    st->gv = make_buf(st->device, SDL_GPU_BUFFERUSAGE_VERTEX, kGround, sizeof(kGround));
    st->gi = make_buf(st->device, SDL_GPU_BUFFERUSAGE_INDEX,  kGroundIdx, sizeof(kGroundIdx));

    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D; ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT;
    ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET | SDL_GPU_TEXTUREUSAGE_SAMPLER;
    ti.width=SHADOW; ti.height=SHADOW; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->shadow = SDL_CreateGPUTexture(st->device, &ti);

    SDL_GPUSamplerCreateInfo si{}; si.min_filter=SDL_GPU_FILTER_NEAREST; si.mag_filter=SDL_GPU_FILTER_NEAREST;
    si.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->sampler = SDL_CreateGPUSampler(st->device, &si);

    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    st->shadowPipe = build_pipeline(st, true,  swap);  if(!st->shadowPipe) return fail("shadow pipeline");
    st->mainPipe   = build_pipeline(st, false, swap);  if(!st->mainPipe)   return fail("main pipeline");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) make_depth(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;

        // Orbiting directional light.
        float lx = 3.5f*SDL_cosf(t*0.5f), lz = 3.5f*SDL_sinf(t*0.5f), ly = 4.5f;
        Mat4 lightView = mat4_look_at(lx,ly,lz, 0,0,0, 0,1,0);
        Mat4 lightProj = mat4_ortho(-4,4, -4,4, 0.1f, 20.0f);
        Mat4 lightVP   = mat4_mul(lightProj, lightView);

        Mat4 camView = mat4_translate(0,-0.5f,-6.0f);
        Mat4 camProj = mat4_perspective(55.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);

        Mat4 cubeModel = mat4_mul(mat4_translate(0,0.6f,0), mat4_rotate_y(t*0.7f));
        Mat4 groundModel = mat4_identity();

        // ---- Pass 1: depth from the light ----
        SDL_GPUDepthStencilTargetInfo sd{}; sd.texture=st->shadow; sd.clear_depth=1.0f;
        sd.load_op=SDL_GPU_LOADOP_CLEAR; sd.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPURenderPass *p1 = SDL_BeginGPURenderPass(cmd, nullptr, 0, &sd);
        SDL_BindGPUGraphicsPipeline(p1, st->shadowPipe);
        struct { Mat4 lightVP, model; } su;
        // cube
        su.lightVP=lightVP; su.model=cubeModel;
        SDL_GPUBufferBinding cvb{}; cvb.buffer=st->cv; SDL_BindGPUVertexBuffers(p1,0,&cvb,1);
        SDL_GPUBufferBinding cib{}; cib.buffer=st->ci; SDL_BindGPUIndexBuffer(p1,&cib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&su,sizeof(su));
        SDL_DrawGPUIndexedPrimitives(p1,36,1,0,0,0);
        // ground
        su.model=groundModel;
        SDL_GPUBufferBinding gvb{}; gvb.buffer=st->gv; SDL_BindGPUVertexBuffers(p1,0,&gvb,1);
        SDL_GPUBufferBinding gib{}; gib.buffer=st->gi; SDL_BindGPUIndexBuffer(p1,&gib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&su,sizeof(su));
        SDL_DrawGPUIndexedPrimitives(p1,6,1,0,0,0);
        SDL_EndGPURenderPass(p1);

        // ---- Pass 2: lit + shadowed to the screen ----
        SDL_GPUColorTargetInfo c{}; c.texture=swap; c.clear_color=SDL_FColor{0.05f,0.06f,0.09f,1};
        c.load_op=SDL_GPU_LOADOP_CLEAR; c.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo d{}; d.texture=st->depth; d.clear_depth=1.0f;
        d.load_op=SDL_GPU_LOADOP_CLEAR; d.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p2 = SDL_BeginGPURenderPass(cmd, &c, 1, &d);
        SDL_BindGPUGraphicsPipeline(p2, st->mainPipe);
        SDL_GPUTextureSamplerBinding ts{}; ts.texture=st->shadow; ts.sampler=st->sampler;
        SDL_BindGPUFragmentSamplers(p2,0,&ts,1);

        struct { Mat4 mvp, lightVP, model; } mu;
        struct { float lightDir[4]; float baseColor[4]; } fu;
        fu.lightDir[0]=-lx; fu.lightDir[1]=-ly; fu.lightDir[2]=-lz; fu.lightDir[3]=0;

        // cube
        mu.mvp=mat4_mul(camProj, mat4_mul(camView, cubeModel)); mu.lightVP=lightVP; mu.model=cubeModel;
        fu.baseColor[0]=0.85f; fu.baseColor[1]=0.45f; fu.baseColor[2]=0.30f; fu.baseColor[3]=1;
        SDL_BindGPUVertexBuffers(p2,0,&cvb,1); SDL_BindGPUIndexBuffer(p2,&cib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&mu,sizeof(mu)); SDL_PushGPUFragmentUniformData(cmd,0,&fu,sizeof(fu));
        SDL_DrawGPUIndexedPrimitives(p2,36,1,0,0,0);
        // ground
        mu.mvp=mat4_mul(camProj, mat4_mul(camView, groundModel)); mu.model=groundModel;
        fu.baseColor[0]=0.55f; fu.baseColor[1]=0.58f; fu.baseColor[2]=0.62f;
        SDL_BindGPUVertexBuffers(p2,0,&gvb,1); SDL_BindGPUIndexBuffer(p2,&gib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&mu,sizeof(mu)); SDL_PushGPUFragmentUniformData(cmd,0,&fu,sizeof(fu));
        SDL_DrawGPUIndexedPrimitives(p2,6,1,0,0,0);
        SDL_EndGPURenderPass(p2);
    }

    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->sampler) SDL_ReleaseGPUSampler(st->device, st->sampler);
        if (st->shadow)  SDL_ReleaseGPUTexture(st->device, st->shadow);
        if (st->depth)   SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->gi) SDL_ReleaseGPUBuffer(st->device, st->gi);
        if (st->gv) SDL_ReleaseGPUBuffer(st->device, st->gv);
        if (st->ci) SDL_ReleaseGPUBuffer(st->device, st->ci);
        if (st->cv) SDL_ReleaseGPUBuffer(st->device, st->cv);
        if (st->mainPipe)   SDL_ReleaseGPUGraphicsPipeline(st->device, st->mainPipe);
        if (st->shadowPipe) SDL_ReleaseGPUGraphicsPipeline(st->device, st->shadowPipe);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Visualise the shadow map: add a small fullscreen-corner pass that samples it and shows the depth as greyscale (recipe 12's fullscreen triangle).
  2. Soften the edges with PCF — sample the map at a 3×3 grid of offsets and average the lit/shadowed results.
  3. Tune the bias until acne and peter-panning both disappear, then make it slope-scaled: bias = max(0.005 * (1 - dot(N, L)), 0.0008).
  4. Add a second caster. The algorithm doesn't change — more geometry into pass 1 just deepens the map.
