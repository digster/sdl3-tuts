Recipe 26 — Screen-Space Ambient Occlusion

Add contact shadow for free from data you already have. Sample a hemisphere of points around each pixel in the deferred G-buffer, count how many are buried behind nearby geometry, and darken creases and corners the ambient term alone would leave flat.

What you'll learn

  • Why a flat ambient term looks lifeless, and what occlusion adds.
  • Reusing the deferred G-buffer in view space for screen-space effects.
  • A hemisphere sample kernel plus a tiled noise texture for cheap, blurrable AO.
  • Re-projecting view-space samples to the screen with the projection matrix.

Prerequisites

Recipe 25 — Deferred Shading for the multi-target G-buffer, and recipe 19 for the chained fullscreen-pass pattern. SSAO is one more pass on the buffers recipe 25 already produces.

Walkthrough

Occlusion from the depth buffer

Ambient light is a constant fudge for "light bouncing around the room", so flat surfaces and tight corners get the same brightness — wrong, because corners trap light. SSAO recovers that: for each pixel it asks how much of the hemisphere above this surface is blocked by nearby geometry? and darkens accordingly, entirely from the G-buffer.

The G-buffer, in view space

SSAO works in view space so it can re-project sample points with just the projection matrix. The geometry pass therefore writes view-space position and normal (multiply by view · model), alongside albedo — the same MRT layout as recipe 25, only the space changes. 

o.viewPos    = (modelView * float4(in.pos, 1.0)).xyz;   // modelView = view * model
o.viewNormal = (modelView * float4(in.nrm, 0.0)).xyz;

Kernel + noise

We pre-generate a kernel of ~24 points in a unit hemisphere, clustered near the origin so nearby occluders matter most. A tiny 4×4 noise texture of random in-plane vectors, tiled across the screen, randomly rotates the kernel per pixel — trading ugly banding for fine grain that a cheap blur erases. 

for (int i = 0; i < KERNEL; ++i) {                 // hemisphere, +Z
    float3 s = { SDL_randf()*2-1, SDL_randf()*2-1, SDL_randf() };
    float len = SDL_sqrtf(s.x*s.x + s.y*s.y + s.z*s.z);
    float scale = (float)i / KERNEL;
    scale = 0.1f + 0.9f * scale * scale;            // bias samples toward the centre
    kernel[i] = { s.x/len*scale, s.y/len*scale, s.z/len*scale, 0 };
}

The SSAO pass

Build a TBN frame from the pixel's normal and the noise vector, place each kernel sample in view space, project it to a screen UV, and read the G-buffer depth there. If the stored surface is closer to the camera than the sample, the sample is buried — occluded. A range check ignores occluders too far away to matter. 

# ssaoFS (excerpt)
float3 samplePos = P + TBN * u.kernel[i].xyz * u.radius;   // view space
float4 off = u.proj * float4(samplePos, 1.0);
off.xyz /= off.w;
float2 uv = off.xy * 0.5 + 0.5; uv.y = 1.0 - uv.y;          // match the post-pass UV
float sampleDepth = gPos.sample(s, uv).z;                  // view-space z there
float range = smoothstep(0.0, 1.0, u.radius / abs(P.z - sampleDepth));
occlusion += (sampleDepth >= samplePos.z + u.bias ? 1.0 : 0.0) * range;
Sign and bias are tunable. The depth comparison assumes a right-handed view space (camera down −Z), matching this site's mat4_perspective. If AO appears inverted or haloed, flip the comparison or nudge bias/radius — the same backend-sensitivity you met with shadow mapping. The final blur is a 4×4 box that hides the 4×4 noise tiling exactly.

Complete listing

A pile of cubes and a floor, rendered to a view-space G-buffer, then darkened in their crevices by SSAO (blurred) before a single directional light composites the result. Press Space to view the raw AO buffer.

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_scale(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[0]=x;r.m[5]=y;r.m[10]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}

static const char *kGeomVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 vpos; float3 vnrm; float4 base; };
struct VUBO  { float4x4 mvp; float4x4 modelView; float4 base; };
vertex VSOut geomVS(VSIn in [[stage_in]], constant VUBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp       * float4(in.pos, 1.0);
    o.vpos     = (u.modelView * float4(in.pos, 1.0)).xyz;   // view-space position
    o.vnrm     = (u.modelView * float4(in.nrm, 0.0)).xyz;   // view-space normal
    o.base     = u.base;
    return o;
}
)msl";
static const char *kGeomFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 vpos; float3 vnrm; float4 base; };
struct GBuf  { float4 albedo [[color(0)]]; float4 normal [[color(1)]]; float4 position [[color(2)]]; };
fragment GBuf geomFS(VSOut in [[stage_in]]) {
    GBuf o;
    o.albedo   = in.base;
    o.normal   = float4(normalize(in.vnrm), 1.0);
    o.position = float4(in.vpos, 1.0);
    return o;
}
)msl";

static const char *kPostVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
vertex VSOut postVS(uint vid [[vertex_id]]) {
    const float2 P[3] = { float2(-1,-1), float2(3,-1), float2(-1,3) };
    VSOut o; o.position = float4(P[vid], 0, 1);
    o.uv = P[vid] * 0.5 + 0.5; o.uv.y = 1.0 - o.uv.y;
    return o;
}
)msl";
static const char *kSSAOFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
struct SSAO  { float4x4 proj; float4 kernel[24]; float2 noiseScale; float radius; float bias; };
fragment float4 ssaoFS(VSOut in [[stage_in]],
                       texture2d<float> gPos   [[texture(0)]], sampler sc  [[sampler(0)]],
                       texture2d<float> gNorm  [[texture(1)]], sampler sc2 [[sampler(1)]],
                       texture2d<float> noiseT [[texture(2)]], sampler sr  [[sampler(2)]],
                       constant SSAO &u [[buffer(0)]]) {
    float3 P = gPos.sample(sc, in.uv).xyz;
    if (P.z == 0.0) return float4(1.0);                       // background: no occlusion
    float3 N = normalize(gNorm.sample(sc2, in.uv).xyz);
    float3 rv = normalize(noiseT.sample(sr, in.uv * u.noiseScale).xyz * 2.0 - 1.0);
    float3 T = normalize(rv - N * dot(rv, N));                // Gram-Schmidt
    float3 B = cross(N, T);
    float3x3 TBN = float3x3(T, B, N);
    float occ = 0.0;
    for (int i = 0; i < 24; ++i) {
        float3 sp = P + TBN * u.kernel[i].xyz * u.radius;     // sample in view space
        float4 off = u.proj * float4(sp, 1.0);
        off.xyz /= off.w;
        float2 uv = off.xy * 0.5 + 0.5; uv.y = 1.0 - uv.y;
        float sd = gPos.sample(sc, uv).z;
        float range = smoothstep(0.0, 1.0, u.radius / abs(P.z - sd));
        occ += (sd >= sp.z + u.bias ? 1.0 : 0.0) * range;
    }
    float ao = 1.0 - occ / 24.0;
    return float4(ao, ao, ao, 1.0);
}
)msl";
static const char *kBlurFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
struct Blur  { float2 texel; float2 pad; };
fragment float4 blurFS(VSOut in [[stage_in]],
                       texture2d<float> ao [[texture(0)]], sampler s [[sampler(0)]],
                       constant Blur &u [[buffer(0)]]) {
    float sum = 0.0;
    for (int x = -2; x < 2; ++x)
        for (int y = -2; y < 2; ++y)
            sum += ao.sample(s, in.uv + float2(x, y) * u.texel).r;
    return float4(float3(sum / 16.0), 1.0);                   // 4x4 box blur
}
)msl";
static const char *kCompositeFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
struct Comp  { float4 lightDir; float mode; float pad0, pad1, pad2; };
fragment float4 compositeFS(VSOut in [[stage_in]],
                            texture2d<float> gAlbedo [[texture(0)]], sampler s0 [[sampler(0)]],
                            texture2d<float> gNormal [[texture(1)]], sampler s1 [[sampler(1)]],
                            texture2d<float> aoTex   [[texture(2)]], sampler s2 [[sampler(2)]],
                            constant Comp &u [[buffer(0)]]) {
    float ao = aoTex.sample(s2, in.uv).r;
    if (u.mode > 0.5) return float4(ao, ao, ao, 1.0);          // raw-AO view
    float3 albedo = gAlbedo.sample(s0, in.uv).rgb;
    float3 N = gNormal.sample(s1, in.uv).xyz;
    if (dot(N, N) < 0.5) return float4(0.02, 0.02, 0.03, 1.0); // background
    N = normalize(N);
    float3 L = normalize(u.lightDir.xyz);                      // view-space light dir
    float diff = max(dot(N, L), 0.0);
    float3 ambient = albedo * 0.35 * ao;                       // occlusion darkens ambient
    float3 lit = ambient + albedo * diff * 0.8;
    return float4(lit, 1.0);
}
)msl";

static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);

static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static SDL_GPUTexture *create_color_target(SDL_GPUDevice *dev, Uint32 w, Uint32 h, SDL_GPUTextureFormat fmt){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D; ti.format=fmt;
    ti.usage=SDL_GPU_TEXTUREUSAGE_COLOR_TARGET | SDL_GPU_TEXTUREUSAGE_SAMPLER;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;

static SDL_GPUTexture *create_depth(SDL_GPUDevice *dev, Uint32 w, Uint32 h){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}
// Build a fullscreen-post pipeline (postVS + the given fragment shader).
static SDL_GPUGraphicsPipeline *make_post(SDL_GPUDevice *dev, const char *fs, const char *e,
                                          Uint32 nsamp, Uint32 nubo, SDL_GPUTextureFormat fmt){
    SDL_GPUShader *vs=make_shader(dev,kPostVS,SDL_GPU_SHADERSTAGE_VERTEX,"postVS",0,0);
    SDL_GPUShader *f =make_shader(dev,fs,SDL_GPU_SHADERSTAGE_FRAGMENT,e,nubo,nsamp);
    SDL_GPUColorTargetDescription ct{}; ct.format=fmt;
    SDL_GPUGraphicsPipelineCreateInfo p{};
    p.vertex_shader=vs; p.fragment_shader=f; p.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST;
    p.target_info.color_target_descriptions=&ct; p.target_info.num_color_targets=1;
    SDL_GPUGraphicsPipeline *pl=SDL_CreateGPUGraphicsPipeline(dev,&p);
    SDL_ReleaseGPUShader(dev,vs); SDL_ReleaseGPUShader(dev,f);
    return pl;
}
static void resize_targets(AppState *st, Uint32 w, Uint32 h){
    SDL_GPUTexture *old[6] = { st->gAlbedo, st->gNormal, st->gPosition, st->depth, st->aoTex, st->aoBlur };
    for (SDL_GPUTexture *t : old) if (t) SDL_ReleaseGPUTexture(st->device, t);
    st->gAlbedo   = create_color_target(st->device, w, h, kRGBA8);
    st->gNormal   = create_color_target(st->device, w, h, kRGBA16F);
    st->gPosition = create_color_target(st->device, w, h, kRGBA16F);
    st->depth     = create_depth(st->device, w, h);
    st->aoTex     = create_color_target(st->device, w, h, kRGBA8);
    st->aoBlur    = create_color_target(st->device, w, h, kRGBA8);
    st->fb_w=w; st->fb_h=h;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("SSAO", "1.0", "com.example.sdl3.ssao");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("SSAO", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    SDL_GPUBufferCreateInfo vbi{}; vbi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; vbi.size=sizeof(kCube); st->vbuf=SDL_CreateGPUBuffer(st->device,&vbi);
    SDL_GPUBufferCreateInfo ibi{}; ibi.usage=SDL_GPU_BUFFERUSAGE_INDEX;  ibi.size=sizeof(kIdx);  st->ibuf=SDL_CreateGPUBuffer(st->device,&ibi);
    upload_buffer(st->device, st->vbuf, kCube, sizeof(kCube));
    upload_buffer(st->device, st->ibuf, kIdx,  sizeof(kIdx));

    // Clamp sampler for the G-buffer, repeat sampler for the tiled noise.
    SDL_GPUSamplerCreateInfo sc{}; sc.min_filter=SDL_GPU_FILTER_NEAREST; sc.mag_filter=SDL_GPU_FILTER_NEAREST;
    sc.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; sc.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->clamp = SDL_CreateGPUSampler(st->device, &sc);
    SDL_GPUSamplerCreateInfo sr{}; sr.min_filter=SDL_GPU_FILTER_NEAREST; sr.mag_filter=SDL_GPU_FILTER_NEAREST;
    sr.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_REPEAT; sr.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_REPEAT;
    st->repeat = SDL_CreateGPUSampler(st->device, &sr);

    // 4x4 noise of random in-plane rotation vectors, encoded *0.5+0.5 in RGBA8.
    Uint8 npx[4*4*4];
    for (int i = 0; i < 16; ++i) {
        float x = SDL_randf()*2-1, y = SDL_randf()*2-1;
        npx[i*4+0]=(Uint8)((x*0.5f+0.5f)*255); npx[i*4+1]=(Uint8)((y*0.5f+0.5f)*255); npx[i*4+2]=128; npx[i*4+3]=255;
    }
    SDL_GPUTextureCreateInfo ni{}; ni.type=SDL_GPU_TEXTURETYPE_2D; ni.format=kRGBA8; ni.usage=SDL_GPU_TEXTUREUSAGE_SAMPLER;
    ni.width=4; ni.height=4; ni.layer_count_or_depth=1; ni.num_levels=1; ni.sample_count=SDL_GPU_SAMPLECOUNT_1;
    st->noise = SDL_CreateGPUTexture(st->device, &ni);
    {   // upload the 4x4 noise
        SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=sizeof(npx);
        SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(st->device,&tbi);
        void *m=SDL_MapGPUTransferBuffer(st->device,tb,false); std::memcpy(m,npx,sizeof(npx)); SDL_UnmapGPUTransferBuffer(st->device,tb);
        SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(st->device); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
        SDL_GPUTextureTransferInfo src{}; src.transfer_buffer=tb;
        SDL_GPUTextureRegion reg{}; reg.texture=st->noise; reg.w=4; reg.h=4; reg.d=1;
        SDL_UploadToGPUTexture(cp,&src,&reg,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
        SDL_ReleaseGPUTransferBuffer(st->device,tb);
    }

    // Pre-generate the hemisphere kernel into the persistent SSAO uniform.
    for (int i = 0; i < KERNEL; ++i) {
        float sx=SDL_randf()*2-1, sy=SDL_randf()*2-1, sz=SDL_randf();
        float len=SDL_sqrtf(sx*sx+sy*sy+sz*sz); if (len<1e-5f) len=1;
        float scale=(float)i/KERNEL; scale=0.1f+0.9f*scale*scale;
        st->ssaoUBO.kernel[i][0]=sx/len*scale; st->ssaoUBO.kernel[i][1]=sy/len*scale;
        st->ssaoUBO.kernel[i][2]=sz/len*scale; st->ssaoUBO.kernel[i][3]=0;
    }
    st->ssaoUBO.radius=0.6f; st->ssaoUBO.bias=0.025f;

    // --- geometry pipeline: view-space MRT G-buffer ---
    SDL_GPUShader *gvs=make_shader(st->device,kGeomVS,SDL_GPU_SHADERSTAGE_VERTEX,"geomVS",1,0);
    SDL_GPUShader *gfs=make_shader(st->device,kGeomFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"geomFS",0,0);
    SDL_GPUVertexBufferDescription gvb{}; gvb.slot=0; gvb.pitch=sizeof(Vertex); gvb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute ga[2]{};
    ga[0].location=0; ga[0].buffer_slot=0; ga[0].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; ga[0].offset=offsetof(Vertex,x);
    ga[1].location=1; ga[1].buffer_slot=0; ga[1].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; ga[1].offset=offsetof(Vertex,nx);
    SDL_GPUVertexInputState gvis{}; gvis.vertex_buffer_descriptions=&gvb; gvis.num_vertex_buffers=1; gvis.vertex_attributes=ga; gvis.num_vertex_attributes=2;
    SDL_GPUColorTargetDescription gct[3]{}; gct[0].format=kRGBA8; gct[1].format=kRGBA16F; gct[2].format=kRGBA16F;
    SDL_GPUGraphicsPipelineCreateInfo gp{};
    gp.vertex_shader=gvs; gp.fragment_shader=gfs; gp.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; gp.vertex_input_state=gvis;
    gp.target_info.color_target_descriptions=gct; gp.target_info.num_color_targets=3;
    gp.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; gp.target_info.has_depth_stencil_target=true;
    gp.depth_stencil_state.enable_depth_test=true; gp.depth_stencil_state.enable_depth_write=true; gp.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    gp.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->geom=SDL_CreateGPUGraphicsPipeline(st->device,&gp);
    SDL_ReleaseGPUShader(st->device,gvs); SDL_ReleaseGPUShader(st->device,gfs);
    if(!st->geom) return fail("geometry pipeline");

    // --- fullscreen pipelines ---
    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    st->ssao      = make_post(st->device, kSSAOFS,      "ssaoFS",      3, 1, kRGBA8);
    st->blur      = make_post(st->device, kBlurFS,      "blurFS",      1, 1, kRGBA8);
    st->composite = make_post(st->device, kCompositeFS, "compositeFS", 3, 1, swap);
    if (!st->ssao || !st->blur || !st->composite) return fail("post pipelines");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE) st->mode ^= 1;   // toggle raw-AO view
    }
    return SDL_APP_CONTINUE;
}

static void draw_cube(SDL_GPUCommandBuffer *cmd, SDL_GPURenderPass *pass,
                      const Mat4 &proj, const Mat4 &view, const Mat4 &model, float r, float g, float b)
{
    Mat4 mv = mat4_mul(view, model);
    VUBO v{}; v.mvp = mat4_mul(proj, mv); v.modelView = mv;
    v.base[0]=r; v.base[1]=g; v.base[2]=b; v.base[3]=1;
    SDL_PushGPUVertexUniformData(cmd, 0, &v, sizeof(v));
    SDL_DrawGPUIndexedPrimitives(pass, 36, 1, 0, 0, 0);
}

// Begin a single-target fullscreen pass, bind the pipeline + samplers, push the UBO, draw 3.
static void post_pass(SDL_GPUCommandBuffer *cmd, SDL_GPUGraphicsPipeline *pipe, SDL_GPUTexture *target,
                      SDL_GPUTextureSamplerBinding *binds, Uint32 nbinds, const void *ubo, Uint32 ubo_size)
{
    SDL_GPUColorTargetInfo c{}; c.texture=target; c.clear_color=SDL_FColor{0,0,0,1};
    c.load_op=SDL_GPU_LOADOP_CLEAR; c.store_op=SDL_GPU_STOREOP_STORE;
    SDL_GPURenderPass *p = SDL_BeginGPURenderPass(cmd, &c, 1, nullptr);
    SDL_BindGPUGraphicsPipeline(p, pipe);
    SDL_BindGPUFragmentSamplers(p, 0, binds, nbinds);
    SDL_PushGPUFragmentUniformData(cmd, 0, ubo, ubo_size);
    SDL_DrawGPUPrimitives(p, 3, 1, 0, 0);
    SDL_EndGPURenderPass(p);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) resize_targets(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;
        Mat4 proj = mat4_perspective(55.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);
        Mat4 view = mat4_mul(mat4_translate(0,-1.0f,-7.0f), mat4_rotate_y(t*0.2f));

        // ---- Pass 1: geometry into the view-space G-buffer ----
        SDL_GPUColorTargetInfo g[3]{};
        g[0].texture=st->gAlbedo;   g[0].clear_color=SDL_FColor{0,0,0,1};
        g[1].texture=st->gNormal;   g[1].clear_color=SDL_FColor{0,0,0,0};
        g[2].texture=st->gPosition; g[2].clear_color=SDL_FColor{0,0,0,0};
        for (int i=0;i<3;++i){ g[i].load_op=SDL_GPU_LOADOP_CLEAR; g[i].store_op=SDL_GPU_STOREOP_STORE; }
        SDL_GPUDepthStencilTargetInfo dti{}; dti.texture=st->depth; dti.clear_depth=1.0f;
        dti.load_op=SDL_GPU_LOADOP_CLEAR; dti.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p1 = SDL_BeginGPURenderPass(cmd, g, 3, &dti);
        SDL_BindGPUGraphicsPipeline(p1, st->geom);
        SDL_GPUBufferBinding vb{}; vb.buffer=st->vbuf; SDL_BindGPUVertexBuffers(p1,0,&vb,1);
        SDL_GPUBufferBinding ib{}; ib.buffer=st->ibuf; SDL_BindGPUIndexBuffer(p1,&ib, SDL_GPU_INDEXELEMENTSIZE_16BIT);
        draw_cube(cmd, p1, proj, view, mat4_mul(mat4_translate(0,-1.6f,0), mat4_scale(10,0.3f,10)), 0.85f,0.85f,0.88f);
        for (int i = 0; i < 6; ++i) {
            float a = i / 6.0f * 6.2831853f;
            Mat4 m = mat4_mul(mat4_translate(SDL_cosf(a)*2.2f, -1.0f + (i%2)*0.8f, SDL_sinf(a)*2.2f), mat4_rotate_y(t + a));
            draw_cube(cmd, p1, proj, view, m, 0.8f, 0.78f, 0.74f);
        }
        SDL_EndGPURenderPass(p1);

        // ---- Pass 2: SSAO ----
        st->ssaoUBO.proj = proj;
        st->ssaoUBO.noiseScale[0] = sw / 4.0f; st->ssaoUBO.noiseScale[1] = sh / 4.0f;
        SDL_GPUTextureSamplerBinding sb[3]{};
        sb[0].texture=st->gPosition; sb[0].sampler=st->clamp;
        sb[1].texture=st->gNormal;   sb[1].sampler=st->clamp;
        sb[2].texture=st->noise;     sb[2].sampler=st->repeat;
        post_pass(cmd, st->ssao, st->aoTex, sb, 3, &st->ssaoUBO, sizeof(st->ssaoUBO));

        // ---- Pass 3: blur the AO ----
        BlurUBO bu{ { 1.0f/sw, 1.0f/sh }, {0,0} };
        SDL_GPUTextureSamplerBinding bb{}; bb.texture=st->aoTex; bb.sampler=st->clamp;
        post_pass(cmd, st->blur, st->aoBlur, &bb, 1, &bu, sizeof(bu));

        // ---- Pass 4: composite to the swapchain ----
        CompUBO cu{}; cu.lightDir[0]=0.4f; cu.lightDir[1]=0.7f; cu.lightDir[2]=0.6f; cu.lightDir[3]=0;
        cu.mode = (float)st->mode;
        SDL_GPUColorTargetInfo sc{}; sc.texture=swap; sc.clear_color=SDL_FColor{0,0,0,1};
        sc.load_op=SDL_GPU_LOADOP_CLEAR; sc.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPURenderPass *p4 = SDL_BeginGPURenderPass(cmd, &sc, 1, nullptr);
        SDL_BindGPUGraphicsPipeline(p4, st->composite);
        SDL_GPUTextureSamplerBinding cb[3]{};
        cb[0].texture=st->gAlbedo; cb[0].sampler=st->clamp;
        cb[1].texture=st->gNormal; cb[1].sampler=st->clamp;
        cb[2].texture=st->aoBlur;  cb[2].sampler=st->clamp;
        SDL_BindGPUFragmentSamplers(p4, 0, cb, 3);
        SDL_PushGPUFragmentUniformData(cmd, 0, &cu, sizeof(cu));
        SDL_DrawGPUPrimitives(p4, 3, 1, 0, 0);
        SDL_EndGPURenderPass(p4);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        SDL_GPUTexture *tex[7] = { st->gAlbedo, st->gNormal, st->gPosition, st->depth, st->aoTex, st->aoBlur, st->noise };
        for (SDL_GPUTexture *x : tex) if (x) SDL_ReleaseGPUTexture(st->device, x);
        if (st->clamp)  SDL_ReleaseGPUSampler(st->device, st->clamp);
        if (st->repeat) SDL_ReleaseGPUSampler(st->device, st->repeat);
        if (st->ibuf) SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf) SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        SDL_GPUGraphicsPipeline *pl[4] = { st->geom, st->ssao, st->blur, st->composite };
        for (SDL_GPUGraphicsPipeline *x : pl) if (x) SDL_ReleaseGPUGraphicsPipeline(st->device, x);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Press Space to see the raw, blurred AO — then comment out the blur pass and watch the 4×4 noise grain reappear.
  2. Sweep radius and bias (bind them to keys). Too large a radius makes distant geometry occlude; too small and the effect vanishes.
  3. Modulate the diffuse term by AO too, not just ambient, for a stronger (if less physically-honest) look.
  4. Drop the kernel to 8 samples and add a second, wider blur — often cheaper than 24 samples for a similar result.
