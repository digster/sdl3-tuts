Recipe 25 — Deferred Shading

Decouple lighting from geometry. Write surface data — albedo, normal, position — into a multi-target G-buffer in one pass, then light the whole screen in a second pass. The cost of many lights stops depending on how much geometry you draw.

What you'll learn

  • Rendering to multiple colour targets (MRT) from one fragment shader.
  • What a G-buffer stores and why floating-point targets avoid encoding normals.
  • A screen-space lighting pass that loops dozens of lights over G-buffer samples.
  • Why deferred turns light cost from lights × fragments into lights + fragments.

Prerequisites

Recipe 22 — Point & Spot Lights (the forward version this fixes) and recipe 12 for offscreen targets and the fullscreen triangle. Deferred is render-to-texture taken to several targets at once.

Walkthrough

The G-buffer

Instead of lighting each surface as it's drawn, the geometry pass records what it would need to light it later: base colour, world-space normal and world-space position — one texture each. Normals and positions go in R16G16B16A16_FLOAT targets so signed, unbounded values survive without packing. 

st->gAlbedo   = create_color_target(dev, w, h, SDL_GPU_TEXTUREFORMAT_R8G8B8A8_UNORM);
st->gNormal   = create_color_target(dev, w, h, SDL_GPU_TEXTUREFORMAT_R16G16B16A16_FLOAT);
st->gPosition = create_color_target(dev, w, h, SDL_GPU_TEXTUREFORMAT_R16G16B16A16_FLOAT);

One fragment shader, three outputs

MRT is a fragment shader that returns a struct, each member tagged with a [[color(n)]] attribute matching a bound target. The geometry pass fills all three in a single draw. 

struct GBuf {
    float4 albedo   [[color(0)]];
    float4 normal   [[color(1)]];
    float4 position [[color(2)]];
};
fragment GBuf geomFS(VSOut in [[stage_in]]) {
    GBuf o;
    o.albedo   = in.base;
    o.normal   = float4(normalize(in.normal), 1.0);   // raw, thanks to the float target
    o.position = float4(in.world, 1.0);
    return o;
}

The pipeline lists three colour target descriptions and the render pass binds three SDL_GPUColorTargetInfos — albedo, normal, position — plus the shared depth buffer.

Light the screen, not the scene

The second pass is a single fullscreen triangle. For each pixel it reads the G-buffer, reconstructs the surface, and loops over every light. The lighting cost is now per screen pixel, independent of how many triangles produced that pixel — so dozens of lights stay cheap. 

float3 albedo = gAlbedo.sample(smp, uv).rgb;
float3 N      = gNormal.sample(smp, uv).xyz;
if (dot(N, N) < 0.5) return float4(BG, 1.0);          // background: nothing was drawn
float3 P = gPosition.sample(smp, uv).xyz;
float3 V = normalize(u.view_pos.xyz - P);
float3 lit = albedo * 0.06;
for (int i = 0; i < int(u.count.x); ++i) { /* accumulate each light */ }
The trade-off. Deferred fixes the many-lights problem but costs G-buffer bandwidth and — famously — can't blend transparent surfaces (they have no single depth) or vary materials per light easily. Real engines run deferred for opaque geometry and a forward pass for transparency.

Complete listing

A floor and a ring of cubes are rasterised once into a three-target G-buffer, then lit by sixteen coloured point lights orbiting overhead in a single screen-space pass. Add more lights and the geometry pass doesn't care. 

/*
 * 25-deferred-shading.cpp — MRT G-buffer (albedo/normal/position) + a screen-space
 * lighting pass looping many point lights. Built on recipes 12 and 22.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <cstring>
#include <cstddef>

#define MAX_LIGHTS 32

struct Vertex { float x, y, z, nx, ny, nz; };
static const Vertex kCube[24] = {
    {  0.5f,-0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f,-0.5f, 1,0,0 },{  0.5f, 0.5f, 0.5f, 1,0,0 },{  0.5f,-0.5f, 0.5f, 1,0,0 },
    { -0.5f,-0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f, 0.5f,-1,0,0 },{ -0.5f, 0.5f,-0.5f,-1,0,0 },{ -0.5f,-0.5f,-0.5f,-1,0,0 },
    { -0.5f, 0.5f,-0.5f, 0,1,0 },{ -0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f, 0.5f, 0,1,0 },{  0.5f, 0.5f,-0.5f, 0,1,0 },
    { -0.5f,-0.5f, 0.5f, 0,-1,0 },{ -0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f,-0.5f, 0,-1,0 },{  0.5f,-0.5f, 0.5f, 0,-1,0 },
    { -0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f,-0.5f, 0.5f, 0,0,1 },{  0.5f, 0.5f, 0.5f, 0,0,1 },{ -0.5f, 0.5f, 0.5f, 0,0,1 },
    {  0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f,-0.5f,-0.5f, 0,0,-1 },{ -0.5f, 0.5f,-0.5f, 0,0,-1 },{  0.5f, 0.5f,-0.5f, 0,0,-1 },
};
static const Uint16 kIdx[36] = {
    0,1,2, 0,2,3, 4,5,6, 4,6,7, 8,9,10, 8,10,11,
    12,13,14, 12,14,15, 16,17,18, 16,18,19, 20,21,22, 20,22,23,
};

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_scale(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[0]=x;r.m[5]=y;r.m[10]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}

static const char *kGeomVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 world; float3 normal; float4 base; };
struct VUBO  { float4x4 mvp; float4x4 model; float4 base; };
vertex VSOut geomVS(VSIn in [[stage_in]], constant VUBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp   * float4(in.pos, 1.0);
    o.world    = (u.model * float4(in.pos, 1.0)).xyz;
    o.normal   = (u.model * float4(in.nrm, 0.0)).xyz;
    o.base     = u.base;
    return o;
}
)msl";
static const char *kGeomFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 world; float3 normal; float4 base; };
struct GBuf  { float4 albedo [[color(0)]]; float4 normal [[color(1)]]; float4 position [[color(2)]]; };
fragment GBuf geomFS(VSOut in [[stage_in]]) {
    GBuf o;
    o.albedo   = in.base;
    o.normal   = float4(normalize(in.normal), 1.0);
    o.position = float4(in.world, 1.0);
    return o;
}
)msl";

static const char *kPostVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
vertex VSOut postVS(uint vid [[vertex_id]]) {
    const float2 P[3] = { float2(-1,-1), float2(3,-1), float2(-1,3) };
    VSOut o; o.position = float4(P[vid], 0, 1);
    o.uv = P[vid] * 0.5 + 0.5; o.uv.y = 1.0 - o.uv.y;
    return o;
}
)msl";
static const char *kLightFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float2 uv; };
struct Light { float4 pos_range; float4 color_int; };
struct LUBO  { float4 view_pos; float4 count; Light lights[32]; };
fragment float4 lightFS(VSOut in [[stage_in]],
                        texture2d<float> gAlbedo  [[texture(0)]], sampler sA [[sampler(0)]],
                        texture2d<float> gNormal  [[texture(1)]], sampler sN [[sampler(1)]],
                        texture2d<float> gPosition[[texture(2)]], sampler sP [[sampler(2)]],
                        constant LUBO &u [[buffer(0)]]) {
    float3 albedo = gAlbedo.sample(sA, in.uv).rgb;
    float3 N = gNormal.sample(sN, in.uv).xyz;
    if (dot(N, N) < 0.5) return float4(0.02, 0.02, 0.03, 1.0);   // background pixel
    N = normalize(N);
    float3 P = gPosition.sample(sP, in.uv).xyz;
    float3 V = normalize(u.view_pos.xyz - P);
    float3 lit = albedo * 0.06;
    for (int i = 0; i < int(u.count.x); ++i) {
        float3 toL  = u.lights[i].pos_range.xyz - P;
        float  dist = length(toL);
        float3 L    = toL / max(dist, 1e-4);
        float  att  = clamp(1.0 - dist / u.lights[i].pos_range.w, 0.0, 1.0); att *= att;
        float  diff = max(dot(N, L), 0.0);
        float  spec = pow(max(dot(N, normalize(L + V)), 0.0), 32.0);
        lit += (albedo * diff + spec) * u.lights[i].color_int.rgb * u.lights[i].color_int.w * att;
    }
    return float4(lit, 1.0);
}
)msl";

struct VUBO  { Mat4 mvp; Mat4 model; float base[4]; };
struct Light { float pos_range[4]; float color_int[4]; };
struct LUBO  { float view_pos[4]; float count[4]; Light lights[MAX_LIGHTS]; };

struct AppState {
    SDL_Window *window=nullptr; SDL_GPUDevice *device=nullptr;
    SDL_GPUGraphicsPipeline *geom=nullptr, *light=nullptr;
    SDL_GPUBuffer *vbuf=nullptr, *ibuf=nullptr;
    SDL_GPUTexture *gAlbedo=nullptr, *gNormal=nullptr, *gPosition=nullptr, *depth=nullptr;
    SDL_GPUSampler *sampler=nullptr;
    Uint32 fb_w=0, fb_h=0; Uint64 start_ns=0;
};

static const SDL_GPUTextureFormat kRGBA8 = SDL_GPU_TEXTUREFORMAT_R8G8B8A8_UNORM;
static const SDL_GPUTextureFormat kRGBA16F = SDL_GPU_TEXTUREFORMAT_R16G16B16A16_FLOAT;

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static SDL_GPUTexture *create_color_target(SDL_GPUDevice *dev, Uint32 w, Uint32 h, SDL_GPUTextureFormat fmt){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D; ti.format=fmt;
    ti.usage=SDL_GPU_TEXTUREUSAGE_COLOR_TARGET | SDL_GPU_TEXTUREUSAGE_SAMPLER;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}
static SDL_GPUTexture *create_depth(SDL_GPUDevice *dev, Uint32 w, Uint32 h){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}
static void resize_gbuffer(AppState *st, Uint32 w, Uint32 h){
    if (st->gAlbedo)   SDL_ReleaseGPUTexture(st->device, st->gAlbedo);
    if (st->gNormal)   SDL_ReleaseGPUTexture(st->device, st->gNormal);
    if (st->gPosition) SDL_ReleaseGPUTexture(st->device, st->gPosition);
    if (st->depth)     SDL_ReleaseGPUTexture(st->device, st->depth);
    st->gAlbedo   = create_color_target(st->device, w, h, kRGBA8);
    st->gNormal   = create_color_target(st->device, w, h, kRGBA16F);
    st->gPosition = create_color_target(st->device, w, h, kRGBA16F);
    st->depth     = create_depth(st->device, w, h);
    st->fb_w=w; st->fb_h=h;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Deferred Shading", "1.0", "com.example.sdl3.deferred");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Deferred Shading", 800, 600, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    SDL_GPUBufferCreateInfo vbi{}; vbi.usage=SDL_GPU_BUFFERUSAGE_VERTEX; vbi.size=sizeof(kCube); st->vbuf=SDL_CreateGPUBuffer(st->device,&vbi);
    SDL_GPUBufferCreateInfo ibi{}; ibi.usage=SDL_GPU_BUFFERUSAGE_INDEX;  ibi.size=sizeof(kIdx);  st->ibuf=SDL_CreateGPUBuffer(st->device,&ibi);
    upload_buffer(st->device, st->vbuf, kCube, sizeof(kCube));
    upload_buffer(st->device, st->ibuf, kIdx,  sizeof(kIdx));

    SDL_GPUSamplerCreateInfo si{}; si.min_filter=SDL_GPU_FILTER_NEAREST; si.mag_filter=SDL_GPU_FILTER_NEAREST;
    si.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->sampler = SDL_CreateGPUSampler(st->device, &si);

    // --- geometry pipeline: three colour targets (MRT) + depth ---
    SDL_GPUShader *gvs=make_shader(st->device,kGeomVS,SDL_GPU_SHADERSTAGE_VERTEX,"geomVS",1,0);
    SDL_GPUShader *gfs=make_shader(st->device,kGeomFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"geomFS",0,0);
    SDL_GPUVertexBufferDescription gvb{}; gvb.slot=0; gvb.pitch=sizeof(Vertex); gvb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute ga[2]{};
    ga[0].location=0; ga[0].buffer_slot=0; ga[0].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; ga[0].offset=offsetof(Vertex,x);
    ga[1].location=1; ga[1].buffer_slot=0; ga[1].format=SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3; ga[1].offset=offsetof(Vertex,nx);
    SDL_GPUVertexInputState gvis{}; gvis.vertex_buffer_descriptions=&gvb; gvis.num_vertex_buffers=1; gvis.vertex_attributes=ga; gvis.num_vertex_attributes=2;
    SDL_GPUColorTargetDescription gct[3]{}; gct[0].format=kRGBA8; gct[1].format=kRGBA16F; gct[2].format=kRGBA16F;
    SDL_GPUGraphicsPipelineCreateInfo gp{};
    gp.vertex_shader=gvs; gp.fragment_shader=gfs; gp.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; gp.vertex_input_state=gvis;
    gp.target_info.color_target_descriptions=gct; gp.target_info.num_color_targets=3;     // MRT
    gp.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; gp.target_info.has_depth_stencil_target=true;
    gp.depth_stencil_state.enable_depth_test=true; gp.depth_stencil_state.enable_depth_write=true; gp.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    gp.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;
    st->geom = SDL_CreateGPUGraphicsPipeline(st->device, &gp);
    SDL_ReleaseGPUShader(st->device,gvs); SDL_ReleaseGPUShader(st->device,gfs);
    if(!st->geom) return fail("geometry pipeline");

    // --- lighting pipeline: fullscreen triangle -> swapchain, 3 G-buffer samplers ---
    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    SDL_GPUShader *lvs=make_shader(st->device,kPostVS,SDL_GPU_SHADERSTAGE_VERTEX,"postVS",0,0);
    SDL_GPUShader *lfs=make_shader(st->device,kLightFS,SDL_GPU_SHADERSTAGE_FRAGMENT,"lightFS",1,3);
    SDL_GPUColorTargetDescription lct{}; lct.format=swap;
    SDL_GPUGraphicsPipelineCreateInfo lp{};
    lp.vertex_shader=lvs; lp.fragment_shader=lfs; lp.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST;
    lp.target_info.color_target_descriptions=&lct; lp.target_info.num_color_targets=1;
    st->light = SDL_CreateGPUGraphicsPipeline(st->device, &lp);
    SDL_ReleaseGPUShader(st->device,lvs); SDL_ReleaseGPUShader(st->device,lfs);
    if(!st->light) return fail("lighting pipeline");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;

static void draw_cube(SDL_GPUCommandBuffer *cmd, SDL_GPURenderPass *pass,
                      const Mat4 &vp, const Mat4 &model, float r, float g, float b)
{
    VUBO v{}; v.mvp = mat4_mul(vp, model); v.model = model;
    v.base[0]=r; v.base[1]=g; v.base[2]=b; v.base[3]=1;
    SDL_PushGPUVertexUniformData(cmd, 0, &v, sizeof(v));
    SDL_DrawGPUIndexedPrimitives(pass, 36, 1, 0, 0, 0);
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        if (sw != st->fb_w || sh != st->fb_h) resize_gbuffer(st, sw, sh);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;
        Mat4 view = mat4_mul(mat4_translate(0,-1.5f,-9.0f), mat4_rotate_y(t*0.2f));
        Mat4 vp   = mat4_mul(mat4_perspective(55.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f), view);

        // ---- Pass 1: geometry into the three-target G-buffer ----
        SDL_GPUColorTargetInfo g[3]{};
        g[0].texture=st->gAlbedo;   g[0].clear_color=SDL_FColor{0,0,0,1};
        g[1].texture=st->gNormal;   g[1].clear_color=SDL_FColor{0,0,0,0};   // 0 ⇒ "no surface"
        g[2].texture=st->gPosition; g[2].clear_color=SDL_FColor{0,0,0,0};
        for (int i = 0; i < 3; ++i) { g[i].load_op=SDL_GPU_LOADOP_CLEAR; g[i].store_op=SDL_GPU_STOREOP_STORE; }
        SDL_GPUDepthStencilTargetInfo dti{}; dti.texture=st->depth; dti.clear_depth=1.0f;
        dti.load_op=SDL_GPU_LOADOP_CLEAR; dti.store_op=SDL_GPU_STOREOP_DONT_CARE;

        SDL_GPURenderPass *p1 = SDL_BeginGPURenderPass(cmd, g, 3, &dti);
        SDL_BindGPUGraphicsPipeline(p1, st->geom);
        SDL_GPUBufferBinding vb{}; vb.buffer=st->vbuf; SDL_BindGPUVertexBuffers(p1,0,&vb,1);
        SDL_GPUBufferBinding ib{}; ib.buffer=st->ibuf; SDL_BindGPUIndexBuffer(p1,&ib, SDL_GPU_INDEXELEMENTSIZE_16BIT);
        draw_cube(cmd, p1, vp, mat4_mul(mat4_translate(0,-1.1f,0), mat4_scale(12,0.2f,12)), 0.8f,0.8f,0.85f); // floor
        for (int i = 0; i < 8; ++i) {                          // a ring of cubes
            float a = i / 8.0f * 6.2831853f;
            Mat4 m = mat4_mul(mat4_translate(SDL_cosf(a)*3.0f, -0.4f, SDL_sinf(a)*3.0f), mat4_rotate_y(t + a));
            draw_cube(cmd, p1, vp, m, 0.85f, 0.8f, 0.75f);
        }
        SDL_EndGPURenderPass(p1);

        // ---- Pass 2: light the screen from the G-buffer ----
        LUBO lub{};
        lub.view_pos[0]=0; lub.view_pos[1]=1.5f; lub.view_pos[2]=9.0f;
        lub.count[0]=16;
        for (int i = 0; i < 16; ++i) {
            float a = t * 0.6f + i / 16.0f * 6.2831853f;
            float rad = 3.4f;
            lub.lights[i].pos_range[0]=SDL_cosf(a)*rad;
            lub.lights[i].pos_range[1]=0.4f + SDL_sinf(t + i)*0.6f;
            lub.lights[i].pos_range[2]=SDL_sinf(a)*rad;
            lub.lights[i].pos_range[3]=4.5f;                    // range
            lub.lights[i].color_int[0]=0.5f + 0.5f*SDL_sinf(i*1.3f);
            lub.lights[i].color_int[1]=0.5f + 0.5f*SDL_sinf(i*1.3f + 2.0f);
            lub.lights[i].color_int[2]=0.5f + 0.5f*SDL_sinf(i*1.3f + 4.0f);
            lub.lights[i].color_int[3]=1.6f;                    // intensity
        }

        SDL_GPUColorTargetInfo cti{}; cti.texture=swap; cti.clear_color=SDL_FColor{0,0,0,1};
        cti.load_op=SDL_GPU_LOADOP_CLEAR; cti.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPURenderPass *p2 = SDL_BeginGPURenderPass(cmd, &cti, 1, nullptr);
        SDL_BindGPUGraphicsPipeline(p2, st->light);
        SDL_GPUTextureSamplerBinding gb[3]{};
        gb[0].texture=st->gAlbedo;   gb[0].sampler=st->sampler;
        gb[1].texture=st->gNormal;   gb[1].sampler=st->sampler;
        gb[2].texture=st->gPosition; gb[2].sampler=st->sampler;
        SDL_BindGPUFragmentSamplers(p2, 0, gb, 3);
        SDL_PushGPUFragmentUniformData(cmd, 0, &lub, sizeof(lub));
        SDL_DrawGPUPrimitives(p2, 3, 1, 0, 0);
        SDL_EndGPURenderPass(p2);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->sampler)   SDL_ReleaseGPUSampler(st->device, st->sampler);
        if (st->gAlbedo)   SDL_ReleaseGPUTexture(st->device, st->gAlbedo);
        if (st->gNormal)   SDL_ReleaseGPUTexture(st->device, st->gNormal);
        if (st->gPosition) SDL_ReleaseGPUTexture(st->device, st->gPosition);
        if (st->depth)     SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->ibuf) SDL_ReleaseGPUBuffer(st->device, st->ibuf);
        if (st->vbuf) SDL_ReleaseGPUBuffer(st->device, st->vbuf);
        if (st->light) SDL_ReleaseGPUGraphicsPipeline(st->device, st->light);
        if (st->geom)  SDL_ReleaseGPUGraphicsPipeline(st->device, st->geom);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. Push count to 32 and scatter the lights — the geometry pass time is unchanged, only the lighting pass grows. That invariance is the whole point.
  2. Visualise the G-buffer: bind one target and output it directly (albedo, normal*0.5+0.5, or position scaled) to see what deferred stores.
  3. Add light volumes: instead of a fullscreen triangle, draw a sphere per light so only affected pixels are shaded — the next optimisation real engines make.
  4. Pack the G-buffer tighter: store depth + reconstruct position from it, dropping the position target entirely to save bandwidth.

Deferred is the backbone that the last two recipes build on. With the G-buffer in hand, screen-space ambient occlusion adds contact shadow from the same buffers, and the track closes with a GPU-driven particle system — the two techniques that round out a modern real-time renderer.
