Recipe 31 — Water & Planar Reflections

The 3D track's finale. Render the world a second time through a mirror camera, store it in an offscreen texture, then project it back onto a rippling water plane — with animated normals and a Fresnel blend that fades the reflection in at grazing angles.

What you'll learn

  • Reflecting the camera across a plane by mirroring world space (scale(1,−1,1)).
  • Rendering that mirrored view into an offscreen colour target.
  • Projective texture lookup: sampling the reflection at the fragment's own screen position.
  • Animated wave normals for distortion and a Fresnel term for angle-dependent reflectivity.

Prerequisites

Recipe 12 — Render-to-Texture for the offscreen colour target, recipe 09 for the lit scene, and recipe 20 for the blend that lets the water sit over the scene. This ties them into one reflective surface.

Walkthrough

A mirror camera

A perfectly flat reflection is just the scene rendered from a camera mirrored across the water plane. For the plane y = 0 that mirror is a single scale: negate the world Y before the normal view transform. Because reflecting flips triangle winding, the reflection pass turns culling off (or reverses the front face). It also clips away anything below the surface (real world y < 0) so submerged geometry can't leak into the mirror — a clipBelow flag switches that discard on for the reflection pass and off for the real scene. 

Mat4 reflectVP = mat4_mul(proj, mat4_mul(view, mat4_scale(1, -1, 1)));   // mirror about y=0
// render every scene object (but NOT the water) into reflectTex with reflectVP,
// with clipBelow = 1 so the fragment shader discards world.y < 0

Project the reflection back

The reflection texture is aligned to the screen, so the water samples it at its own screen position. Pass the water fragment's clip-space coordinate through as a varying and do the perspective divide per fragment — that's projective texturing, the same divide the GPU does for gl_Position. 

# water fragment (excerpt)
float2 uv = in.clip.xy / in.clip.w * 0.5 + 0.5;   // screen position of this fragment
uv.y = 1.0 - uv.y;
float2 wave = float2(sin(in.wpos.x*0.6 + u.time), cos(in.wpos.z*0.6 + u.time)) * 0.015;
float3 refl = reflectTex.sample(smp, uv + wave).rgb;  // distorted reflection

Fresnel: reflect more at grazing angles

Real water is nearly a mirror when you look across it and nearly clear when you look straight down. The Schlick–Fresnel approximation captures that: with the plane normal (0,1,0), the reflectivity rises as the view direction's Y component shrinks. Blend the tinted water colour toward the reflection by that factor. 

# water fragment (excerpt)
float3 V = normalize(u.eye.xyz - in.wpos);
float fres = pow(1.0 - clamp(V.y, 0.0, 1.0), 3.0);    // grazing -> ~1, top-down -> ~0
float3 col = mix(u.waterColor.rgb, refl, clamp(fres + 0.15, 0.0, 1.0));
Two draws of the scene. Planar reflection costs a full extra render of everything above the water. That's why games reserve it for one prominent surface — a lake, a polished floor — and fall back to cubemaps (recipe 24) or screen-space reflections elsewhere. The offscreen target is exactly recipe 12's render-to-texture, reused.

Complete listing

Coloured cubes hovering over a water plane, orbited by the camera. The water mirrors them with a gentle ripple, and the reflection strengthens as your view flattens toward the horizon.

struct Mat4 { float m[16]; };
static Mat4 mat4_identity(){ Mat4 r{}; r.m[0]=r.m[5]=r.m[10]=r.m[15]=1; return r; }
static Mat4 mat4_translate(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[12]=x;r.m[13]=y;r.m[14]=z; return r;}
static Mat4 mat4_scale(float x,float y,float z){ Mat4 r=mat4_identity(); r.m[0]=x;r.m[5]=y;r.m[10]=z; return r;}
static Mat4 mat4_rotate_y(float a){ float c=SDL_cosf(a),s=SDL_sinf(a); Mat4 r=mat4_identity(); r.m[0]=c;r.m[2]=-s;r.m[8]=s;r.m[10]=c; return r;}
static Mat4 mat4_perspective(float fovy,float aspect,float n,float f){ float t=1/SDL_tanf(fovy*0.5f); Mat4 r{}; r.m[0]=t/aspect; r.m[5]=t; r.m[10]=f/(n-f); r.m[11]=-1; r.m[14]=(n*f)/(n-f); return r;}
static Mat4 mat4_mul(const Mat4 &a,const Mat4 &b){ Mat4 r{}; for(int c=0;c<4;++c) for(int row=0;row<4;++row){ float s=0; for(int k=0;k<4;++k) s+=a.m[k*4+row]*b.m[c*4+k]; r.m[c*4+row]=s; } return r;}
static Mat4 mat4_look_at(float ex,float ey,float ez, float cx,float cy,float cz, float ux,float uy,float uz){
    float fx=cx-ex, fy=cy-ey, fz=cz-ez; float fl=SDL_sqrtf(fx*fx+fy*fy+fz*fz); fx/=fl; fy/=fl; fz/=fl;
    float sx=fy*uz-fz*uy, sy=fz*ux-fx*uz, sz=fx*uy-fy*ux; float sl=SDL_sqrtf(sx*sx+sy*sy+sz*sz); sx/=sl; sy/=sl; sz/=sl;
    float ux2=sy*fz-sz*fy, uy2=sz*fx-sx*fz, uz2=sx*fy-sy*fx;
    Mat4 r=mat4_identity();
    r.m[0]=sx; r.m[4]=sy; r.m[8]=sz;  r.m[1]=ux2; r.m[5]=uy2; r.m[9]=uz2;
    r.m[2]=-fx; r.m[6]=-fy; r.m[10]=-fz;
    r.m[12]=-(sx*ex+sy*ey+sz*ez); r.m[13]=-(ux2*ex+uy2*ey+uz2*ez); r.m[14]=(fx*ex+fy*ey+fz*ez);
    return r;
}

static const char *kSceneVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float3 wn; float3 wpos; };
struct UBO   { float4x4 mvp; float4x4 model; };
vertex VSOut sceneVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp * float4(in.pos, 1.0);
    o.wn   = (u.model * float4(in.nrm, 0.0)).xyz;
    o.wpos = (u.model * float4(in.pos, 1.0)).xyz;    // world position for the water clip
    return o;
}
)msl";
static const char *kSceneFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float3 wn; float3 wpos; };
struct FBO   { float4 color; float4 lightDir; float clipBelow; float p0, p1, p2; };
fragment float4 sceneFS(VSOut in [[stage_in]], constant FBO &u [[buffer(0)]]) {
    // In the reflection pass, drop geometry below the water so it can't leak into the mirror.
    if (u.clipBelow > 0.5 && in.wpos.y < 0.0) discard_fragment();
    float diff = max(dot(normalize(in.wn), normalize(-u.lightDir.xyz)), 0.0);
    return float4(u.color.rgb * (0.25 + 0.8 * diff), 1.0);
}
)msl";

static const char *kWaterVS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSIn  { float3 pos [[attribute(0)]]; float3 nrm [[attribute(1)]]; };
struct VSOut { float4 position [[position]]; float4 clip; float3 wpos; };
struct UBO   { float4x4 mvp; };
vertex VSOut waterVS(VSIn in [[stage_in]], constant UBO &u [[buffer(0)]]) {
    VSOut o;
    o.position = u.mvp * float4(in.pos, 1.0);
    o.clip = o.position;                     // copy for projective sampling
    o.wpos = in.pos;                         // water authored in world space
    return o;
}
)msl";
static const char *kWaterFS = R"msl(
#include <metal_stdlib>
using namespace metal;
struct VSOut { float4 position [[position]]; float4 clip; float3 wpos; };
struct FBO   { float4 eye; float4 waterColor; float time; float p0,p1,p2; };
fragment float4 waterFS(VSOut in [[stage_in]], constant FBO &u [[buffer(0)]],
                        texture2d<float> reflectTex [[texture(0)]], sampler smp [[sampler(0)]]) {
    float2 uv = in.clip.xy / in.clip.w * 0.5 + 0.5;
    uv.y = 1.0 - uv.y;
    float2 wave = float2(sin(in.wpos.x*0.6 + u.time), cos(in.wpos.z*0.6 + u.time)) * 0.015;
    float3 refl = reflectTex.sample(smp, clamp(uv + wave, 0.001, 0.999)).rgb;
    float3 V = normalize(u.eye.xyz - in.wpos);
    float fres = pow(1.0 - clamp(V.y, 0.0, 1.0), 3.0);
    float3 col = mix(u.waterColor.rgb, refl, clamp(fres + 0.15, 0.0, 1.0));
    return float4(col, 1.0);
}
)msl";

struct SceneVUBO { Mat4 mvp, model; };
struct SceneFUBO { float color[4]; float lightDir[4]; float clipBelow, p0, p1, p2; };
struct WaterVUBO { Mat4 mvp; };
struct WaterFUBO { float eye[4]; float waterColor[4]; float time, p0, p1, p2; };

// Four cubes that bob through the surface, so the reflection clip does visible work.
static const int NUM_OBJ = 4;
static Mat4 obj_model(int i, float t) {
    float a = i / (float)NUM_OBJ * 6.2831853f + t * 0.3f;
    return mat4_mul(mat4_translate(SDL_cosf(a)*6.0f, 0.9f + SDL_sinf(t + i)*1.1f, SDL_sinf(a)*6.0f),
                    mat4_rotate_y(t + i));
}
static void obj_color(int i, float *c) {
    const float pal[4][3] = { {0.9f,0.4f,0.4f}, {0.4f,0.8f,0.5f}, {0.5f,0.6f,0.95f}, {0.9f,0.8f,0.4f} };
    c[0]=pal[i][0]; c[1]=pal[i][1]; c[2]=pal[i][2]; c[3]=1;
}

struct AppState {
    SDL_Window *window=nullptr; SDL_GPUDevice *device=nullptr;
    SDL_GPUGraphicsPipeline *scene=nullptr, *water=nullptr;
    SDL_GPUBuffer *cv=nullptr, *ci=nullptr, *wv=nullptr, *wi=nullptr;
    SDL_GPUTexture *reflectTex=nullptr, *reflectDepth=nullptr, *depth=nullptr;
    SDL_GPUSampler *sampler=nullptr;
    Uint32 fb_w=0, fb_h=0; Uint64 start_ns=0;
};

static SDL_AppResult fail(const char *w){ SDL_Log("%s: %s", w, SDL_GetError()); return SDL_APP_FAILURE; }
static SDL_GPUShader *make_shader(SDL_GPUDevice *d, const char *src, SDL_GPUShaderStage s, const char *e, Uint32 ubo, Uint32 samplers){
    SDL_GPUShaderCreateInfo i{}; i.code=(const Uint8*)src; i.code_size=std::strlen(src);
    i.entrypoint=e; i.format=SDL_GPU_SHADERFORMAT_MSL; i.stage=s; i.num_uniform_buffers=ubo; i.num_samplers=samplers;
    return SDL_CreateGPUShader(d,&i);
}
static void upload_buffer(SDL_GPUDevice *dev, SDL_GPUBuffer *dst, const void *src, Uint32 size){
    SDL_GPUTransferBufferCreateInfo tbi{}; tbi.usage=SDL_GPU_TRANSFERBUFFERUSAGE_UPLOAD; tbi.size=size;
    SDL_GPUTransferBuffer *tb=SDL_CreateGPUTransferBuffer(dev,&tbi);
    void *m=SDL_MapGPUTransferBuffer(dev,tb,false); std::memcpy(m,src,size); SDL_UnmapGPUTransferBuffer(dev,tb);
    SDL_GPUCommandBuffer *cmd=SDL_AcquireGPUCommandBuffer(dev); SDL_GPUCopyPass *cp=SDL_BeginGPUCopyPass(cmd);
    SDL_GPUTransferBufferLocation s{}; s.transfer_buffer=tb; SDL_GPUBufferRegion r{}; r.buffer=dst; r.size=size;
    SDL_UploadToGPUBuffer(cp,&s,&r,false); SDL_EndGPUCopyPass(cp); SDL_SubmitGPUCommandBuffer(cmd);
    SDL_ReleaseGPUTransferBuffer(dev,tb);
}
static SDL_GPUBuffer *make_buf(SDL_GPUDevice *d, SDL_GPUBufferUsageFlags u, const void *src, Uint32 size){
    SDL_GPUBufferCreateInfo bi{}; bi.usage=u; bi.size=size; SDL_GPUBuffer *b=SDL_CreateGPUBuffer(d,&bi);
    upload_buffer(d,b,src,size); return b;
}
static SDL_GPUTexture *create_color_target(SDL_GPUDevice *dev, Uint32 w, Uint32 h, SDL_GPUTextureFormat fmt){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D; ti.format=fmt;
    ti.usage=SDL_GPU_TEXTUREUSAGE_COLOR_TARGET | SDL_GPU_TEXTUREUSAGE_SAMPLER;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}
static SDL_GPUTexture *create_depth(SDL_GPUDevice *dev, Uint32 w, Uint32 h){
    SDL_GPUTextureCreateInfo ti{}; ti.type=SDL_GPU_TEXTURETYPE_2D;
    ti.format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; ti.usage=SDL_GPU_TEXTUREUSAGE_DEPTH_STENCIL_TARGET;
    ti.width=w; ti.height=h; ti.layer_count_or_depth=1; ti.num_levels=1; ti.sample_count=SDL_GPU_SAMPLECOUNT_1;
    return SDL_CreateGPUTexture(dev,&ti);
}
static void resize_targets(AppState *st, Uint32 w, Uint32 h, SDL_GPUTextureFormat fmt){
    SDL_GPUTexture *old[3] = { st->reflectTex, st->reflectDepth, st->depth };
    for (SDL_GPUTexture *x : old) if (x) SDL_ReleaseGPUTexture(st->device, x);
    st->reflectTex   = create_color_target(st->device, w, h, fmt);
    st->reflectDepth = create_depth(st->device, w, h);
    st->depth        = create_depth(st->device, w, h);
    st->fb_w=w; st->fb_h=h;
}

// Build the scene pipeline (depth-tested, lit) or the water pipeline (samples reflectTex).
static SDL_GPUGraphicsPipeline *build_pipeline(AppState *st, bool isWater, SDL_GPUTextureFormat fmt)
{
    SDL_GPUShader *vs=make_shader(st->device, isWater?kWaterVS:kSceneVS, SDL_GPU_SHADERSTAGE_VERTEX,
                                  isWater?"waterVS":"sceneVS", 1, 0);
    SDL_GPUShader *fs=make_shader(st->device, isWater?kWaterFS:kSceneFS, SDL_GPU_SHADERSTAGE_FRAGMENT,
                                  isWater?"waterFS":"sceneFS", 1, isWater?1u:0u);
    SDL_GPUVertexBufferDescription vb{}; vb.slot=0; vb.pitch=sizeof(Vertex); vb.input_rate=SDL_GPU_VERTEXINPUTRATE_VERTEX;
    SDL_GPUVertexAttribute a[2]{};
    a[0]={0,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,px)};
    a[1]={1,0,SDL_GPU_VERTEXELEMENTFORMAT_FLOAT3, offsetof(Vertex,nx)};
    SDL_GPUVertexInputState vi{}; vi.vertex_buffer_descriptions=&vb; vi.num_vertex_buffers=1; vi.vertex_attributes=a; vi.num_vertex_attributes=2;
    SDL_GPUColorTargetDescription ct{}; ct.format=fmt;
    SDL_GPUGraphicsPipelineCreateInfo pc{};
    pc.vertex_shader=vs; pc.fragment_shader=fs; pc.primitive_type=SDL_GPU_PRIMITIVETYPE_TRIANGLELIST; pc.vertex_input_state=vi;
    pc.target_info.color_target_descriptions=&ct; pc.target_info.num_color_targets=1;
    pc.target_info.depth_stencil_format=SDL_GPU_TEXTUREFORMAT_D32_FLOAT; pc.target_info.has_depth_stencil_target=true;
    pc.depth_stencil_state.enable_depth_test=true; pc.depth_stencil_state.enable_depth_write=true; pc.depth_stencil_state.compare_op=SDL_GPU_COMPAREOP_LESS;
    pc.rasterizer_state.cull_mode=SDL_GPU_CULLMODE_NONE;   // reflection flips winding
    SDL_GPUGraphicsPipeline *p=SDL_CreateGPUGraphicsPipeline(st->device,&pc);
    SDL_ReleaseGPUShader(st->device,vs); SDL_ReleaseGPUShader(st->device,fs);
    return p;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Water Reflections", "1.0", "com.example.sdl3.water");
    if (!SDL_Init(SDL_INIT_VIDEO)) return fail("init");
    AppState *st = new AppState(); *appstate = st;
    st->device = SDL_CreateGPUDevice(SDL_GPU_SHADERFORMAT_MSL, true, nullptr);
    st->window = SDL_CreateWindow("Water & Planar Reflections", 900, 640, SDL_WINDOW_RESIZABLE);
    SDL_ClaimWindowForGPUDevice(st->device, st->window);

    st->cv = make_buf(st->device, SDL_GPU_BUFFERUSAGE_VERTEX, kCube,  sizeof(kCube));
    st->ci = make_buf(st->device, SDL_GPU_BUFFERUSAGE_INDEX,  kIdx,   sizeof(kIdx));
    st->wv = make_buf(st->device, SDL_GPU_BUFFERUSAGE_VERTEX, kWater, sizeof(kWater));
    st->wi = make_buf(st->device, SDL_GPU_BUFFERUSAGE_INDEX,  kWaterIdx, sizeof(kWaterIdx));

    SDL_GPUSamplerCreateInfo si{}; si.min_filter=SDL_GPU_FILTER_LINEAR; si.mag_filter=SDL_GPU_FILTER_LINEAR;
    si.address_mode_u=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE; si.address_mode_v=SDL_GPU_SAMPLERADDRESSMODE_CLAMP_TO_EDGE;
    st->sampler = SDL_CreateGPUSampler(st->device, &si);

    SDL_GPUTextureFormat swap = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
    st->scene = build_pipeline(st, false, swap);  if(!st->scene) return fail("scene pipeline");
    st->water = build_pipeline(st, true,  swap);  if(!st->water) return fail("water pipeline");

    st->start_ns = SDL_GetTicksNS();
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

// Draw the four cubes with the given view-projection into the bound scene pass.
// clipBelow = 1 in the reflection pass discards geometry below the water plane.
static void draw_objects(SDL_GPUCommandBuffer *cmd, SDL_GPURenderPass *pass, AppState *st,
                         const Mat4 &vp, float t, const float *lightDir, float clipBelow)
{
    SDL_GPUBufferBinding vb{}; vb.buffer=st->cv; SDL_GPUBufferBinding ib{}; ib.buffer=st->ci;
    SDL_BindGPUVertexBuffers(pass,0,&vb,1); SDL_BindGPUIndexBuffer(pass,&ib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
    for (int i = 0; i < NUM_OBJ; ++i) {
        Mat4 model = obj_model(i, t);
        SceneVUBO vu{ mat4_mul(vp, model), model };
        SceneFUBO fu{}; obj_color(i, fu.color); fu.clipBelow = clipBelow;
        fu.lightDir[0]=lightDir[0]; fu.lightDir[1]=lightDir[1]; fu.lightDir[2]=lightDir[2];
        SDL_PushGPUVertexUniformData(cmd,0,&vu,sizeof(vu));
        SDL_PushGPUFragmentUniformData(cmd,0,&fu,sizeof(fu));
        SDL_DrawGPUIndexedPrimitives(pass,36,1,0,0,0);
    }
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    SDL_GPUCommandBuffer *cmd = SDL_AcquireGPUCommandBuffer(st->device);
    SDL_GPUTexture *swap=nullptr; Uint32 sw=0, sh=0;
    SDL_WaitAndAcquireGPUSwapchainTexture(cmd, st->window, &swap, &sw, &sh);

    if (swap) {
        SDL_GPUTextureFormat fmt = SDL_GetGPUSwapchainTextureFormat(st->device, st->window);
        if (sw != st->fb_w || sh != st->fb_h) resize_targets(st, sw, sh, fmt);
        float aspect = sh > 0 ? (float)sw/(float)sh : 1.0f;

        float ex = SDL_cosf(t*0.2f)*14.0f, ez = SDL_sinf(t*0.2f)*14.0f, ey = 5.0f;
        Mat4 view = mat4_look_at(ex, ey, ez, 0, 1.0f, 0, 0, 1, 0);
        Mat4 proj = mat4_perspective(55.0f*SDL_PI_F/180.0f, aspect, 0.1f, 100.0f);
        Mat4 vp   = mat4_mul(proj, view);
        Mat4 reflectVP = mat4_mul(proj, mat4_mul(view, mat4_scale(1, -1, 1)));   // mirror about y=0
        float lightDir[3] = { -0.4f, -0.8f, -0.3f };

        // ---- Pass 1: the scene through the mirror camera -> reflectTex ----
        SDL_GPUColorTargetInfo rc{}; rc.texture=st->reflectTex; rc.clear_color=SDL_FColor{0.55f,0.68f,0.85f,1};
        rc.load_op=SDL_GPU_LOADOP_CLEAR; rc.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo rd{}; rd.texture=st->reflectDepth; rd.clear_depth=1.0f;
        rd.load_op=SDL_GPU_LOADOP_CLEAR; rd.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p1 = SDL_BeginGPURenderPass(cmd, &rc, 1, &rd);
        SDL_BindGPUGraphicsPipeline(p1, st->scene);
        draw_objects(cmd, p1, st, reflectVP, t, lightDir, 1.0f);   // clip below the water
        SDL_EndGPURenderPass(p1);

        // ---- Pass 2: the real scene + the reflective water ----
        SDL_GPUColorTargetInfo c{}; c.texture=swap; c.clear_color=SDL_FColor{0.55f,0.68f,0.85f,1};
        c.load_op=SDL_GPU_LOADOP_CLEAR; c.store_op=SDL_GPU_STOREOP_STORE;
        SDL_GPUDepthStencilTargetInfo d{}; d.texture=st->depth; d.clear_depth=1.0f;
        d.load_op=SDL_GPU_LOADOP_CLEAR; d.store_op=SDL_GPU_STOREOP_DONT_CARE;
        SDL_GPURenderPass *p2 = SDL_BeginGPURenderPass(cmd, &c, 1, &d);

        SDL_BindGPUGraphicsPipeline(p2, st->scene);
        draw_objects(cmd, p2, st, vp, t, lightDir, 0.0f);         // real scene: no clip

        SDL_BindGPUGraphicsPipeline(p2, st->water);
        SDL_GPUTextureSamplerBinding rb{}; rb.texture=st->reflectTex; rb.sampler=st->sampler;
        SDL_BindGPUFragmentSamplers(p2, 0, &rb, 1);
        WaterVUBO wv{ vp };
        WaterFUBO wf{}; wf.eye[0]=ex; wf.eye[1]=ey; wf.eye[2]=ez;
        wf.waterColor[0]=0.10f; wf.waterColor[1]=0.20f; wf.waterColor[2]=0.28f; wf.waterColor[3]=1;
        wf.time = t;
        SDL_GPUBufferBinding wvb{}; wvb.buffer=st->wv; SDL_GPUBufferBinding wib{}; wib.buffer=st->wi;
        SDL_BindGPUVertexBuffers(p2,0,&wvb,1); SDL_BindGPUIndexBuffer(p2,&wib,SDL_GPU_INDEXELEMENTSIZE_16BIT);
        SDL_PushGPUVertexUniformData(cmd,0,&wv,sizeof(wv));
        SDL_PushGPUFragmentUniformData(cmd,0,&wf,sizeof(wf));
        SDL_DrawGPUIndexedPrimitives(p2,6,1,0,0,0);

        SDL_EndGPURenderPass(p2);
    }
    SDL_SubmitGPUCommandBuffer(cmd);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (!st) return;
    if (st->device) {
        if (st->sampler) SDL_ReleaseGPUSampler(st->device, st->sampler);
        if (st->reflectTex)   SDL_ReleaseGPUTexture(st->device, st->reflectTex);
        if (st->reflectDepth) SDL_ReleaseGPUTexture(st->device, st->reflectDepth);
        if (st->depth)        SDL_ReleaseGPUTexture(st->device, st->depth);
        if (st->wi) SDL_ReleaseGPUBuffer(st->device, st->wi);
        if (st->wv) SDL_ReleaseGPUBuffer(st->device, st->wv);
        if (st->ci) SDL_ReleaseGPUBuffer(st->device, st->ci);
        if (st->cv) SDL_ReleaseGPUBuffer(st->device, st->cv);
        if (st->water) SDL_ReleaseGPUGraphicsPipeline(st->device, st->water);
        if (st->scene) SDL_ReleaseGPUGraphicsPipeline(st->device, st->scene);
        if (st->window) SDL_ReleaseWindowFromGPUDevice(st->device, st->window);
        SDL_DestroyGPUDevice(st->device);
    }
    SDL_DestroyWindow(st->window);
    delete st;
}

Try it yourself

  1. The below-water clip is a hard plane — feather it: fade the reflection toward the water colour over a thin band around y = 0 so the cut line where cubes pierce the surface isn't razor-sharp.
  2. Add refraction: render a second offscreen pass without the mirror and blend it in under the Fresnel term for a see-through shallow look.
  3. Sample a scrolling normal map instead of the analytic sine waves for finer, more convincing ripples.
  4. Combine with recipe 28's terrain: set the water plane at a lake height and watch the hills reflect.
