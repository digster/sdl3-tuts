Recipe 16 — Render Targets & Screen Effects

Render the whole game into an offscreen texture at a fixed low resolution, then blit it to the window — and get crisp pixel-art scaling, letterboxing, screen shake and flash transitions for free.

What you'll learn

  • Creating a target texture with SDL_TEXTUREACCESS_TARGET and drawing into it with SDL_SetRenderTarget.
  • Decoupling your game's internal resolution from the window size.
  • Integer/aspect-fit scaling with letterbox bars when you present the target.
  • Two cheap effects the offscreen pass makes trivial: screen shake and a full-screen flash/fade.

Prerequisites

Recipe 05 for textures and recipe 04 for delta-time timers. This is the 2D cousin of the 3D and bgfx render-to-texture recipes.

Walkthrough

A texture you can draw onto

Pass SDL_TEXTUREACCESS_TARGET to SDL_CreateTexture and the renderer can point its output at that texture instead of the window. We size it to a fixed internal resolution — say 320×180 — and render the entire game there, never caring how big the window is. 

st->target = SDL_CreateTexture(st->renderer, SDL_PIXELFORMAT_RGBA8888,
                              SDL_TEXTUREACCESS_TARGET, GAME_W, GAME_H);
// Pixel-art crispness: nearest-neighbour when we scale it up.
SDL_SetTextureScaleMode(st->target, SDL_SCALEMODE_NEAREST);

Render into it, then back to the screen

Bind the target, draw the scene in internal coordinates, then bind nullptr to go back to the window and draw the target as one big textured quad. Everything between the two SDL_SetRenderTarget calls lands in the texture. 

SDL_SetRenderTarget(st->renderer, st->target);   // ---- offscreen ----
gfx::clear(st->renderer, 18, 20, 30);
/* ...draw the game at 320x180... */

SDL_SetRenderTarget(st->renderer, nullptr);      // ---- back to window ----
gfx::clear(st->renderer, 0, 0, 0);
SDL_RenderTexture(st->renderer, st->target, nullptr, &dst);

Letterbox scaling

To fill the window without distorting the art, scale by the smaller of the two axis ratios and centre the result. The leftover space becomes the black bars — the classic letterbox/pillarbox. 

static SDL_FRect fit_rect(int win_w, int win_h)
{
    float scale = SDL_min((float)win_w / GAME_W, (float)win_h / GAME_H);
    float w = GAME_W * scale, h = GAME_H * scale;
    return SDL_FRect{ (win_w - w) * 0.5f, (win_h - h) * 0.5f, w, h };
}

Shake and flash, almost for free

Because the whole frame is one texture, you can perturb the entire image at present time. Screen shake is a random offset added to the destination rect, its magnitude decaying over time. A flash is a translucent full-window rectangle whose alpha fades out — great for hits, pickups and scene transitions. 

// Decaying shake offset.
float s = st->shake;                                  // 0..1, ticks down each frame
dst.x += (SDL_randf() * 2 - 1) * 8.0f * s;
dst.y += (SDL_randf() * 2 - 1) * 8.0f * s;

// Fading white flash over the top.
if (st->flash > 0) {
    SDL_SetRenderDrawBlendMode(st->renderer, SDL_BLENDMODE_BLEND);
    SDL_SetRenderDrawColor(st->renderer, 255, 255, 255, (Uint8)(st->flash * 255));
    SDL_RenderFillRect(st->renderer, nullptr);        // nullptr = the whole target
}
Why fixed internal resolution? Your gameplay, collision and art all work in one stable coordinate space no matter the monitor. The single scale-up at the end keeps pixels sharp and your code resolution-independent — the bedrock of pixel-art games.

Complete listing

A bouncing square rendered at 320×180, scaled to fill the window with letterbox bars. Press Space to shake the screen and flash white; resize the window and the bars adjust while the pixels stay crisp. 

/*
 * 16-render-targets.cpp — fixed-res offscreen target, letterbox present,
 * screen shake + flash.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include "gfx.hpp"

static const int GAME_W = 320, GAME_H = 180;

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    SDL_Texture  *target = nullptr;
    float  bx = 40, by = 30, vx = 90, vy = 70;    // bouncing square
    float  shake = 0.0f, flash = 0.0f;             // both decay to zero
    Uint64 last_ns = 0;
};

static SDL_FRect fit_rect(int win_w, int win_h)
{
    float scale = SDL_min((float)win_w / GAME_W, (float)win_h / GAME_H);
    float w = GAME_W * scale, h = GAME_H * scale;
    return SDL_FRect{ (win_w - w) * 0.5f, (win_h - h) * 0.5f, w, h };
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Render Targets", "1.0", "com.example.sdl3.rendertarget");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Render Targets", 960, 540,
                                     SDL_WINDOW_RESIZABLE, &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);

    st->target = SDL_CreateTexture(st->renderer, SDL_PIXELFORMAT_RGBA8888,
                                   SDL_TEXTUREACCESS_TARGET, GAME_W, GAME_H);
    if (!st->target) { SDL_Log("target: %s", SDL_GetError()); delete st; return SDL_APP_FAILURE; }
    SDL_SetTextureScaleMode(st->target, SDL_SCALEMODE_NEAREST);

    st->last_ns = SDL_GetTicksNS();
    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE) { st->shake = 1.0f; st->flash = 0.8f; }
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->last_ns) / 1.0e9f;
    st->last_ns = now;

    // --- update (internal-resolution space) ---
    st->bx += st->vx * dt; st->by += st->vy * dt;
    if (st->bx < 0 || st->bx > GAME_W - 24) { st->vx = -st->vx; st->shake = SDL_max(st->shake, 0.5f); }
    if (st->by < 0 || st->by > GAME_H - 24) { st->vy = -st->vy; st->shake = SDL_max(st->shake, 0.5f); }
    st->shake = SDL_max(0.0f, st->shake - dt * 2.0f);
    st->flash = SDL_max(0.0f, st->flash - dt * 2.0f);

    // --- pass A: draw the game into the target ---
    SDL_SetRenderTarget(st->renderer, st->target);
    gfx::clear(st->renderer, 18, 20, 30);
    for (int i = 0; i < 5; ++i) {           // a little parallax grid
        gfx::set_color(st->renderer, 40, 46, 64, 255);
        gfx::fill_rect(st->renderer, i * 64.0f, 0, 2, GAME_H);
    }
    gfx::set_color(st->renderer, 120, 200, 255, 255);
    gfx::fill_rect(st->renderer, st->bx, st->by, 24, 24);

    // --- pass B: present the target to the window, scaled + shaken ---
    SDL_SetRenderTarget(st->renderer, nullptr);
    gfx::clear(st->renderer, 0, 0, 0);

    int win_w = 960, win_h = 540;
    SDL_GetRenderOutputSize(st->renderer, &win_w, &win_h);
    SDL_FRect dst = fit_rect(win_w, win_h);
    float s = st->shake;
    dst.x += (SDL_randf() * 2 - 1) * 10.0f * s;
    dst.y += (SDL_randf() * 2 - 1) * 10.0f * s;
    SDL_RenderTexture(st->renderer, st->target, nullptr, &dst);

    if (st->flash > 0) {
        SDL_SetRenderDrawBlendMode(st->renderer, SDL_BLENDMODE_BLEND);
        SDL_SetRenderDrawColor(st->renderer, 255, 255, 255, (Uint8)(st->flash * 255));
        SDL_RenderFillRect(st->renderer, nullptr);
    }

    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        if (st->target) SDL_DestroyTexture(st->target);
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Force integer scaling only (round the scale down to a whole number). Pixels become perfectly uniform — no shimmering — at the cost of slightly thicker bars.
  2. Add a second target the same size and cross-fade between them to transition between scenes.
  3. Tint the flash per event: white for a pickup, red for damage. Drive its colour from state instead of hard-coding white.
  4. Sample the target a second time, offset and dimmed, before the main blit — a one-line bloom-ish glow.

That's the 2D track end-to-end: from a blank window to a controller-driven, audio-mixing, save-backed game rendered through an offscreen pixel-art pipeline.
