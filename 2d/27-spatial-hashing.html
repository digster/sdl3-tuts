Recipe 27 — Spatial Hashing

Collision that scales. Checking every pair of N objects is O(n²) and melts at a few hundred entities; a uniform-grid broad phase only compares objects in the same or neighbouring cells. Toggle the two and watch the millisecond counter.

What you'll learn

  • Why brute-force pair testing is O(n²) and where it falls over.
  • A uniform spatial hash: bucket objects by (x/cell, y/cell) each frame.
  • Testing only same-cell and neighbouring-cell pairs — the broad phase / narrow phase split.
  • Measuring the difference with SDL_GetPerformanceCounter and a live overlay.

Prerequisites

Recipe 07 — Collision Detection for the circle-circle test this accelerates, and recipe 11 for updating a pool of moving objects each frame.

Walkthrough

The O(n²) wall

The naive collision loop compares object i against every j > i. That's n(n−1)/2 tests: fine for 50 objects (~1 200 tests), brutal for 800 (~320 000). Most of those tests are wasted on pairs nowhere near each other. A broad phase cheaply rejects distant pairs so the exact narrow-phase test only runs on plausible ones.

Bucket by cell

Pick a cell size roughly the diameter of an object. Each frame, clear the grid and drop every object into the bucket for its cell. Hash the cell coordinate to an index; a vector<vector<int>> of object indices is the simplest store. 

static const int CELL = 32;
static const int GW = (WIN_W + CELL - 1) / CELL, GH = (WIN_H + CELL - 1) / CELL;

std::vector<std::vector<int>> grid(GW * GH);
for (auto &b : grid) b.clear();
for (int i = 0; i < N; ++i) {
    int cx = SDL_clamp((int)(obj[i].x / CELL), 0, GW - 1);
    int cy = SDL_clamp((int)(obj[i].y / CELL), 0, GH - 1);
    grid[cy * GW + cx].push_back(i);
}

Only nearby cells

Two objects can only touch if they share a cell or sit in adjacent ones. So for each cell, test its members against each other and against the members of its neighbours. Visiting only the "forward" half of the 3×3 neighbourhood (this cell plus four of the eight around it) tests each pair exactly once. 

// Offsets that cover every unordered pair once: self + E, SW, S, SE.
static const int OX[5] = { 0, 1, -1, 0, 1 };
static const int OY[5] = { 0, 0,  1, 1, 1 };

for (int cy = 0; cy < GH; ++cy)
  for (int cx = 0; cx < GW; ++cx) {
    const auto &here = grid[cy * GW + cx];
    for (int o = 0; o < 5; ++o) {
        int nx = cx + OX[o], ny = cy + OY[o];
        if (nx < 0 || nx >= GW || ny < 0 || ny >= GH) continue;
        const auto &other = grid[ny * GW + nx];
        for (size_t a = 0; a < here.size(); ++a)
            for (size_t b = (o == 0 ? a + 1 : 0); b < other.size(); ++b)
                resolve(obj[here[a]], obj[other[b]]);   // narrow phase
    }
  }
Cell size is the tuning knob. Too large and each bucket holds many objects (back toward O(n²) locally); too small and a single object spans several cells and you re-test it. About one object-diameter is the sweet spot for uniformly-sized entities. For wildly different sizes, reach for a loose grid or a quadtree instead.

Complete listing

Several hundred bouncing balls resolving elastic collisions. Press Space to switch between the grid broad phase and brute force, and ↑/↓ to add or remove balls. The corner shows the collision-pass time in milliseconds — grid stays flat as brute force climbs. 

/*
 * 27-spatial-hashing.cpp — uniform-grid broad phase vs. brute force for
 * circle-circle collision across hundreds of moving balls, timed live.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <vector>
#include "gfx.hpp"

static const int WIN_W = 900, WIN_H = 640;
static const int CELL = 28;
static const int GW = (WIN_W + CELL - 1) / CELL, GH = (WIN_H + CELL - 1) / CELL;
static const float RADIUS = 7.0f;

struct Ball { float x, y, vx, vy; };

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    std::vector<Ball> balls;
    std::vector<std::vector<int>> grid;    // GW*GH buckets of ball indices
    bool   use_grid = true;
    double ms = 0.0;                        // last collision-pass time
    Uint64 last_ns = 0;
};

static void spawn(AppState *st, int n)
{
    for (int i = 0; i < n; ++i) {
        float a = SDL_randf() * 6.2831853f, sp = 60 + SDL_randf() * 90;
        st->balls.push_back({ 40 + SDL_randf() * (WIN_W - 80),
                              40 + SDL_randf() * (WIN_H - 80),
                              SDL_cosf(a) * sp, SDL_sinf(a) * sp });
    }
}

// Narrow phase: push two overlapping balls apart and swap the velocity component
// along the contact normal (equal-mass elastic response).
static void resolve(Ball &a, Ball &b)
{
    float dx = b.x - a.x, dy = b.y - a.y;
    float d2 = dx*dx + dy*dy, r = RADIUS * 2;
    if (d2 >= r*r || d2 < 1e-6f) return;
    float d = SDL_sqrtf(d2), nx = dx / d, ny = dy / d;
    float overlap = (r - d) * 0.5f;
    a.x -= nx * overlap; a.y -= ny * overlap;
    b.x += nx * overlap; b.y += ny * overlap;
    float va = a.vx * nx + a.vy * ny, vb = b.vx * nx + b.vy * ny, diff = vb - va;
    a.vx += diff * nx; a.vy += diff * ny;
    b.vx -= diff * nx; b.vy -= diff * ny;
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Spatial Hashing", "1.0", "com.example.sdl3.spatialhash");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Spatial Hashing", WIN_W, WIN_H, 0, &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);
    st->grid.resize(GW * GH);
    spawn(st, 400);
    st->last_ns = SDL_GetTicksNS();
    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE) st->use_grid = !st->use_grid;
        if (e->key.scancode == SDL_SCANCODE_UP)   spawn(st, 100);
        if (e->key.scancode == SDL_SCANCODE_DOWN && st->balls.size() > 100)
            st->balls.resize(st->balls.size() - 100);
    }
    return SDL_APP_CONTINUE;
}

static void collide_bruteforce(AppState *st)
{
    int n = (int)st->balls.size();
    for (int i = 0; i < n; ++i)
        for (int j = i + 1; j < n; ++j) resolve(st->balls[i], st->balls[j]);
}

static void collide_grid(AppState *st)
{
    for (auto &b : st->grid) b.clear();
    int n = (int)st->balls.size();
    for (int i = 0; i < n; ++i) {
        int cx = SDL_clamp((int)(st->balls[i].x / CELL), 0, GW - 1);
        int cy = SDL_clamp((int)(st->balls[i].y / CELL), 0, GH - 1);
        st->grid[cy * GW + cx].push_back(i);
    }
    // self + E, SW, S, SE covers every unordered neighbour pair exactly once.
    static const int OX[5] = { 0, 1, -1, 0, 1 };
    static const int OY[5] = { 0, 0,  1, 1, 1 };
    for (int cy = 0; cy < GH; ++cy)
        for (int cx = 0; cx < GW; ++cx) {
            const auto &here = st->grid[cy * GW + cx];
            for (int o = 0; o < 5; ++o) {
                int nx = cx + OX[o], ny = cy + OY[o];
                if (nx < 0 || nx >= GW || ny < 0 || ny >= GH) continue;
                const auto &other = st->grid[ny * GW + nx];
                for (size_t a = 0; a < here.size(); ++a)
                    for (size_t b = (o == 0 ? a + 1 : 0); b < other.size(); ++b)
                        resolve(st->balls[here[a]], st->balls[other[b]]);
            }
        }
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->last_ns) / 1.0e9f;
    st->last_ns = now;

    for (auto &b : st->balls) {                 // integrate + bounce off walls
        b.x += b.vx * dt; b.y += b.vy * dt;
        if (b.x < RADIUS)         { b.x = RADIUS;         b.vx = -b.vx; }
        if (b.x > WIN_W - RADIUS) { b.x = WIN_W - RADIUS; b.vx = -b.vx; }
        if (b.y < RADIUS)         { b.y = RADIUS;         b.vy = -b.vy; }
        if (b.y > WIN_H - RADIUS) { b.y = WIN_H - RADIUS; b.vy = -b.vy; }
    }

    // Time only the collision pass with the high-resolution counter.
    Uint64 t0 = SDL_GetPerformanceCounter();
    if (st->use_grid) collide_grid(st); else collide_bruteforce(st);
    Uint64 t1 = SDL_GetPerformanceCounter();
    double this_ms = (double)(t1 - t0) * 1000.0 / SDL_GetPerformanceFrequency();
    st->ms = st->ms * 0.9 + this_ms * 0.1;      // smooth the readout

    gfx::clear(st->renderer, 16, 18, 26);
    gfx::set_color(st->renderer, 150, 200, 240, 255);
    for (const auto &b : st->balls) gfx::fill_circle(st->renderer, b.x, b.y, RADIUS);

    // Overlay: a bar whose length encodes the collision time, plus mode/count pips.
    gfx::set_color(st->renderer, st->use_grid ? 120 : 230, st->use_grid ? 220 : 110, 130, 255);
    gfx::fill_rect(st->renderer, 12, 12, (float)SDL_min(400.0, st->ms * 40.0), 12);
    gfx::set_color(st->renderer, 90, 96, 110, 255);
    gfx::rect(st->renderer, 12, 12, 400, 12);
    int pips = (int)st->balls.size() / 100;    // one pip per 100 balls
    gfx::set_color(st->renderer, 200, 210, 230, 255);
    for (int i = 0; i < pips; ++i) gfx::fill_rect(st->renderer, 12.0f + i * 10.0f, 32, 6, 6);

    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Push the count to a few thousand and watch the brute-force bar hit the ceiling while the grid bar barely moves — the O(n²) vs O(n) gap made visible.
  2. Sweep CELL from tiny to huge and note the U-shaped cost: both extremes are slow, one diameter is best.
  3. Count the actual narrow-phase tests each mode performs and print the ratio — it's usually one to two orders of magnitude.
  4. Store one flat vector<int> with per-cell start offsets (a counting sort) instead of a vector-of-vectors, to kill the per-frame allocations entirely.
