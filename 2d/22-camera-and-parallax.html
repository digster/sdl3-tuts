Recipe 22 — Camera & Parallax Scrolling

Wrap world-to-screen conversion in a tiny Camera, follow a player with smoothing, then scroll several procedural background layers at fractional rates so distant scenery drifts slower than the foreground — the depth illusion every side-scroller leans on.

What you'll learn

  • A reusable Camera that maps world coordinates to screen coordinates (pan + zoom).
  • Smooth, frame-rate-independent camera follow with an exponential lerp.
  • The parallax trick: scroll each layer by camera.x · depth, where depth ∈ (0, 1].
  • Tiling infinite repeating scenery with a modulo loop instead of storing every object.

Prerequisites

Recipe 10 — Tilemap Rendering introduced a scrolling camera offset; this recipe promotes that idea to a proper Camera abstraction and layers it for depth. Recipe 04 supplies the delta-time the smoothing needs.

Walkthrough

The camera as a coordinate transform

A 2D camera is nothing more than a translation (and optional scale) applied to every world point before it's drawn. Store the world point the camera is centred on, and convert by subtracting it, scaling by zoom, and shifting to the screen centre. 

struct Camera { float x = 0, y = 0, zoom = 1.0f; };

static float world_to_screen_x(const Camera &c, float wx, int sw)
{
    return (wx - c.x) * c.zoom + sw * 0.5f;   // c.x ends up at the screen centre
}

Follow with a smoothed lerp

Snapping the camera straight to the player is jarring. Instead ease toward the target each frame. A raw lerp(cam, target, k) is frame-rate dependent, so we derive k from dt with an exponential — the same smoothing at 60 and 144 fps. 

float k = 1.0f - SDL_expf(-8.0f * dt);     // frame-rate-independent smoothing
st->cam.x += (st->player_x - st->cam.x) * k;

Parallax: scale the scroll by depth

Each background layer scrolls by the camera's position multiplied by a depth factor. A far layer at depth = 0.1 moves a tenth as fast as the foreground (depth = 1.0), so it reads as distant. The whole effect is one multiply per layer. 

float scroll = st->cam.x * layer_depth;     // 0.1 = far, 1.0 = foreground plane

Infinite scenery with a modulo loop

We don't store the hills — we generate them. Given a repeat period, find which copies fall within the screen and draw only those. A deterministic hash of the copy index varies each hill's height so the ridge never looks tiled. 

int first = (int)SDL_floorf(scroll / period) - 1;
int last  = (int)SDL_floorf((scroll + sw) / period) + 1;
for (int k = first; k <= last; ++k) {
    float sx = k * period - scroll;        // screen x of copy k
    /* draw one motif at sx */
}
Depth ordering matters. Draw back-to-front: sky, far layers, near layers, then the player and foreground. Painter's-algorithm layering is what sells the parallax — a near hill must occlude a far one.

Complete listing

Drive a glider left and right with the arrow keys. The camera eases to follow it while four procedurally generated layers — stars, two hill ridges and the ground — scroll at increasing rates, so the world feels deep even though it's all flat shapes. 

/*
 * 22-camera-and-parallax.cpp — a Camera abstraction with smoothed follow and
 * multi-layer parallax scrolling, all procedural.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include "gfx.hpp"

static const int WIN_W = 800, WIN_H = 450;

struct Camera { float x = 0, y = 0, zoom = 1.0f; };

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    Camera cam;
    float  player_x = 0, player_y = WIN_H - 120.0f;
    Uint64 last_ns = 0;
};

/* Cheap deterministic hash -> 0..1, so repeated scenery looks varied but stable. */
static float hash01(int k)
{
    Uint32 h = (Uint32)k * 2654435761u;
    h ^= h >> 15;
    return (h & 0xffff) / 65535.0f;
}

static float world_to_screen_x(const Camera &c, float wx) { return (wx - c.x) * c.zoom + WIN_W * 0.5f; }

/* One ridge of repeating triangular hills, scrolled by `scroll`. */
static void draw_hills(SDL_Renderer *r, float scroll, float base_y, float min_h, float max_h,
                       float period, Uint8 cr, Uint8 cg, Uint8 cb)
{
    gfx::set_color(r, cr, cg, cb, 255);
    int first = (int)SDL_floorf(scroll / period) - 1;
    int last  = (int)SDL_floorf((scroll + WIN_W) / period) + 1;
    for (int k = first; k <= last; ++k) {
        float sx = k * period - scroll;
        float h  = min_h + hash01(k) * (max_h - min_h);
        gfx::fill_triangle(r, sx - period * 0.6f, base_y, sx, base_y - h, sx + period * 0.6f, base_y);
    }
    gfx::fill_rect(r, 0, base_y, (float)WIN_W, (float)WIN_H - base_y);  // fill below the ridge
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Camera and Parallax", "1.0", "com.example.sdl3.parallax");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Camera and Parallax", WIN_W, WIN_H, 0,
                                     &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);
    st->cam.x = st->player_x;
    st->last_ns = SDL_GetTicksNS();
    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *, SDL_Event *e)
{
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN && e->key.scancode == SDL_SCANCODE_ESCAPE)
        return SDL_APP_SUCCESS;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->last_ns) / 1.0e9f;
    st->last_ns = now;

    // --- input: move the player, ease the camera to follow ---
    const bool *keys = SDL_GetKeyboardState(nullptr);
    float speed = 320.0f;
    if (keys[SDL_SCANCODE_LEFT])  st->player_x -= speed * dt;
    if (keys[SDL_SCANCODE_RIGHT]) st->player_x += speed * dt;

    float k = 1.0f - SDL_expf(-8.0f * dt);          // frame-rate-independent smoothing
    st->cam.x += (st->player_x - st->cam.x) * k;

    // --- draw back to front ---
    gfx::clear(st->renderer, 24, 26, 42);            // night sky

    // stars (depth 0.10)
    float star_scroll = st->cam.x * 0.10f;
    int sfirst = (int)SDL_floorf(star_scroll / 60.0f) - 1;
    int slast  = (int)SDL_floorf((star_scroll + WIN_W) / 60.0f) + 1;
    gfx::set_color(st->renderer, 220, 224, 255, 255);
    for (int i = sfirst; i <= slast; ++i) {
        float sx = i * 60.0f - star_scroll;
        float sy = 20.0f + hash01(i * 7 + 1) * 180.0f;
        gfx::fill_rect(st->renderer, sx, sy, 2, 2);
    }

    draw_hills(st->renderer, st->cam.x * 0.30f, 300, 60, 130, 260, 46, 52, 86);   // far ridge
    draw_hills(st->renderer, st->cam.x * 0.60f, 360, 80, 170, 200, 64, 74, 120);  // near ridge
    draw_hills(st->renderer, st->cam.x * 1.00f, 410, 30,  60, 150, 86, 102, 150);  // ground

    // player (foreground plane, depth 1.0)
    float px = world_to_screen_x(st->cam, st->player_x);
    gfx::set_color(st->renderer, 255, 210, 120, 255);
    gfx::fill_triangle(st->renderer, px - 18, st->player_y, px + 22, st->player_y - 6, px - 18, st->player_y - 16);
    gfx::set_color(st->renderer, 255, 150, 90, 255);
    gfx::fill_rect(st->renderer, px - 6, st->player_y - 12, 10, 6);

    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Add vertical parallax: let the player jump, follow with cam.y, and scale each layer's vertical scroll by the same depth factor.
  2. Animate zoom — bind +/- to cam.zoom. Note the foreground scales while distant layers barely react, reinforcing the depth.
  3. Add a fifth, ultra-near layer at depth = 1.3 (a few foreground grass blades). Layers can scroll faster than the player for an aggressive in-your-face effect.
  4. Introduce a dead-zone: only move the camera once the player drifts more than 80px from centre, so small movements don't shift the world.
