Recipe 25 — Streaming Textures & Pixel Effects

Write pixels by hand. A streaming texture hands you its raw memory every frame, so you can compute colour per pixel on the CPU — a classic rolling fire and a plasma field — and upload the whole buffer with one SDL_UnlockTexture.

What you'll learn

  • The three texture access modes and why effects need SDL_TEXTUREACCESS_STREAMING.
  • Locking a texture for CPU writes with SDL_LockTexture / SDL_UnlockTexture.
  • Why you must honour the returned pitch (bytes per row), not assume width × 4.
  • Two per-pixel effects — a propagating fire and a trigonometric plasma — rendered at a small internal size and scaled up.

Prerequisites

Recipe 05 — Loading Textures for textures and destination rects, and recipe 16 for the render-at-a-small-size-then-scale idea. Where recipe 16 let the GPU fill a target texture, this recipe fills the pixels itself, on the CPU.

Walkthrough

Three access modes

A texture is created with one of three access hints. STATIC (recipe 05) is uploaded rarely with SDL_UpdateTexture. TARGET (recipe 16) is drawn into by the GPU. STREAMING is the one you rewrite from the CPU every frame — SDL keeps it in memory you can map, so a per-pixel effect costs one lock and one upload, not thousands of draw calls. 

st->tex = SDL_CreateTexture(r, SDL_PIXELFORMAT_ARGB8888,
                           SDL_TEXTUREACCESS_STREAMING, FW, FH);   // small internal size
SDL_SetTextureScaleMode(st->tex, SDL_SCALEMODE_NEAREST);           // crisp, chunky pixels

Lock, write, unlock

SDL_LockTexture gives you a raw pointer plus a pitch: the number of bytes from the start of one row to the start of the next. It is frequently larger than width × 4 because drivers pad rows for alignment, so you must step by pitch and never assume a tight buffer. 

void *pixels = nullptr; int pitch = 0;
SDL_LockTexture(st->tex, nullptr, &pixels, &pitch);
for (int y = 0; y < FH; ++y) {
    Uint32 *row = (Uint32 *)((Uint8 *)pixels + y * pitch);   // pitch is BYTES
    for (int x = 0; x < FW; ++x) row[x] = argb;              // one packed pixel
}
SDL_UnlockTexture(st->tex);                                   // uploads the whole buffer

The fire, one row at a time

The rolling-fire trick keeps a heat field the size of the texture. The bottom row is reseeded hot each frame; every pixel above cools as it copies heat upward from its neighbours. A palette maps heat → black → red → orange → yellow → white. 

for (int x = 0; x < FW; ++x)                        // reseed the base
    heat[(FH - 1) * FW + x] = on ? 200 + (Uint8)(SDL_rand(56)) : 0;

for (int y = 0; y < FH - 1; ++y)                    // propagate + cool upward
    for (int x = 0; x < FW; ++x) {
        int below = (y + 1) * FW + ((x + (int)SDL_rand(3) - 1 + FW) % FW);
        int decay = (int)SDL_rand(3);               // random cooling
        int v = heat[below] - decay;
        heat[y * FW + x] = (Uint8)(v < 0 ? 0 : v);
    }
Small buffer, big window. Fire and plasma are computed at a low internal resolution (a couple hundred pixels wide) and stretched to fill the window by SDL_RenderTexture. That keeps the per-pixel CPU work cheap while SDL_SCALEMODE_NEAREST gives the crunchy, deliberately-pixelated look.

Complete listing

A streaming texture updated every frame. Press Space to switch between the rolling fire and a plasma field; hold the mouse button to paint extra heat into the fire. 

/*
 * 25-streaming-textures.cpp — a STREAMING texture written per-pixel on the CPU:
 * a propagating fire and a plasma field, computed small and scaled to the window.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include <vector>
#include "gfx.hpp"

static const int WIN_W = 800, WIN_H = 600;
static const int FW = 256, FH = 192;                 // internal effect resolution

// heat -> ARGB fire palette (black through white). 256 entries.
static Uint32 fire_color(Uint8 h)
{
    // Piecewise ramp: red rises first, then green, then blue at the top.
    Uint8 r = (Uint8)SDL_min(255, h * 3);
    Uint8 g = (Uint8)SDL_min(255, h > 85 ? (h - 85) * 3 : 0);
    Uint8 b = (Uint8)SDL_min(255, h > 170 ? (h - 170) * 3 : 0);
    return (0xFFu << 24) | (r << 16) | (g << 8) | b;      // ARGB8888
}

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    SDL_Texture  *tex = nullptr;
    std::vector<Uint8> heat;          // FW*FH heat field for the fire
    bool   plasma = false;           // Space toggles fire / plasma
    bool   painting = false;         // mouse held: inject heat
    float  mx = 0, my = 0;           // mouse in internal-texture space
    Uint64 start_ns = 0;
};

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Streaming Textures", "1.0", "com.example.sdl3.streaming");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Streaming Textures & Pixel Effects", WIN_W, WIN_H, 0,
                                     &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);

    st->tex = SDL_CreateTexture(st->renderer, SDL_PIXELFORMAT_ARGB8888,
                                SDL_TEXTUREACCESS_STREAMING, FW, FH);
    SDL_SetTextureScaleMode(st->tex, SDL_SCALEMODE_NEAREST);
    st->heat.assign(FW * FH, 0);
    st->start_ns = SDL_GetTicksNS();
    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE)  st->plasma = !st->plasma;
    }
    if (e->type == SDL_EVENT_MOUSE_BUTTON_DOWN) st->painting = true;
    if (e->type == SDL_EVENT_MOUSE_BUTTON_UP)   st->painting = false;
    if (e->type == SDL_EVENT_MOUSE_MOTION) {
        st->mx = e->motion.x * (float)FW / WIN_W;    // window -> internal texture space
        st->my = e->motion.y * (float)FH / WIN_H;
    }
    return SDL_APP_CONTINUE;
}

// --- Fire: reseed base, propagate + cool upward. ---
static void step_fire(AppState *st)
{
    Uint8 *h = st->heat.data();
    for (int x = 0; x < FW; ++x)
        h[(FH - 1) * FW + x] = 180 + (Uint8)SDL_rand(76);   // hot base row

    if (st->painting) {                                     // inject a hot blob at the cursor
        int cx = (int)st->mx, cy = (int)st->my;
        for (int y = -4; y <= 4; ++y)
            for (int x = -4; x <= 4; ++x) {
                int px = cx + x, py = cy + y;
                if (px >= 0 && px < FW && py >= 0 && py < FH) h[py * FW + px] = 255;
            }
    }
    for (int y = 0; y < FH - 1; ++y)                        // each cell cools its neighbour below
        for (int x = 0; x < FW; ++x) {
            int src_x = (x + (int)SDL_rand(3) - 1 + FW) % FW;
            int v = h[(y + 1) * FW + src_x] - (int)SDL_rand(3);
            h[y * FW + x] = (Uint8)(v < 0 ? 0 : v);
        }
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    float t = (SDL_GetTicksNS() - st->start_ns) / 1.0e9f;

    if (!st->plasma) step_fire(st);

    // ---- Write the whole texture through the locked pointer ----
    void *pixels = nullptr; int pitch = 0;
    if (SDL_LockTexture(st->tex, nullptr, &pixels, &pitch)) {
        for (int y = 0; y < FH; ++y) {
            Uint32 *row = (Uint32 *)((Uint8 *)pixels + y * pitch);   // step by BYTES
            for (int x = 0; x < FW; ++x) {
                if (st->plasma) {
                    // Sum of sines: a smooth, animated colour field.
                    float v = SDL_sinf(x * 0.06f + t)
                            + SDL_sinf(y * 0.08f - t * 1.3f)
                            + SDL_sinf((x + y) * 0.05f + t * 0.7f);
                    Uint8 r = (Uint8)(128 + 127 * SDL_sinf(v * 1.05f));
                    Uint8 g = (Uint8)(128 + 127 * SDL_sinf(v * 1.05f + 2.094f));
                    Uint8 b = (Uint8)(128 + 127 * SDL_sinf(v * 1.05f + 4.188f));
                    row[x] = (0xFFu << 24) | (r << 16) | (g << 8) | b;
                } else {
                    row[x] = fire_color(st->heat[y * FW + x]);
                }
            }
        }
        SDL_UnlockTexture(st->tex);              // pushes the buffer to the GPU
    }

    gfx::clear(st->renderer, 0, 0, 0);
    SDL_RenderTexture(st->renderer, st->tex, nullptr, nullptr);   // stretch to the window
    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        if (st->tex) SDL_DestroyTexture(st->tex);
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Only part of the texture changed? Pass a non-null rect to SDL_LockTexture to lock and upload just that sub-region.
  2. Swap the fire palette for a green "toxic" or blue "cold" ramp by remapping the channels in fire_color.
  3. Compare against SDL_UpdateTexture: fill a scratch array and upload it in one call. Which is clearer? (Locking avoids a second buffer and copy.)
  4. Make the plasma react to the mouse: fold mx/my into the sine arguments so the field warps around the cursor.
