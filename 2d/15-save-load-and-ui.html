Recipe 15 — Save/Load & Simple UI

Persist a high score to the right place on disk, and build clickable buttons with hover and a tiny menu state machine — no UI library, just rectangles and the mouse.

What you'll learn

  • Where save data belongs: SDL_GetPrefPath, not next to the binary.
  • Reading and writing a small file with SDL_LoadFile and SDL_IOStream.
  • An immediate-mode button: hit-test a rectangle against the cursor, react to hover and click.
  • A minimal screen/menu state machine to glue it together.

Prerequisites

Recipe 03 for mouse events and recipe 09 for the bitmap-font draw_text — we reuse it verbatim to label the buttons (its glyph table is elided in the listing below).

Walkthrough

Where does save data go?

Writing next to your executable breaks the moment the app is installed somewhere read-only — which is every real install. SDL hands you the correct per-user, writable directory for your app with SDL_GetPrefPath(org, app). It creates the folder if needed and returns a string you own. 

char *base = SDL_GetPrefPath("ExampleCo", "Recipes");   // e.g. ~/Library/Application Support/...
SDL_snprintf(st->save_path, sizeof(st->save_path), "%ssave.dat", base);
SDL_free(base);

Loading: one call

SDL_LoadFile slurps an entire file into a freshly allocated buffer and tells you the size. A missing file just returns nullptr — that's our "no save yet" case, not an error to panic over. 

size_t len = 0;
void *data = SDL_LoadFile(st->save_path, &len);
if (data && len >= sizeof(int)) SDL_memcpy(&st->hi_score, data, sizeof(int));
SDL_free(data);                                  // free() handles nullptr fine

Saving: open, write, close

For writing we open an SDL_IOStream in binary-write mode, push the bytes, and close it. Here the payload is a single int; for anything structured you'd serialise a struct (mind endianness if the file must be portable across machines). 

static void save_score(const char *path, int score)
{
    SDL_IOStream *io = SDL_IOFromFile(path, "wb");
    if (!io) { SDL_Log("save: %s", SDL_GetError()); return; }
    SDL_WriteIO(io, &score, sizeof(score));
    SDL_CloseIO(io);
}

A button you can click

No widget toolkit required. A button is a rectangle, a label, and two questions each frame: is the cursor inside it (hover), and was it clicked this frame? We track the mouse position from SDL_EVENT_MOUSE_MOTION and set a one-frame clicked flag on SDL_EVENT_MOUSE_BUTTON_DOWN. 

static bool in_rect(float px, float py, const SDL_FRect &r)
{
    return px >= r.x && px < r.x + r.w && py >= r.y && py < r.y + r.h;
}

// Draws the button, returns true if it was clicked this frame.
static bool button(AppState *st, const SDL_FRect &r, const char *label)
{
    bool hot = in_rect(st->mx, st->my, r);
    if (hot) gfx::set_color(st->renderer, 90, 130, 200, 255);
    else     gfx::set_color(st->renderer, 60,  80, 120, 255);
    gfx::fill_rect(st->renderer, r.x, r.y, r.w, r.h);
    draw_text(st->renderer, label, r.x + 16, r.y + 14, 2.0f);
    return hot && st->clicked;
}
This is "immediate mode". The button has no stored object — it's drawn and tested in the same call, every frame. It's the simplest UI style there is, and plenty for menus, debug panels and tools.

Complete listing

A title screen with Play and Reset buttons. Playing bumps the score; beating the stored best saves it to the prefs directory, so it survives a restart. Paste glyph_for / draw_glyph / draw_text from recipe 09 where marked. 

/*
 * 15-save-load-and-ui.cpp — persistent high score + immediate-mode buttons.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include "gfx.hpp"

enum Screen { MENU, PLAYING };

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    Screen screen = MENU;
    int    score = 0;
    int    hi_score = 0;
    float  mx = 0, my = 0;       // cursor position
    bool   clicked = false;      // true for the frame a click happened
    char   save_path[1024] = {0};
};

// ---- bitmap font from recipe 09 (glyph_for / draw_glyph / draw_text) ----
// Paste those three functions here unchanged; elided for length.
static void draw_text(SDL_Renderer *r, const char *s, float x, float y, float scale);

static bool in_rect(float px, float py, const SDL_FRect &r)
{
    return px >= r.x && px < r.x + r.w && py >= r.y && py < r.y + r.h;
}

static bool button(AppState *st, const SDL_FRect &r, const char *label)
{
    bool hot = in_rect(st->mx, st->my, r);
    gfx::set_color(st->renderer, hot ? 90 : 60, hot ? 130 : 80, hot ? 200 : 120, 255);
    gfx::fill_rect(st->renderer, r.x, r.y, r.w, r.h);
    draw_text(st->renderer, label, r.x + 16, r.y + 14, 2.0f);
    return hot && st->clicked;
}

static void load_score(AppState *st)
{
    size_t len = 0;
    void *data = SDL_LoadFile(st->save_path, &len);
    if (data && len >= sizeof(int)) SDL_memcpy(&st->hi_score, data, sizeof(int));
    SDL_free(data);
}

static void save_score(const char *path, int score)
{
    SDL_IOStream *io = SDL_IOFromFile(path, "wb");
    if (!io) { SDL_Log("save: %s", SDL_GetError()); return; }
    SDL_WriteIO(io, &score, sizeof(score));
    SDL_CloseIO(io);
}

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Save/Load & UI", "1.0", "com.example.sdl3.saveui");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Save/Load & UI", 640, 400,
                                     SDL_WINDOW_RESIZABLE, &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);

    char *base = SDL_GetPrefPath("ExampleCo", "Recipes");
    SDL_snprintf(st->save_path, sizeof(st->save_path), "%ssave.dat", base);
    SDL_free(base);
    load_score(st);

    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    switch (e->type) {
    case SDL_EVENT_QUIT: return SDL_APP_SUCCESS;
    case SDL_EVENT_KEY_DOWN:
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (st->screen == PLAYING && e->key.scancode == SDL_SCANCODE_SPACE) {
            st->score += 10;
            if (st->score > st->hi_score) { st->hi_score = st->score; save_score(st->save_path, st->hi_score); }
        }
        break;
    case SDL_EVENT_MOUSE_MOTION:
        st->mx = e->motion.x; st->my = e->motion.y;
        break;
    case SDL_EVENT_MOUSE_BUTTON_DOWN:
        if (e->button.button == SDL_BUTTON_LEFT) {
            st->mx = e->button.x; st->my = e->button.y;
            st->clicked = true;
        }
        break;
    default: break;
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    gfx::clear(st->renderer, 18, 20, 30);

    char buf[64];
    if (st->screen == MENU) {
        draw_text(st->renderer, "RECIPE 15", 40, 40, 4.0f);
        SDL_snprintf(buf, sizeof(buf), "BEST %d", st->hi_score);
        draw_text(st->renderer, buf, 40, 110, 2.0f);

        if (button(st, SDL_FRect{ 40, 170, 200, 56 }, "PLAY"))  { st->screen = PLAYING; st->score = 0; }
        if (button(st, SDL_FRect{ 40, 244, 200, 56 }, "RESET")) {
            st->hi_score = 0; save_score(st->save_path, 0);
        }
    } else {
        draw_text(st->renderer, "SPACE SCORES", 40, 40, 3.0f);
        SDL_snprintf(buf, sizeof(buf), "SCORE %d", st->score);
        draw_text(st->renderer, buf, 40, 110, 2.0f);
        SDL_snprintf(buf, sizeof(buf), "BEST %d", st->hi_score);
        draw_text(st->renderer, buf, 40, 150, 2.0f);
        if (button(st, SDL_FRect{ 40, 220, 200, 56 }, "MENU")) st->screen = MENU;
    }

    st->clicked = false;             // consume the click after the UI saw it
    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Store more than a number. Serialise a small struct (name, score, settings) and version it with a leading magic/version field so old saves migrate.
  2. Add a keyboard fallback for the menu — arrow keys move a highlight, Enter activates — so the UI works without a mouse.
  3. Switch saving to SDL_SaveFile(path, data, size), the one-call counterpart to SDL_LoadFile, and compare.
  4. Make the click "fire on release inside the same button you pressed" instead of on press — the standard, forgiving button behaviour.
