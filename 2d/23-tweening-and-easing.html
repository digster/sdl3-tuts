Recipe 23 — Tweening & Easing

Replace linear motion with personality: a handful of easing functions and one reusable Tween struct give you snappy UI pops, springy menus and weighty movement — animation that accelerates, overshoots and settles instead of sliding mechanically.

What you'll learn

  • What an easing function is: a reshaping of normalized time t ∈ [0,1].
  • The classic family — quad in/out, back (overshoot), elastic (spring) and bounce.
  • A data-driven Tween that hosts any easing via a function pointer.
  • Driving both a comparison strip and a one-shot UI "pop" from the same struct.

Prerequisites

Recipe 04 — Delta Time for the per-frame dt the tween integrates. Everything else here is plain math on top of the gfx:: helpers from recipe 02.

Walkthrough

An easing is a curve from t to t

Linear motion divides distance evenly across time. Easing bends that line: feed in progress t from 0 to 1 and get back a shaped progress. Most stay within [0,1]; a few deliberately don't. 

typedef float (*EaseFn)(float);

static float ease_linear(float t)      { return t; }
static float ease_in_quad(float t)     { return t * t; }                 // slow start
static float ease_out_quad(float t)    { return t * (2.0f - t); }        // slow stop
static float ease_in_out_quad(float t) { return t < 0.5f ? 2*t*t : -1 + (4 - 2*t)*t; }

Overshoot and spring deliberately leave [0,1]

The expressive curves are the ones that exceed their endpoints partway through. back overshoots the target then settles; elastic oscillates like a spring; bounce drops and rebounds. Don't clamp them — the overshoot is the effect. 

static float ease_out_back(float t)
{
    const float s = 1.70158f;          // overshoot amount
    float u = t - 1.0f;
    return 1.0f + (s + 1.0f) * u*u*u + s * u*u;
}

static float ease_out_bounce(float t)
{
    if (t < 1.0f / 2.75f)        return 7.5625f * t * t;
    else if (t < 2.0f / 2.75f)   { t -= 1.5f  / 2.75f; return 7.5625f * t * t + 0.75f; }
    else if (t < 2.5f / 2.75f)   { t -= 2.25f / 2.75f; return 7.5625f * t * t + 0.9375f; }
    else                        { t -= 2.625f/ 2.75f; return 7.5625f * t * t + 0.984375f; }
}

One Tween struct, any curve

Because every easing has the same signature, a single struct can drive all of them. It banks elapsed time, normalizes to [0,1], and applies its function. A loop flag makes it ping-pong forever (for a comparison strip) or run once (for a UI pop). 

struct Tween {
    float  dur;            // seconds for one sweep
    EaseFn ease;
    float  t   = 0.0f;
    int    dir = 1;        // +1 forward, -1 back
    bool   loop = true;

    void update(float dt) {
        t += dir * dt;
        if (t >= dur) { t = dur; if (loop) dir = -1; }
        if (t <= 0.0f) { t = 0.0f; if (loop) dir = +1; }
    }
    float p() const { return ease(t / dur); }   // shaped progress
};

float x = from + (to - from) * tween.p();       // map progress onto anything
Tween values, not just positions. The output is a plain scalar, so the same struct eases colour, alpha, scale, camera zoom, audio gain — anything you can lerp. That generality is why one little struct replaces a pile of bespoke animation code.

Complete listing

Six tracks each show one easing: a faint curve drawn from the function plus a dot driven by a looping Tween, so you can watch the shapes race side by side. Press Space to fire a one-shot ease_out_back pop on the button at the bottom. 

/*
 * 23-tweening-and-easing.cpp — an easing-function library and a reusable Tween.
 */
#define SDL_MAIN_USE_CALLBACKS 1
#include <SDL3/SDL.h>
#include <SDL3/SDL_main.h>
#include "gfx.hpp"

static const int WIN_W = 760, WIN_H = 480;

typedef float (*EaseFn)(float);

static float ease_linear(float t)      { return t; }
static float ease_in_quad(float t)     { return t * t; }
static float ease_out_quad(float t)    { return t * (2.0f - t); }
static float ease_in_out_quad(float t) { return t < 0.5f ? 2*t*t : -1.0f + (4.0f - 2*t)*t; }

static float ease_out_back(float t)
{
    const float s = 1.70158f;
    float u = t - 1.0f;
    return 1.0f + (s + 1.0f) * u*u*u + s * u*u;
}
static float ease_out_elastic(float t)
{
    if (t == 0.0f || t == 1.0f) return t;
    const float p = 0.3f;
    return SDL_powf(2.0f, -10.0f * t) * SDL_sinf((t - p/4.0f) * (2.0f*SDL_PI_F)/p) + 1.0f;
}
static float ease_out_bounce(float t)
{
    if (t < 1.0f / 2.75f)        return 7.5625f * t * t;
    else if (t < 2.0f / 2.75f)   { t -= 1.5f  / 2.75f; return 7.5625f * t * t + 0.75f; }
    else if (t < 2.5f / 2.75f)   { t -= 2.25f / 2.75f; return 7.5625f * t * t + 0.9375f; }
    else                        { t -= 2.625f/ 2.75f; return 7.5625f * t * t + 0.984375f; }
}

struct Tween {
    float  dur;
    EaseFn ease;
    float  t = 0.0f;
    int    dir = 1;
    bool   loop = true;
    void update(float dt) {
        t += dir * dt;
        if (t >= dur)  { t = dur;  if (loop) dir = -1; }
        if (t <= 0.0f) { t = 0.0f; if (loop) dir = +1; }
    }
    float p() const { return ease(t / dur); }
};

struct Row { EaseFn ease; Uint8 r, g, b; };
static const int NROWS = 6;
static const Row ROWS[NROWS] = {
    { ease_linear,       150, 160, 180 },
    { ease_in_out_quad,  120, 200, 255 },
    { ease_in_quad,      120, 230, 150 },
    { ease_out_back,     255, 200,  90 },
    { ease_out_elastic,  230, 120, 200 },
    { ease_out_bounce,   235, 220, 120 },
};

struct AppState {
    SDL_Window   *window = nullptr;
    SDL_Renderer *renderer = nullptr;
    Tween  rows[NROWS];
    Tween  pop{ 0.5f, ease_out_back, 0.5f, 1, false };  // starts settled
    Uint64 last_ns = 0;
};

SDL_AppResult SDL_AppInit(void **appstate, int, char *[])
{
    SDL_SetAppMetadata("Tweening and Easing", "1.0", "com.example.sdl3.tween");
    if (!SDL_Init(SDL_INIT_VIDEO)) { SDL_Log("%s", SDL_GetError()); return SDL_APP_FAILURE; }

    AppState *st = new AppState();
    if (!SDL_CreateWindowAndRenderer("Tweening and Easing", WIN_W, WIN_H, 0,
                                     &st->window, &st->renderer)) {
        SDL_Log("%s", SDL_GetError()); delete st; return SDL_APP_FAILURE;
    }
    SDL_SetRenderVSync(st->renderer, 1);
    for (int i = 0; i < NROWS; ++i) st->rows[i] = Tween{ 1.6f, ROWS[i].ease };
    st->last_ns = SDL_GetTicksNS();
    *appstate = st;
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppEvent(void *appstate, SDL_Event *e)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (e->type == SDL_EVENT_QUIT) return SDL_APP_SUCCESS;
    if (e->type == SDL_EVENT_KEY_DOWN) {
        if (e->key.scancode == SDL_SCANCODE_ESCAPE) return SDL_APP_SUCCESS;
        if (e->key.scancode == SDL_SCANCODE_SPACE) { st->pop.t = 0.0f; st->pop.dir = 1; } // fire one-shot
    }
    return SDL_APP_CONTINUE;
}

SDL_AppResult SDL_AppIterate(void *appstate)
{
    AppState *st = static_cast<AppState *>(appstate);
    Uint64 now = SDL_GetTicksNS();
    float dt = (now - st->last_ns) / 1.0e9f;
    st->last_ns = now;

    for (int i = 0; i < NROWS; ++i) st->rows[i].update(dt);
    st->pop.update(dt);

    gfx::clear(st->renderer, 18, 20, 30);

    const float x0 = 60, trackW = WIN_W - 200.0f, amp = 36.0f, rowH = 60.0f;
    for (int i = 0; i < NROWS; ++i) {
        float baseY = 60.0f + i * rowH;
        // faint easing curve, sampled straight from the function
        gfx::set_color(st->renderer, 56, 62, 84, 255);
        float prevx = x0, prevy = baseY - ROWS[i].ease(0.0f) * amp;
        for (int s = 1; s <= 48; ++s) {
            float t  = s / 48.0f;
            float cx = x0 + t * trackW;
            float cy = baseY - ROWS[i].ease(t) * amp;
            gfx::line(st->renderer, prevx, prevy, cx, cy);
            prevx = cx; prevy = cy;
        }
        // dot driven by the looping Tween
        float pr = st->rows[i].p();
        gfx::set_color(st->renderer, ROWS[i].r, ROWS[i].g, ROWS[i].b, 255);
        gfx::fill_circle(st->renderer, x0 + pr * trackW, baseY - pr * amp, 8);
    }

    // one-shot pop button (scale eased about its centre)
    float scale = 0.7f + 0.45f * st->pop.p();
    float cx = WIN_W * 0.5f, cy = WIN_H - 46.0f, hw = 70.0f * scale, hh = 24.0f * scale;
    gfx::set_color(st->renderer, 90, 140, 240, 255);
    gfx::fill_rect(st->renderer, cx - hw, cy - hh, hw * 2, hh * 2);
    gfx::set_color(st->renderer, 200, 220, 255, 255);
    gfx::rect(st->renderer, cx - hw, cy - hh, hw * 2, hh * 2);

    SDL_RenderPresent(st->renderer);
    return SDL_APP_CONTINUE;
}

void SDL_AppQuit(void *appstate, SDL_AppResult)
{
    AppState *st = static_cast<AppState *>(appstate);
    if (st) {
        SDL_DestroyRenderer(st->renderer);
        SDL_DestroyWindow(st->window);
        delete st;
    }
}

Try it yourself

  1. Add the in variants by mirroring: ease_in_x(t) = 1 - ease_out_x(1 - t). One identity gives you the whole back/elastic/bounce family for free.
  2. Chain tweens: when one finishes, start the next (move, then scale, then fade). A tiny state index over an array of Tweens makes a sequence.
  3. Ease a colour: lerp two SDL_Colors by pop.p() so the button shifts hue as it pops.
  4. Add a delay field to Tween (count dt down before advancing t) to stagger a row of UI elements into a cascade.
